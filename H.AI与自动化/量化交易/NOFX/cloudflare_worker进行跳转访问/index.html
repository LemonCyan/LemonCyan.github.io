export default {
  async fetch(request) {
    //
    // 目标网站（必须修改）
    //
    const TARGET = "https://fapi.binance.com";
  
    // ========== Step 1：构建真实请求 URL ==========
    const url = new URL(request.url);
    const targetUrl = TARGET + url.pathname + url.search;
  
    // ========== Step 2：构建伪装浏览器 Header ==========
    const browserHeaders = new Headers(request.headers);
  
    // 替换 User-Agent（模拟 Chrome）
    browserHeaders.set(
      "User-Agent",
      "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/123.0.0.0 Safari/537.36"
    );

    // 浏览器常用 Accept
    browserHeaders.set(
      "Accept",
"text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8"
    );
  
    // 语言
    browserHeaders.set("Accept-Language", "zh-CN,zh;q=0.9,en;q=0.8");
  
    // 禁止 Cloudflare 自动删除部分 header
    browserHeaders.set("Cache-Control", "no-cache");
  
    // 防止压缩（便于调试）
    browserHeaders.set("Accept-Encoding", "identity");
  
    // 保留 Cookie（如果有）
    if (request.headers.get("Cookie")) {
      browserHeaders.set("Cookie", request.headers.get("Cookie"));
    }
  
    // ========== Step 3：发起请求 ==========
    const response = await fetch(targetUrl, {
      method: request.method,
      headers: browserHeaders,
      body: request.body,
      redirect: "manual", // 我们手动处理重定向
      cf: {
        // Worker 能控制的最强 TLS 设置（仍无法伪造真正浏览器指纹）
        minTLSVersion: "TLSv1.2",
        tlsCipherSuites: [
          "ECDHE-ECDSA-AES128-GCM-SHA256",
          "ECDHE-RSA-AES128-GCM-SHA256",
          "ECDHE-ECDSA-AES256-GCM-SHA384",
          "ECDHE-RSA-AES256-GCM-SHA384"
        ]
      }
    });
  
    // ========== Step 4：手动处理重定向 ==========
    const status = response.status;
  
    // 如果是 301 / 302 / 307 / 308
    if ([301, 302, 303, 307, 308].includes(status)) {
      const location = response.headers.get("Location");
  
      if (!location) {
        return new Response("Redirect received without Location header", {
          status: 500
        });
      }

      // 自动跟随到新网址
      return fetch(location, {
        method: "GET",
        headers: browserHeaders,
        redirect: "manual"
      });
    }
  
    // ========== Step 5：返回目标网站的响应 ==========
    // 保留原始响应头（去掉 Cloudflare 不允许的）
    const newHeaders = new Headers(response.headers);
  
    // 允许跨域（如果你需要）
    newHeaders.set("Access-Control-Allow-Origin", "*");
  
    return new Response(response.body, {
      status: status,
      headers: newHeaders
    });
  }
};

/**
 * Binance API Proxy Worker
 * 用于绕过地理位置限制，代理币安期货 API 请求
 */

export default {
  async fetch(request, env, ctx) {
    // 处理 CORS 预检请求
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        headers: {
          'Access-Control-Allow-Origin': '*',
          'Access-Control-Allow-Methods': 'GET, OPTIONS',
          'Access-Control-Allow-Headers': 'Content-Type, Authorization',
          'Access-Control-Max-Age': '86400',
        },
      });
    }

    // 只允许 GET 请求
    if (request.method !== 'GET') {
      return new Response(JSON.stringify({ 
        error: 'Method not allowed',
        message: 'Only GET requests are supported' 
      }), {
        status: 405,
        headers: {
          'Content-Type': 'application/json',
          'Access-Control-Allow-Origin': '*',
        },
      });
    }

    try {
      // 解析请求 URL
      const url = new URL(request.url);
      
      // 提取币安 API 路径和查询参数
      // 例如: /fapi/v1/klines?symbol=BTCUSDT&interval=15m&limit=100
      const binancePath = url.pathname + url.search;
      
      // 构建币安 API 完整 URL
      const binanceURL = `https://fapi.binance.com${binancePath}`;
      
      // 创建缓存键
      const cacheKey = new Request(binanceURL, {
        method: 'GET',
        headers: request.headers,
      });
      
      // 尝试从缓存获取（使用 Cache API）
      const cache = caches.default;
      const cachedResponse = await cache.match(cacheKey);
      
      if (cachedResponse) {
        // 返回缓存响应（添加 CORS 头）
        const headers = new Headers(cachedResponse.headers);
        headers.set('Access-Control-Allow-Origin', '*');
        headers.set('Access-Control-Allow-Methods', 'GET, OPTIONS');
        headers.set('Access-Control-Allow-Headers', 'Content-Type, Authorization');
        headers.set('X-Cache-Status', 'HIT');
        headers.set('X-Worker-Version', '1.0');
        
        return new Response(cachedResponse.body, {
          status: cachedResponse.status,
          statusText: cachedResponse.statusText,
          headers: headers,
        });
      }

      // 转发请求到币安 API
      const response = await fetch(binanceURL, {
        method: 'GET',
        headers: {
          'User-Agent': 'Mozilla/5.0 (compatible; NOFX/1.0)',
          'Accept': 'application/json',
          'Accept-Language': 'en-US,en;q=0.9',
        },
        // 设置超时（30秒）
        signal: AbortSignal.timeout(30000),
      });

      // 检查响应状态
      if (!response.ok) {
        const errorText = await response.text();
        return new Response(JSON.stringify({
          error: 'Binance API error',
          status: response.status,
          message: errorText,
        }), {
          status: response.status,
          headers: {
            'Content-Type': 'application/json',
            'Access-Control-Allow-Origin': '*',
            'X-Worker-Version': '1.0',
          },
        });
      }

      // 克隆响应以便缓存
      const responseClone = response.clone();
      
      // 确定缓存时间（根据 API 端点）
      let cacheTime = 60; // 默认 60 秒
      if (binancePath.includes('/klines')) {
        // K线数据：根据时间间隔设置缓存
        const interval = url.searchParams.get('interval') || '1m';
        const intervalMinutes = getIntervalMinutes(interval);
        cacheTime = Math.max(30, intervalMinutes); // 至少缓存 30 秒
      } else if (binancePath.includes('/premiumIndex') || binancePath.includes('/openInterest')) {
        cacheTime = 300; // 资金费率等：缓存 5 分钟
      } else if (binancePath.includes('/ticker/price')) {
        cacheTime = 10; // 价格：缓存 10 秒
      }

      // 只缓存成功的响应
      if (response.status === 200) {
        const cacheResponse = new Response(responseClone.body, {
          status: responseClone.status,
          statusText: responseClone.statusText,
          headers: {
            ...responseClone.headers,
            'Cache-Control': `public, max-age=${cacheTime}`,
          },
        });
        
        // 异步缓存响应
        ctx.waitUntil(cache.put(cacheKey, cacheResponse));
      }

      // 返回响应（添加 CORS 头）
      const headers = new Headers(response.headers);
      headers.set('Access-Control-Allow-Origin', '*');
      headers.set('Access-Control-Allow-Methods', 'GET, OPTIONS');
      headers.set('Access-Control-Allow-Headers', 'Content-Type, Authorization');
      headers.set('X-Cache-Status', 'MISS');
      headers.set('X-Worker-Version', '1.0');
      headers.set('X-Cache-Time', cacheTime.toString());

      return new Response(response.body, {
        status: response.status,
        statusText: response.statusText,
        headers: headers,
      });
      
    } catch (error) {
      // 错误处理
      console.error('Worker error:', error);
      
      return new Response(JSON.stringify({ 
        error: 'Internal server error',
        message: error.message,
        type: error.name,
      }), {
        status: 500,
        headers: {
          'Content-Type': 'application/json',
          'Access-Control-Allow-Origin': '*',
          'X-Worker-Version': '1.0',
        },
      });
    }
  },
};

/**
 * 将时间间隔字符串转换为分钟数
 */
function getIntervalMinutes(interval) {
  const map = {
    '1m': 1,
    '3m': 3,
    '5m': 5,
    '15m': 15,
    '30m': 30,
    '1h': 60,
    '2h': 120,
    '4h': 240,
    '6h': 360,
    '8h': 480,
    '12h': 720,
    '1d': 1440,
    '3d': 4320,
    '1w': 10080,
  };
  return map[interval] || 60;
}
