Qt_Build_QMake与CMake对比

QMake 和 CMake 对比及实践指南 #

QMake 和 CMake 都是构建系统，负责将源代码转换为可执行程序。两者的核心作用相同：根据项目配置生成适用于特定编译器的 Makefile。

用一个类比来理解：构建系统类似于作曲家，Makefile 是乐谱，编译器则是演奏乐章的乐队。QMake 和 CMake 就是两位风格不同的作曲家——前者专注于 Qt 领域，后者通用性更强。

一、核心特性对比 #

1.1 适用场景 #

构建系统 最佳使用场景 典型项目类型
QMake 纯 Qt 项目，与 Qt Creator 深度集成 桌面 GUI 应用、Qt Quick 应用
CMake 跨平台混合项目，需要集成第三方库或多语言模块 大型工程、嵌入式系统、跨语言项目

设计动机说明：

  • QMake 由 Qt 团队开发，目标是简化 Qt 项目构建，.pro 文件语法与 Qt 模块命名高度一致
  • CMake 是通用构建系统，支持 C/C++/Fortran 等多种语言，Qt 从 5.x 开始官方支持 CMake，Qt 6 已将 CMake 作为推荐构建系统 [1]

1.2 配置复杂度 #

graph LR
    A[QMake] -->|.pro 文件| B[简洁直观]
    C[CMake] -->|CMakeLists.txt| D[功能强大但学习曲线陡峭]
    B --> E[适合快速原型开发]
    D --> F[适合复杂工程管理]
  • QMake：通过 .pro 文件定义项目，语法简洁，5 分钟即可上手
  • CMake：需要理解 CMakeLists.txt 语法、变量作用域、模块系统，完整掌握需要 1-2 周

1.3 社区与生态 #

维度 QMake CMake
官方支持 Qt 官方维护 Kitware 维护，Qt 官方适配
文档质量 Qt 文档集成，示例丰富 官方参考手册 + 大量社区教程
IDE 支持 Qt Creator 原生支持 CLion、VS Code、Visual Studio 原生支持
第三方库集成 需手动配置路径 通过 find_package 自动定位
✦提示

Qt 6 项目推荐使用 CMake。Qt 官方从 Qt 6 开始已将 CMake 作为默认构建系统，QMake 仅保留兼容支持 [1]。

二、CMake 核心语法详解 #

CMake 通过 CMakeLists.txt 文件描述构建规则。以下是关键命令的详细说明：

2.1 项目定义命令 #

cmake
cmake_minimum_required(VERSION 3.16)  # 指定最低 CMake 版本（Qt5 至少 3.10，Qt6 至少 3.16）
project(MyApp VERSION 1.0.0 LANGUAGES CXX)  # 定义项目名称、版本、语言

2.2 Qt 元对象系统配置 #

Qt 使用 MOC（元对象编译器）、UIC（UI 文件编译器）和 RCC（资源文件编译器）处理特殊文件。CMake 提供自动化支持：

cmake
set(CMAKE_INCLUDE_CURRENT_DIR ON)  # 将当前目录加入 include 路径

set(CMAKE_AUTOMOC ON)  # 自动处理 Q_OBJECT 宏（必需）
set(CMAKE_AUTOUIC ON)  # 自动处理 .ui 文件
set(CMAKE_AUTORCC ON)  # 自动处理 .qrc 资源文件

自定义生成目录（避免污染源文件）：

cmake
set(CMAKE_AUTOUIC_DIR ${CMAKE_BINARY_DIR}/ui)   # .ui 生成的 .h 文件目录
set(CMAKE_AUTOMOC_DIR ${CMAKE_BINARY_DIR}/moc)  # MOC 生成的 .cpp 文件目录
set(CMAKE_AUTORCC_DIR ${CMAKE_BINARY_DIR}/rcc)  # .qrc 生成的 .cpp 文件目录
file(MAKE_DIRECTORY ${CMAKE_AUTOUIC_DIR} ${CMAKE_AUTOMOC_DIR} ${CMAKE_AUTORCC_DIR})
ℹ笔记

CMake 中必须设置 CMAKE_AUTOMOC ON 才能自动处理 Qt 的 MOC 元对象编译器，否则包含 Q_OBJECT 宏的类将无法正确编译。

2.3 目标构建命令 #

添加可执行文件 #

cmake
add_executable(${PROJECT_NAME} main.cpp widget.cpp)

添加库文件 #

cmake
# 静态库
add_library(mylib STATIC lib.cpp)

# 动态库（共享库）
add_library(mylib SHARED lib.cpp)

# 接口库（仅头文件）
add_library(mylib INTERFACE)

添加子目录 #

cmake
add_subdirectory(src)      # 添加 src 目录，查找其中的 CMakeLists.txt
add_subdirectory(plugins)  # 递归处理子项目

2.4 依赖管理命令 #

查找 Qt 模块 #

cmake
# Qt5 项目
find_package(Qt5 COMPONENTS Core Gui Widgets Network REQUIRED)

# Qt5/Qt6 兼容写法
find_package(QT NAMES Qt6 Qt5 COMPONENTS Widgets REQUIRED)
find_package(Qt${QT_VERSION_MAJOR} COMPONENTS Widgets REQUIRED)

链接库文件 #

cmake
target_link_libraries(${PROJECT_NAME} PRIVATE
    Qt5::Core
    Qt5::Gui
    Qt5::Widgets
)

# 或使用版本变量（兼容 Qt5/Qt6）
target_link_libraries(${PROJECT_NAME} PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)

2.5 路径与变量设置 #

cmake
# 设置 C++ 标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 设置输出目录
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
set(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)

# 设置模块搜索路径
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_SOURCE_DIR}/cmake")

# 添加头文件搜索路径（现代 CMake 推荐使用 target_include_directories）
target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_SOURCE_DIR}/include)
❢重要

现代 CMake（3.0+）推荐使用 target_* 系列命令替代全局命令，如用 target_include_directories 替代 include_directories，以实现更好的目标隔离 [2]。

三、从 QMake 迁移到 CMake #

3.1 典型 QMake 项目文件 #

# test.pro
QT += core network
QT -= gui

CONFIG += c++11 console
CONFIG -= app_bundle

TARGET = test
TEMPLATE = app

SOURCES += \
    main.cpp \
    test_interface.cpp \
    motomanlibrary.cpp \
    processing.cpp

HEADERS += \
    test_interface.h \
    motomanlibrary.h \
    processing.h

3.2 等效 CMakeLists.txt #

cmake
cmake_minimum_required(VERSION 3.16)
project(test VERSION 1.0.0 LANGUAGES CXX)

# ============ Qt 配置 ============
set(CMAKE_INCLUDE_CURRENT_DIR ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

find_package(Qt5 COMPONENTS Core Network REQUIRED)

# ============ C++ 标准 ============
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ============ 源文件定义 ============
set(SOURCES
    main.cpp
    test_interface.cpp
    motomanlibrary.cpp
    processing.cpp
)

set(HEADERS
    test_interface.h
    motomanlibrary.h
    processing.h
)

# ============ 构建目标 ============
add_executable(${PROJECT_NAME} ${SOURCES} ${HEADERS})

target_link_libraries(${PROJECT_NAME} PRIVATE
    Qt5::Core
    Qt5::Network
)

# ============ 输出路径 ============
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)

3.3 转换对照表 #

QMake 语法 CMake 等效写法
QT += core network find_package(Qt5 COMPONENTS Core Network REQUIRED)
CONFIG += c++11 set(CMAKE_CXX_STANDARD 11)
TARGET = test project(test)
TEMPLATE = app add_executable(...)
TEMPLATE = lib add_library(...)
SOURCES += file.cpp set(SOURCES file.cpp)
INCLUDEPATH += path target_include_directories(target PRIVATE path)
LIBS += -lmylib target_link_libraries(target PRIVATE mylib)

四、编译与构建流程 #

4.1 环境准备 #

Windows 环境变量配置 #

将 Qt 的 bin 目录添加到系统 PATH，确保命令行工具可用：

C:\Qt\5.15.0\msvc2019_64\bin

MinGW 编译器配置 #

使用 MinGW 时需在 CMake 中指定编译器路径：

cmake
# 必须在 project() 之前设置
set(CMAKE_PREFIX_PATH "D:/Qt/Qt5.14.2/5.14.2/mingw73_64")
set(CMAKE_MAKE_PROGRAM "D:/Qt/Qt5.14.2/Tools/mingw730_64/bin/mingw32-make.exe")
set(CMAKE_C_COMPILER "D:/Qt/Qt5.14.2/Tools/mingw730_64/bin/gcc.exe")
set(CMAKE_CXX_COMPILER "D:/Qt/Qt5.14.2/Tools/mingw730_64/bin/g++.exe")

或通过命令行参数指定：

bash
cmake . -G "MinGW Makefiles" \
  -DCMAKE_C_COMPILER="D:/Qt/Qt5.14.2/Tools/mingw730_64/bin/gcc.exe" \
  -DCMAKE_CXX_COMPILER="D:/Qt/Qt5.14.2/Tools/mingw730_64/bin/g++.exe" \
  -DCMAKE_PREFIX_PATH="D:/Qt/Qt5.14.2/5.14.2/mingw73_64" \
  -DCMAKE_MAKE_PROGRAM="D:/Qt/Qt5.14.2/Tools/mingw730_64/bin/mingw32-make.exe"

4.2 标准构建步骤 #

bash
# 1. 创建构建目录（推荐 out-of-source 构建）
mkdir build && cd build

# 2. 配置项目（生成 Makefile）
cmake ..

# 3. 编译项目
cmake --build . --parallel 4  # 使用 4 个并行进程

# 或使用 make（仅限 Unix 系统）
make -j4

4.3 指定生成器 #

bash
# 使用 Ninja 生成器（构建速度更快）
cmake -G Ninja ..

# 使用 Visual Studio 生成器（Windows）
cmake -G "Visual Studio 17 2022" -A x64 ..

# 使用 MinGW Makefiles（Windows）
cmake -G "MinGW Makefiles" ..

4.4 常用配置选项 #

bash
# 指定构建类型
cmake -DCMAKE_BUILD_TYPE=Release ..

# 指定 Qt 安装路径（当系统存在多个 Qt 版本时）
cmake -DCMAKE_PREFIX_PATH=/opt/Qt/5.15.2/gcc_64 ..

# 指定安装目录
cmake -DCMAKE_INSTALL_PREFIX=/usr/local ..

五、IDE 集成 #

5.1 VS Code 集成 #

必需扩展 #

  1. C/C++：提供代码高亮、智能感知、调试支持
  2. CMake：CMakeLists.txt 语法高亮
  3. CMake Tools：项目配置、构建和调试集成

配置步骤 #

  1. 打开项目文件夹（包含 CMakeLists.txt）
  2. VS Code 自动检测 CMake 项目，选择 CMake Kit（推荐 Visual Studio 2019/2022）
  3. 按 F7 构建项目，按 Shift+F5 运行，按 Ctrl+F5 调试

智能感知问题修复 #

Qt 生成的 .ui 文件对应的头文件可能未被自动包含在 IntelliSense 路径中，需手动添加：

cmake
target_include_directories(${PROJECT_NAME} PRIVATE 
    "${CMAKE_BINARY_DIR}/${PROJECT_NAME}_autogen/include_Debug"
)

5.2 Qt Creator 集成 #

Qt Creator 原生支持 CMake 项目：

  1. 选择 文件 → 打开文件或项目，选择 CMakeLists.txt
  2. 选择构建套件（Kit），点击 配置项目
  3. 使用标准的构建和运行按钮

六、完整项目模板 #

6.1 Qt5 Widgets 项目（兼容 Qt6） #

cmake
cmake_minimum_required(VERSION 3.16)
project(MyQtApp VERSION 1.0.0 LANGUAGES CXX)

# ============ Qt 配置 ============
set(CMAKE_INCLUDE_CURRENT_DIR ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 兼容 Qt5 和 Qt6
find_package(QT NAMES Qt6 Qt5 COMPONENTS Widgets REQUIRED)
find_package(Qt${QT_VERSION_MAJOR} COMPONENTS Widgets REQUIRED)

# ============ 源文件 ============
set(SOURCES
    main.cpp
    mainwindow.cpp
    mainwindow.h
    mainwindow.ui
)

# ============ 构建目标 ============
if(ANDROID)
    add_library(${PROJECT_NAME} SHARED ${SOURCES})
else()
    add_executable(${PROJECT_NAME} ${SOURCES})
endif()

target_link_libraries(${PROJECT_NAME} PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)

# ============ 输出路径 ============
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)

七、选型建议 #

flowchart TD
    A[新建 Qt 项目] --> B{项目规模?}
    B -->|小型/原型| C[QMake]
    B -->|中大型| D{是否需要集成非 Qt 库?}
    D -->|否| E[QMake 或 CMake 均可]
    D -->|是| F[CMake]
    C --> G[快速启动，Qt Creator 最佳体验]
    F --> H[统一构建系统，便于持续集成]
    
    I[已有 Qt 项目] --> J{当前使用?}
    J -->|QMake| K{是否计划升级 Qt 6?}
    K -->|是| L[迁移到 CMake]
    K -->|否| M[继续使用 QMake]
    J -->|CMake| N[继续使用 CMake]

总结：

  1. 新项目：优先选择 CMake，尤其是计划使用 Qt 6 的项目
  2. 现有 QMake 项目：若无迁移压力可继续使用，升级 Qt 6 时再迁移
  3. 混合项目：必须使用 CMake，可通过 add_subdirectory 集成各模块
