Qt之QTreeView（一）自定义树形模型（QAbstractItemModel） #

在 Qt 开发中，针对多级嵌套、分层结构的层级数据，QTreeView 是最合适也是性能最好的展示控件。与直接内置模型的 QTreeWidget 相比，使用自定义模型（QAbstractItemModel）配合 QTreeView 可以极大地优化大数据集下的渲染效率，同时更利于业务数据的深度解耦。

本文将演示如何从零构建自定义的树形数据节点和数据模型。

一、 核心概念梳理：Model/View #

在进行自定义树模型开发前，需要先理清以下关系：

  • 便利窗口部件（如 QTreeWidget）：内置了现成的树形数据结构，适用于几十到数百个节点的轻量级场景。
  • 模型/视图部件（如 QTreeView）：视图不持有任何数据，仅负责绘制。数据模型（Model）扮演媒介，根据视图的索引请求（QModelIndex）返回数据。适用于大数据流，性能上限极高。

为了实现自定义树模型，我们需要编写两个核心类：

  1. TreeItem：纯 C++ 类，代表树形结构中的每个“节点”。
  2. TreeModel：继承自 QAbstractItemModel，实现虚函数，将 TreeItem 节点的关系映射成 Qt 框架能理解的坐标索引。

二、 设计树节点类 (TreeItem) #

每个节点都需要记录自己的父节点指针、子节点列表，以及该行所代表的多列数据。

1. treeitem.h #

cpp
#pragma once
#include <QList>
#include <QVariant>

class TreeItem
{
public:
    explicit TreeItem(const QList<QVariant> &data, TreeItem *parent = nullptr);
    ~TreeItem();

    void appendChild(TreeItem *child); // 添加子节点
    TreeItem *child(int row);          // 获取指定索引处的子节点
    int childCount() const;            // 获取子节点总数
    int columnCount() const;           // 获取数据列数
    QVariant data(int column) const;   // 获取某一列的数据
    TreeItem *parent();                // 获取父节点指针
    int row() const;                   // 获取该节点在父节点子列表中的行索引

private:
    QList<TreeItem*> childItems;       // 子节点列表指针
    QList<QVariant> itemData;          // 当前行的各列数据
    TreeItem *parentItem;              // 父节点指针
};

2. treeitem.cpp #

cpp
#include "treeitem.h"

TreeItem::TreeItem(const QList<QVariant> &data, TreeItem *parent)
{
    parentItem = parent;
    itemData = data;
}

TreeItem::~TreeItem()
{
    // 级联释放所有子节点内存，防止内存泄漏
    qDeleteAll(childItems);
}

void TreeItem::appendChild(TreeItem *item)
{
    childItems.append(item);
}

TreeItem *TreeItem::child(int row)
{
    return childItems.value(row);
}

int TreeItem::childCount() const
{
    return childItems.count();
}

int TreeItem::columnCount() const
{
    return itemData.count();
}

QVariant TreeItem::data(int column) const
{
    return itemData.value(column);
}

TreeItem *TreeItem::parent()
{
    return parentItem;
}

int TreeItem::row() const
{
    if (parentItem)
    {
        // 查找自身在父节点 childItems 中的位置作为行号
        return parentItem->childItems.indexOf(const_cast<TreeItem*>(this));
    }
    return 0;
}

三、 实现自定义模型 (TreeModel) #

继承 QAbstractItemModel 时，必须重写以下 6 个基础虚函数，以便视图组件（QTreeView）可以通过 QModelIndex 进行层级遍历。

1. treemodel.h #

cpp
#pragma once
#include <QAbstractItemModel>
#include <QModelIndex>
#include <QVariant>

class TreeItem;

class TreeModel : public QAbstractItemModel
{
    Q_OBJECT

public:
    explicit TreeModel(QObject *parent = nullptr);
    ~TreeModel();

    // 必须重写的 Model 接口虚函数
    QVariant data(const QModelIndex &index, int role) const override;
    Qt::ItemFlags flags(const QModelIndex &index) const override;
    QVariant headerData(int section, Qt::Orientation orientation, int role = Qt::DisplayRole) const override;
    QModelIndex index(int row, int column, const QModelIndex &parent = QModelIndex()) const override;
    QModelIndex parent(const QModelIndex &index) const override;
    int rowCount(const QModelIndex &parent = QModelIndex()) const override;
    int columnCount(const QModelIndex &parent = QModelIndex()) const override;

    // 自定义初始化数据加载接口
    void setXML(const QString &xmlFile);

private:
    void setupModelData(TreeItem *parent);
    TreeItem *findTargetItem(TreeItem* item, int parentId);

    TreeItem *rootItem; // 树结构的虚根节点
};

2. treemodel.cpp #

cpp
#include "treemodel.h"
#include "treeitem.h"

TreeModel::TreeModel(QObject *parent)
    : QAbstractItemModel(parent)
{
    QList<QVariant> rootData;
    rootData << "标题"; // 设定表头文本
    rootItem = new TreeItem(rootData);
}

TreeModel::~TreeModel()
{
    delete rootItem;
}

// 1. 获取树中指定位置节点的数据列数
int TreeModel::columnCount(const QModelIndex &parent) const
{
    if (parent.isValid())
        return static_cast<TreeItem*>(parent.internalPointer())->columnCount();
    else
        return rootItem->columnCount();
}

// 2. 根据 row/column/parent 创建 QModelIndex 实例对象，关键是将底层的 TreeItem 绑定为 internalPointer
QModelIndex TreeModel::index(int row, int column, const QModelIndex &parent) const
{
    if (!hasIndex(row, column, parent))
        return QModelIndex();

    TreeItem *parentItem;
    if (!parent.isValid())
        parentItem = rootItem;
    else
        parentItem = static_cast<TreeItem*>(parent.internalPointer());

    TreeItem *childItem = parentItem->child(row);
    if (childItem)
        return createIndex(row, column, childItem);
    else
        return QModelIndex();
}

// 3. 根据给定的子节点 ModelIndex，逆向查找获取其父节点 ModelIndex
QModelIndex TreeModel::parent(const QModelIndex &index) const
{
    if (!index.isValid())
        return QModelIndex();

    TreeItem *childItem = static_cast<TreeItem*>(index.internalPointer());
    TreeItem *parentItem = childItem->parent();

    if (parentItem == rootItem)
        return QModelIndex(); // 顶级节点的父节点在 View 看来是无效的 QModelIndex()

    return createIndex(parentItem->row(), 0, parentItem);
}

// 4. 获取子节点行数
int TreeModel::rowCount(const QModelIndex &parent) const
{
    TreeItem *parentItem;
    if (parent.column() > 0)
        return 0;

    if (!parent.isValid())
        parentItem = rootItem;
    else
        parentItem = static_cast<TreeItem*>(parent.internalPointer());

    return parentItem->childCount();
}

// 5. 根据 ModelIndex 读取具体的角色数据（DisplayRole 用于显示文本，DecorationRole 用于图标）
QVariant TreeModel::data(const QModelIndex &index, int role) const
{
    if (!index.isValid())
        return QVariant();

    TreeItem *item = static_cast<TreeItem*>(index.internalPointer());
    
    // 假设从我们的本地结构中找到了对应的 ToolUtil 业务数据
    if (role == Qt::DisplayRole)
    {
        return item->data(index.column()); // 返回节点名称或列内容
    }
    else if (role == Qt::DecorationRole && index.column() == 0)
    {
        // 渲染树节点前的装饰图标
        return QIcon(":/Resources/Images/folderIcon.png");
    }
    else if (role == Qt::ToolTipRole)
    {
        return QString("节点提示信息");
    }
    
    return QVariant();
}

// 6. 设置节点的属性标志（只读、使能、可拖拽等）
Qt::ItemFlags TreeModel::flags(const QModelIndex &index) const
{
    if (!index.isValid())
        return Qt::NoItemFlags;

    return QAbstractItemModel::flags(index);
}

// 7. 读取表头名称
QVariant TreeModel::headerData(int section, Qt::Orientation orientation, int role) const
{
    if (orientation == Qt::Horizontal && role == Qt::DisplayRole)
        return rootItem->data(section);

    return QVariant();
}

通过完成以上方法，将其绑定到视图 ui->treeView->setModel(yourTreeModel) 上，一个结构完整的树形图表便可以高性能地呈现在界面上了。
