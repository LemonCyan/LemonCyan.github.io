Qt使用QCamera多媒体模块调用摄像头与截图保存实现 #

在客户端开发中，调用本地摄像头硬件实现视频预览、录像或者进行抓拍截图是常见的多媒体需求。

在 Qt 5 中，Qt Multimedia（多媒体）模块提供了非常健全的类库支持。主要涉及以下三个核心类：

  1. QCamera：摄像物理设备，控制硬件曝光、对焦及捕获模式。
  2. QCameraViewfinder：取景渲染器，作为一个独立的 QWidget 挂载在界面上展示实时视频流。
  3. QCameraImageCapture：截图抓拍器，负责截取当前帧画面并触发图片保存动作。

下面提供一个可以直接编译运行的完整摄像头拍照程序示例。

一、 项目环境配置 #

在使用多媒体模块前，必须在项目的 .pro 配置文件中声明依赖：

# 确保在项目 pro 文件中追加
QT += multimedia multimediawidgets

二、 头文件声明 (CameraDialog.h) #

cpp
#pragma once
#include <QDialog>
#include <QCamera>
#include <QCameraViewfinder>
#include <QCameraImageCapture>
#include <QPushButton>
#include <QLabel>

class CameraDialog : public QDialog
{
    Q_OBJECT
public:
    explicit CameraDialog(QWidget *parent = nullptr);
    ~CameraDialog();

private slots:
    // 捕获一帧并触发 imageCaptured 信号
    void slotCaptureImage();
    
    // 接收捕获完成的图片并显示在右上角预览区域
    void slotDisplayImage(int id, const QImage &image);
    
    // 将预览区域的图片保存至本地磁盘
    void slotSaveImage();

private:
    void initUI();

private:
    // 💡 核心多媒体指针
    QCamera             *m_camera;
    QCameraViewfinder   *m_viewFinder;
    QCameraImageCapture *m_imageCapture;

    // UI 组件
    QLabel              *m_previewLabel;
    QPushButton         *m_captureButton;
    QPushButton         *m_saveButton;
    QPushButton         *m_exitButton;
};

三、 源文件实现 (CameraDialog.cpp) #

cpp
#include "CameraDialog.h"
#include <QHBoxLayout>
#include <QVBoxLayout>
#include <QFileDialog>
#include <QMessageBox>

CameraDialog::CameraDialog(QWidget *parent)
    : QDialog(parent)
{
    this->resize(700, 450);
    this->setWindowTitle(tr("摄像头采集测试"));

    initUI();

    // 1. 实例化摄像头设备（默认使用系统主摄像头）
    m_camera = new QCamera(this);

    // 2. 实例化取景器控件并将其与摄像头关联
    m_viewFinder = new QCameraViewfinder(this);
    m_camera->setViewfinder(m_viewFinder);

    // 3. 实例化抓拍器并绑定目标摄像头设备
    m_imageCapture = new QCameraImageCapture(m_camera, this);

    // 4. 配置摄像头为静态照片捕获模式
    m_camera->setCaptureMode(QCamera::CaptureStillImage);
    
    // 5. 设置抓拍的输出目的地为文件（可同时输出到文件及缓冲区）
    m_imageCapture->setCaptureDestination(QCameraImageCapture::CaptureToFile);

    // 6. 关联信号与槽
    connect(m_captureButton, &QPushButton::clicked, this, &CameraDialog::slotCaptureImage);
    connect(m_saveButton, &QPushButton::clicked, this, &CameraDialog::slotSaveImage);
    connect(m_exitButton, &QPushButton::clicked, this, &CameraDialog::close);
    
    // 当截图完成时，抓拍器会发射 imageCaptured 信号，返回预览图像
    connect(m_imageCapture, &QCameraImageCapture::imageCaptured, this, &CameraDialog::slotDisplayImage);

    // 7. 布局组装
    QHBoxLayout *mainLayout = new QHBoxLayout(this);
    QVBoxLayout *rightLayout = new QVBoxLayout();

    rightLayout->addWidget(m_previewLabel);
    rightLayout->addStretch();
    rightLayout->addWidget(m_captureButton);
    rightLayout->addWidget(m_saveButton);
    rightLayout->addWidget(m_exitButton);

    mainLayout->addWidget(m_viewFinder, 1); // 视频画面占据大部空间
    mainLayout->addLayout(rightLayout);

    // 8. 启动摄像头采集流
    m_camera->start();
}

CameraDialog::~CameraDialog()
{
    m_camera->stop();
}

void CameraDialog::initUI()
{
    m_previewLabel = new QLabel(this);
    m_previewLabel->setFixedSize(160, 120);
    m_previewLabel->setFrameShape(QFrame::Box);
    m_previewLabel->setScaledContents(true); // 让图片自适应缩放填满 Label
    m_previewLabel->setStyleSheet("background-color: black;");

    m_captureButton = new QPushButton(tr("拍照"), this);
    m_saveButton = new QPushButton(tr("保存"), this);
    m_exitButton = new QPushButton(tr("退出"), this);
}

void CameraDialog::slotCaptureImage()
{
    // 调用此函数触发单帧截图
    m_imageCapture->capture();
}

void CameraDialog::slotDisplayImage(int id, const QImage &image)
{
    Q_UNUSED(id);
    // 将 QImage 转换为 QPixmap 并渲染在右侧预览区域
    m_previewLabel->setPixmap(QPixmap::fromImage(image));
}

void CameraDialog::slotSaveImage()
{
    const QPixmap *pixmap = m_previewLabel->pixmap();
    if (!pixmap || pixmap->isNull())
    {
        QMessageBox::warning(this, tr("警告"), tr("预览区无图片，请先拍照！"));
        return;
    }

    // 弹出文件保存对话框供用户选择路径
    QString savePath = QFileDialog::getSaveFileName(this, 
                                                    tr("保存图片"), 
                                                    QDir::currentPath(), 
                                                    tr("Images (*.jpg *.png *.bmp)"));
    if (!savePath.isEmpty())
    {
        if (pixmap->save(savePath))
        {
            QMessageBox::information(this, tr("成功"), tr("图片保存成功！"));
        }
        else
        {
            QMessageBox::critical(this, tr("错误"), tr("图片保存失败，请检查写入权限。"));
        }
    }
}
