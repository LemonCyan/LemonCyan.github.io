Qt模型视图架构实战：自定义进度条委托与表格数据实时动态更新 #

在客户端开发（如文件传输工具、任务管理器）中，表格数据往往需要实时动态变化（例如文件传输进度、速度、剩余时间）。

要实现此类复杂表格，Qt 标准的 QTableWidget 显得力不存心。最专业的方法是采用 Model/View (模型/视图) 架构：

  1. Model：继承 QAbstractTableModel，维护底层数据链表并对视图提供读写标准接口。
  2. View：使用 QTableView 负责行高、列宽和多选交互。
  3. Delegate：继承 QItemDelegate（或 QStyledItemDelegate），接管指定列的绘制（例如直接用 QStyle 绘制一个进度条）。

一、 自定义进度条委托 (ProgressBarDelegate) #

默认情况下，表格单元格只能展示普通文本。为了在“进度”列绘制一个真实的进度条，我们需要重写委托的 paint 函数，利用系统的 QStyle 渲染引擎动态绘制进度条控件：

头文件声明： #

cpp
#pragma once
#include <QItemDelegate>

class ProgressBarDelegate : public QItemDelegate
{
    Q_OBJECT
public:
    explicit ProgressBarDelegate(QObject *parent = nullptr);

    // 💡 接管绘制事件
    void paint(QPainter *painter, 
               const QStyleOptionViewItem &option, 
               const QModelIndex &index) const override;
};

源文件实现： #

cpp
#include "ProgressBarDelegate.h"
#include <QApplication>
#include <QStyleOptionProgressBar>

ProgressBarDelegate::ProgressBarDelegate(QObject *parent)
    : QItemDelegate(parent)
{
}

void ProgressBarDelegate::paint(QPainter *painter, 
                                 const QStyleOptionViewItem &option, 
                                 const QModelIndex &index) const
{
    // 💡 针对第三列 (Column Index = 2) 绘制进度条
    if (index.column() == 2)
    {
        // 从 model 的 DisplayRole 中获取整型数据
        int progress = index.model()->data(index, Qt::DisplayRole).toInt();

        // 构造 QStyleOptionProgressBar 结构体
        QStyleOptionProgressBar progressBarOption;
        progressBarOption.state = QStyle::State_Enabled;
        progressBarOption.direction = QApplication::layoutDirection();
        progressBarOption.rect = option.rect; // 进度条覆盖当前单元格区域
        progressBarOption.fontMetrics = QApplication::fontMetrics();
        progressBarOption.minimum = 0;
        progressBarOption.maximum = 100;
        progressBarOption.textAlignment = Qt::AlignCenter;
        progressBarOption.textVisible = true;
        progressBarOption.progress = progress;
        progressBarOption.text = QString("%1%").arg(progress);

        // 💡 核心：利用当前操作系统的扁平化样式绘制进度条
        QApplication::style()->drawControl(QStyle::CE_ProgressBar, &progressBarOption, painter);
    }
    else
    {
        // 其它普通文本列交给父类默认处理
        QItemDelegate::paint(painter, option, index);
    }
}

二、 构建自定义表格数据模型 (TableModel) #

数据模型用于封装真正的内存数组 QList<QStringList>，并向 QTableView 汇报行数、列数、以及单元格数据。

头文件声明： #

cpp
#pragma once
#include <QAbstractTableModel>
#include <QStringList>
#include <QList>

class TableModel : public QAbstractTableModel
{
    Q_OBJECT
public:
    explicit TableModel(QObject *parent = nullptr);
    ~TableModel();

    // 💡 必须重写的核心虚函数
    int rowCount(const QModelIndex &parent = QModelIndex()) const override;
    int columnCount(const QModelIndex &parent = QModelIndex()) const override;
    QVariant data(const QModelIndex &index, int role = Qt::DisplayRole) const override;
    QVariant headerData(int section, Qt::Orientation orientation, int role = Qt::DisplayRole) const override;
    Qt::ItemFlags flags(const QModelIndex &index) const override;

    // 配置数据接口
    void setModelData(QList<QStringList> *dataList);
    void setHorizontalHeaders(const QStringList &headers);
    
    // 刷新模型视图
    void refreshModel();

signals:
    void updateCount(int count);

private:
    QList<QStringList>  *m_rowList;      // 指向真实数据的外部指针
    QStringList          m_headerList;   // 标题头列表
};

源文件实现： #

cpp
#include "TableModel.h"

TableModel::TableModel(QObject *parent)
    : QAbstractTableModel(parent), m_rowList(nullptr)
{
}

TableModel::~TableModel()
{
}

void TableModel::setModelData(QList<QStringList> *dataList)
{
    m_rowList = dataList;
}

void TableModel::setHorizontalHeaders(const QStringList &headers)
{
    m_headerList = headers;
}

int TableModel::rowCount(const QModelIndex &parent) const
{
    Q_UNUSED(parent)
    return m_rowList ? m_rowList->size() : 0;
}

int TableModel::columnCount(const QModelIndex &parent) const
{
    Q_UNUSED(parent)
    if (m_rowList && m_rowList->size() > 0)
        return m_rowList->at(0).size();
    return m_headerList.size();
}

QVariant TableModel::data(const QModelIndex &index, int role) const
{
    if (!index.isValid() || !m_rowList)
        return QVariant();

    int row = index.row();
    int col = index.column();

    if (row >= m_rowList->size() || col >= m_rowList->at(0).size())
        return QVariant();

    if (role == Qt::DisplayRole)
    {
        return m_rowList->at(row).at(col);
    }
    else if (role == Qt::TextAlignmentRole)
    {
        return int(Qt::AlignLeft | Qt::AlignVCenter); // 靠左垂直居中
    }
    return QVariant();
}

QVariant TableModel::headerData(int section, Qt::Orientation orientation, int role) const
{
    if (role == Qt::DisplayRole && orientation == Qt::Horizontal)
    {
        if (section < m_headerList.size())
            return m_headerList.at(section);
    }
    return QVariant();
}

Qt::ItemFlags TableModel::flags(const QModelIndex &index) const
{
    if (!index.isValid())
        return Qt::NoItemFlags;
    return Qt::ItemIsEnabled | Qt::ItemIsSelectable; // 只读、可选择
}

void TableModel::refreshModel()
{
    // 通知视图模型已被重置，强制重新绘制整张表格
    beginResetModel();
    endResetModel();
    emit updateCount(rowCount());
}

三、 表格视图实现与多行反向删除 (TableView) #

TableView 承载真正的 UI 并负责管理底层数据链表 QList<QStringList>：

cpp
#include <QTableView>
#include <QHeaderView>
#include <QtAlgorithms>

class TableView : public QTableView
{
    Q_OBJECT
public:
    TableView(QWidget *parent = nullptr) : QTableView(parent) {
        // UI 配置
        this->setAlternatingRowColors(true);
        this->setSelectionBehavior(QAbstractItemView::SelectRows);
        this->horizontalHeader()->setStretchLastSection(true);
        this->verticalHeader()->setVisible(false);
        this->setShowGrid(false);
        this->setEditTriggers(QAbstractItemView::NoEditTriggers);
        this->setSelectionMode(QAbstractItemView::ExtendedSelection);

        // 组装 Model 与 Delegate
        m_model = new TableModel(this);
        this->setModel(m_model);
        
        m_delegate = new ProgressBarDelegate(this);
        this->setItemDelegate(m_delegate);

        // 初始化表头
        QStringList header;
        header << tr("任务名") << tr("大小") << tr("进度") << tr("速度") << tr("剩余时间") << tr("状态");
        m_model->setHorizontalHeaders(header);
        m_model->setModelData(&m_dataList);
    }

    void addRow(const QStringList &row) {
        m_dataList.append(row);
        m_model->refreshModel();
    }

    // 💡 优雅地删除多行选中项
    void removeSelectedRows() {
        QModelIndexList selectedIdx = this->selectedIndexes();
        if (selectedIdx.isEmpty()) return;

        // 获取不重复的行号列表
        QList<int> rowsToDelete;
        foreach (const QModelIndex &index, selectedIdx) {
            if (!rowsToDelete.contains(index.row())) {
                rowsToDelete.append(index.row());
            }
        }

        // 从大到小排序，防止由于前面元素被删除导致后面索引发生漂移
        std::sort(rowsToDelete.begin(), rowsToDelete.end(), std::greater<int>());

        foreach (int row, rowsToDelete) {
            m_dataList.removeAt(row);
        }
        m_model->refreshModel();
    }

private:
    QList<QStringList>    m_dataList;
    TableModel           *m_model;
    ProgressBarDelegate  *m_delegate;
};

四、 💡 核心进阶：如何解决更新时选中项丢失的问题？ #

在上述代码中，当我们实时更新下载进度，调用 m_model->refreshModel()：

  • 内部执行了 beginResetModel() 和 endResetModel()。
  • 这会导致表格刷新得非常生硬，并且会瞬间清除用户当前在表格上用鼠标选中的行焦点。

优化方案：使用 dataChanged 信号进行局部更新 #

如果你只是需要实时修改某些行的进度或速度数值，千万不要重置整个模型，直接修改数据链表，并通知视图重新渲染局部单元格即可：

cpp
void TableModel::updateProgress(int row, int progressValue)
{
    if (row < 0 || row >= m_rowList->size()) return;

    // 修改内存数组中的对应行数据
    QStringList &rowList = (*m_rowList)[row];
    rowList[2] = QString::number(progressValue); // 进度列

    // 💡 构造发生变更的单元格索引 (第 row 行，第 2 列)
    QModelIndex changedIndex = createIndex(row, 2);

    // 💡 只触发该单元格的更新重绘，保留用户的滚动条位置和行选中状态
    emit dataChanged(changedIndex, changedIndex, {Qt::DisplayRole});
}

通过使用局部 dataChanged 代替全局重置，程序在处理数千行高频更新时能够极大降低 CPU 开销，且完全不影响用户拖拽和选中单元格的交互操作。
