Qt之自定义提示框（类似QMessageBox） #

在 Qt 中经常会用到提示框来引导用户进行交互操作。虽然内置的 QMessageBox 非常方便，但其控件类型、大小、布局以及默认样式往往无法满足特定的 UI 规划。如果直接把未做过样式处理的默认弹出框交给用户，整体界面格调会有所脱节。

普通的 QMessageBox 初始化效果十分粗糙，也无法随意更改边框样式和控件细节。因此，本文将详细介绍如何通过继承 QDialog 自定义一个精致、功能丰富的模态提示框，支持无边框窗口移动、背景图片绘制以及圆角裁剪等功能。

一、 自定义提示框头文件 (msg_box.h) #

下面是自定义提示框类 MsgBox 的核心声明：

cpp
#ifndef MSG_BOX_H
#define MSG_BOX_H

#include <QDialog>
#include <QLabel>
#include <QPushButton>
#include <QCheckBox>
#include <QMouseEvent>
#include <QPaintEvent>
#include <QPainter>
#include <QBitmap>

// 假设 PushButton 是已经自定义实现的可以加载图标资源的按钮类
#include "push_button.h" 

class MsgBox : public QDialog
{
    Q_OBJECT

public:
    explicit MsgBox(QWidget *parent = nullptr);
    ~MsgBox() = default;

    // 设置提示框的标题、文本、图片、以及隐藏复选框和确认按钮的配置
    void setInfo(QString title_info, QString info, QPixmap pixmap, 
                 bool is_check_hidden = true, bool is_ok_hidden = false);

signals:
    // 复选框被勾选时的信号，携带勾选状态与确认/取消按钮的判别
    void msgChecked(bool is_checked, bool is_ok);
    void okMessageHidden(bool is_checked);

private slots:
    void okOperate();
    void cancelOperate();

protected:
    void paintEvent(QPaintEvent *event) override;
    void mousePressEvent(QMouseEvent *event) override;
    void mouseReleaseEvent(QMouseEvent *event) override;
    void mouseMoveEvent(QMouseEvent *event) override;

private:
    void translateLanguage();

private:
    QLabel *title_label;      // 标题文本
    QLabel *msg_label;        // 提示图标
    QLabel *ask_label;        // 提示详细文本
    QCheckBox *check_box;     // "不再提示" 复选框
    QPushButton *ok_button;   // 确定按钮
    QPushButton *cancel_button; // 取消按钮
    PushButton *close_button;  // 关闭按钮

    QString ok_text;
    QString cancel_text;

    bool mouse_press;         // 记录鼠标是否按下
    QPoint move_point;        // 记录鼠标移动的相对距离
};

#endif // MSG_BOX_H

二、 核心实现文件 (msg_box.cpp) #

cpp
#include "msg_box.h"

MsgBox::MsgBox(QWidget *parent)
    : QDialog(parent)
{
    this->resize(320, 160);
    int width = this->width();
    int height = this->height();
    
    // 初始化鼠标按压状态
    mouse_press = false;
    
    // 设置无边框并且作为模态 Dialog 弹出
    this->setWindowFlags(Qt::FramelessWindowHint | Qt::Dialog);
    
    // 创建自定义关闭按钮并设置其坐标
    close_button = new PushButton(this);
    close_button->loadPixmap("tipclose"); // 载入关闭图片资源
    close_button->setGeometry(width - 30, 0, 30, 30); 
    
    // 设置标题栏 QLabel
    title_label = new QLabel(this);
    title_label->setObjectName(QString::fromUtf8("labelOne"));
    title_label->setGeometry(0, 0, width - 50, 30);
    QFont font = title_label->font();
    font.setBold(true);
    title_label->setFont(font);
    
    // 设置提示图标 QLabel
    msg_label = new QLabel(this);
    msg_label->setGeometry(20, 50, 36, 36);
    msg_label->setScaledContents(true);
    
    // 设置提示文本 QLabel，支持多行自动换行显示
    ask_label = new QLabel(this);
    ask_label->setGeometry(65, 60, width - 100, 25 * 2);
    ask_label->setWordWrap(true);
    ask_label->setAlignment(Qt::AlignTop);
    
    // 设置“记住选择/不再提示”复选框
    check_box = new QCheckBox(this);
    check_box->setGeometry(10, height - 35, 160, 25);
    check_box->setHidden(true);
    check_box->setStyleSheet("background:transparent;");
    
    // 创建确认和取消按钮并做相应坐标排布
    cancel_button = new QPushButton(this);
    cancel_button->resize(70, 25);
    cancel_button->move(width - cancel_button->width() - 10, height - 35);
    
    ok_button = new QPushButton(this);
    ok_button->resize(70, 25);
    ok_button->move(width - ok_button->width() - cancel_button->width() - 20, height - 35);
    
    ok_button->setObjectName(QString::fromUtf8("pushButtonTwo"));
    cancel_button->setObjectName(QString::fromUtf8("pushButtonTwo"));
    
    // 信号槽连接
    connect(ok_button, SIGNAL(clicked()), this, SLOT(okOperate()));
    connect(close_button, SIGNAL(clicked()), this, SLOT(cancelOperate()));
    connect(cancel_button, SIGNAL(clicked()), this, SLOT(cancelOperate()));
    
    this->translateLanguage();
}

void MsgBox::translateLanguage()
{
    close_button->setToolTip(tr("close"));
    check_box->setText(tr("remember"));
    ok_text = tr("ok");
    cancel_text = tr("cancel");
}

void MsgBox::setInfo(QString title_info, QString info, QPixmap pixmap, bool is_check_hidden, bool is_ok_hidden)
{
    title_label->setText(QString("  ") + title_info);
    ask_label->setText(info);
    msg_label->setPixmap(pixmap);
    
    check_box->setChecked(false);
    check_box->setHidden(is_check_hidden);
    ok_button->setHidden(is_ok_hidden);
    
    if (is_ok_hidden) {
        // 若隐藏确定按钮，则取消按钮直接显示“确定”文本
        cancel_button->setText(ok_text);
    } else {
        ok_button->setText(ok_text);
        cancel_button->setText(cancel_text);
    }
    
    // 默认焦点落在取消按钮上，防范误触敲回车键直接确认
    cancel_button->setFocus();
}

// 💡 核心绘制：使用 QBitmap 对无边框窗体做圆角裁剪，并绘制自定义底图
void MsgBox::paintEvent(QPaintEvent *)
{
    QPainter painter(this);
    painter.drawPixmap(rect(), QPixmap(":/icon/tip")); // 绘制背景大图
    
    // 生成圆角遮罩蒙版
    QBitmap bitmap(this->size());
    QPainter painter2(&bitmap);
    painter2.fillRect(bitmap.rect(), Qt::white);
    painter2.setBrush(QColor(0, 0, 0));
    painter2.drawRoundedRect(rect(), 4, 4); // 圆角大小为 4px
    setMask(bitmap);
}

// 💡 窗口拖拽实现
void MsgBox::mousePressEvent(QMouseEvent *event)
{
    if (event->button() == Qt::LeftButton) {
        mouse_press = true;
    }
    move_point = event->globalPos() - pos();
}

void MsgBox::mouseReleaseEvent(QMouseEvent *)
{
    mouse_press = false;
}

void MsgBox::mouseMoveEvent(QMouseEvent *event)
{
    if (mouse_press) {
        QPoint move_pos = event->globalPos();
        move(move_pos - move_point);
    }
}

void MsgBox::okOperate()
{
    if (!check_box->isHidden()) {
        emit msgChecked(check_box->isChecked(), true);
    }
    this->accept();
}

void MsgBox::cancelOperate()
{
    if (!check_box->isHidden()) {
        if (!ok_button->isHidden()) {
            emit msgChecked(check_box->isChecked(), false);
        } else {
            emit okMessageHidden(check_box->isChecked());
        }
    }
    this->reject();
}

三、 使用示例 #

在业务逻辑中，我们可以方便地声明 MsgBox 并获取执行返回值：

cpp
// 示例 1：双按钮确认提示框（例如：文件删除）
MsgBox *msg_box = new MsgBox(this);
msg_box->setInfo(QString("文件删除"), QString("确实要把文件放入回收站吗?"), 
                 QPixmap(":/icon/attention"), true, false);
if (msg_box->exec() == QDialog::Accepted) {
    // 执行删除操作...
}

// 示例 2：单确认按钮警告提示框（例如：格式错误）
MsgBox *msg_box2 = new MsgBox(this);
msg_box2->setInfo(QString("重命名"), QString("文件名不能包含下列任何字符：\n     \\ / : * ? \" < > |"), 
                 QPixmap(":/icon/attention"), true, true);
msg_box2->exec();

// 示例 3：带复选框的提示框
MsgBox *msg_box3 = new MsgBox(this);
msg_box3->setInfo(QString("批量文件删除"), QString("确实要把所有文件放入回收站吗?"), 
                 QPixmap(":/icon/attention"), false, false);
msg_box3->exec();

四、 总结 #

界面样式的实现是不设限制的，开发者既可以通过定制精美的背景图片加无边框来实现，也可以利用纯 QSS 对提示框中的子控件边框与底色进行自定义定义。设计自定义模态对话框的关键在于隐藏系统默认标题栏后，需自主重写鼠标事件以支持窗口的正常拖动与遮罩圆角处理。
