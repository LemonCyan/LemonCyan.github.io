Qt之系统托盘（QSystemTrayIcon详解） #

为了提升用户体验，许多桌面后台程序、即时通讯工具（如 QQ、微信）在被关闭或最小化时并不退出程序，而是隐藏在操作系统的右下角任务栏中，以系统托盘图标的形式继续常驻后台运行。

在 Qt 开发中，我们可以使用 QSystemTrayIcon 类极为轻松地实现这一功能。

一、 系统托盘菜单的构建 #

托盘图标最核心的交互是右键弹出控制菜单。我们首先设计一个托盘专属的菜单类 TrayMenu（继承自 QMenu）：

cpp
// TrayMenu.h
#pragma once
#include <QMenu>
#include <QAction>

class TrayMenu : public QMenu
{
    Q_OBJECT
public:
    explicit TrayMenu(QWidget *parent = nullptr);

signals:
    void showWidget();

private:
    void createActions();

private:
    QAction *action_show;
    QAction *action_quit;
    QAction *action_login_home;
    QAction *action_help;
    QAction *action_about;
    QAction *action_check_update;
    QAction *action_setting;
};

// TrayMenu.cpp
#include "TrayMenu.h"

TrayMenu::TrayMenu(QWidget *parent)
    : QMenu(parent)
{
    createActions();
}

void TrayMenu::createActions()
{
    // 1. 实例化各个行为 Action
    action_show = new QAction(tr("显示"), this);
    action_quit = new QAction(tr("退出"), this);
    action_login_home = new QAction(tr("登录网页"), this);
    action_help = new QAction(tr("帮助"), this);
    action_about = new QAction(tr("关于"), this);
    action_check_update = new QAction(tr("检查更新"), this);
    action_setting = new QAction(tr("设置"), this);

    // 2. 为 Action 配置系统图标
    action_show->setIcon(QIcon(":/icon/open"));
    action_login_home->setIcon(QIcon(":/icon/home"));
    action_help->setIcon(QIcon(":/icon/help"));
    action_about->setIcon(QIcon(":/icon/about"));
    action_check_update->setIcon(QIcon(":/icon/update"));
    action_setting->setIcon(QIcon(":/icon/set"));
    action_quit->setIcon(QIcon(":/icon/quit"));

    // 3. 将行为添加到菜单上并用分割线排版
    addAction(action_show);
    addAction(action_setting);
    addAction(action_login_home);
    addSeparator();
    addAction(action_help);
    addAction(action_about);
    addAction(action_check_update);
    addSeparator();
    addAction(action_quit);

    action_setting->setVisible(false); // 可根据状态动态控制隐藏

    // 4. 关联动作信号（此处以点击“显示”转发给主界面为例）
    connect(action_show, &QAction::triggered, this, &TrayMenu::showWidget);
}

二、 系统托盘主控管理 #

我们在程序主视窗（如 LoginDialog）的构造阶段，将托盘对象与上面的右键菜单进行绑定，并配置单击、双击的交互事件。

cpp
// LoginDialog.h
#pragma once
#include <QDialog>
#include <QSystemTrayIcon>
#include "TrayMenu.h"

class LoginDialog : public QDialog
{
    Q_OBJECT
public:
    explicit LoginDialog(QWidget *parent = nullptr);

private slots:
    // 🔑 响应托盘激活状态变化的槽函数
    void onIconActivated(QSystemTrayIcon::ActivationReason reason);

private:
    QSystemTrayIcon *system_tray;
    TrayMenu *tray_menu;
};

// LoginDialog.cpp
#include "LoginDialog.h"
#include <QMessageBox>

LoginDialog::LoginDialog(QWidget *parent)
    : QDialog(parent)
{
    // 💡 1. 实例化 QSystemTrayIcon 对象
    system_tray = new QSystemTrayIcon(this);

    // 2. 设置托盘常驻提示文字 (ToolTip) 与图标
    system_tray->setToolTip(tr("我就是托盘"));
    system_tray->setIcon(QIcon(":/icon/login"));

    // 3. 实例化右键关联菜单并绑定
    tray_menu = new TrayMenu(this);
    system_tray->setContextMenu(tray_menu);

    // 4. 连接激活事件与双击恢复显示槽函数
    connect(system_tray, &QSystemTrayIcon::activated, this, &LoginDialog::onIconActivated);
    connect(tray_menu, &TrayMenu::showWidget, this, &LoginDialog::showNormal);

    // 5. 显示系统托盘
    system_tray->show();

    // 💡 6. 主动向任务栏上报一条系统通知（气泡弹窗）
    system_tray->showMessage(tr("托盘标题"), tr("托盘已启动并在后台运行。"));
}

void LoginDialog::onIconActivated(QSystemTrayIcon::ActivationReason reason)
{
    switch (reason) {
        // 单击托盘图标，恢复显示主窗口
        case QSystemTrayIcon::Trigger:
            this->showNormal();
            this->raise();
            this->activateWindow();
            break;
            
        // 双击托盘图标，恢复显示主窗口
        case QSystemTrayIcon::DoubleClick:
            this->showNormal();
            break;
            
        default:
            break;
    }
}

三、 总结 #

通过结合使用 QSystemTrayIcon 的以下三大接口，可以轻松构建出符合商业规范的托盘工具：

  • setContextMenu(QMenu*)：直接绑定托盘的右击弹出选项。
  • activated(QSystemTrayIcon::ActivationReason) 信号：精准响应用户的单击 (Trigger)、双击 (DoubleClick) 或中键点击 (MiddleClick)。
  • showMessage(QString title, QString msg)：向系统右下角推送通知气泡。
