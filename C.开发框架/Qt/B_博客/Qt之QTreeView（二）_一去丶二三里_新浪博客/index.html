Qt之QTreeView（二）自定义项绘制与树形扩展折叠按钮 #

在《Qt之QTreeView（一）》中，我们详细实现了如何通过继承 QAbstractItemModel 实现自定义树形模型。但在实际应用开发中，除了底层树形数据的定义，我们往往还需要对每一行节点的展现形式进行微调。比如：

  • 默认的 QTreeView 展开/折叠三角形把手（Branch）是在最左侧文字前方的。我们希望将其隐藏，替换成我们自定义绘制在行末最右侧的按钮。
  • 鼠标悬停到该按钮上时，鼠标形状变成手型（PointingHandCursor），并且弹出 ToolTip 气泡提示当前是“展开”还是“折叠”。
  • 点击该按钮时，直接控制 QTreeView 折叠或者展开对应的父节点。

这种深度定制视图内交互与绘制逻辑的需求，最经典的解决方案就是通过继承 QStyledItemDelegate 委托来实现。

一、 自定义委托类设计 (ItemDelegate) #

我们需要让委托类持有 QTreeView 的指针，以便能够判定每个索引当前是否处于展开状态（QTreeView::isExpanded），并在点击时操作对应的展开状态。

1. 头文件 (ItemDelegate.h) #

cpp
#pragma once
#include <QStyledItemDelegate>
#include <QTreeView>

class ItemDelegate : public QStyledItemDelegate
{
    Q_OBJECT
public:
    explicit ItemDelegate(QObject *parent = nullptr);
    ~ItemDelegate() {}

    // 设置与委托关联的 TreeView 实例
    void setView(QTreeView *treeView);
    // 配置小图标的尺寸大小
    void setIconSize(QSize size);
    // 配置展开与折叠状态的图标路径
    void setExpandIcon(const QString &expandIconName);
    void setCollapseIcon(const QString &collapseIconName);

    // 重写绘制流程
    void paint(QPainter *painter, 
               const QStyleOptionViewItem &option, 
               const QModelIndex &index) const override;

    // 重写事件过滤，捕获鼠标动作
    bool editorEvent(QEvent *event, 
                     QAbstractItemModel *model, 
                     const QStyleOptionViewItem &option, 
                     const QModelIndex &index) override;

signals:
    void toggleExpanded(const QModelIndex &index); // 发射控制展开信号

private:
    QTreeView *m_treeView;
    int m_pixmapWidth;
    int m_pixmapHeight;
    QString m_expandIconName;
    QString m_collapseIconName;
};

2. 源文件 (ItemDelegate.cpp) #

cpp
#include "ItemDelegate.h"
#include <QPainter>
#include <QMouseEvent>
#include <QApplication>
#include <QToolTip>

ItemDelegate::ItemDelegate(QObject *parent)
    : QStyledItemDelegate(parent)
    , m_treeView(nullptr)
    , m_pixmapWidth(9)
    , m_pixmapHeight(9)
{
    // 初始化默认展开折叠箭头图标路径 (可以放置在 Qt 的 qrc 资源中)
    m_expandIconName = ":/Images/arrowBottom.png";
    m_collapseIconName = ":/Images/arrowTop.png";
}

void ItemDelegate::setView(QTreeView *treeView)
{
    m_treeView = treeView;
}

void ItemDelegate::setIconSize(QSize size)
{
    m_pixmapWidth = size.width();
    m_pixmapHeight = size.height();
}

void ItemDelegate::setExpandIcon(const QString &expandIconName)
{
    m_expandIconName = expandIconName;
}

void ItemDelegate::setCollapseIcon(const QString &collapseIconName)
{
    m_collapseIconName = collapseIconName;
}

// 1. 自定义绘制树节点行末尾的动作图标
void ItemDelegate::paint(QPainter *painter, 
                         const QStyleOptionViewItem &option, 
                         const QModelIndex &index) const
{
    QStyleOptionViewItem viewOption(option);
    
    // 隐藏选中行时的虚线焦点外边框
    if (viewOption.state & QStyle::State_HasFocus)
    {
        viewOption.state = viewOption.state ^ QStyle::State_HasFocus;
    }

    // 调用基类默认行为绘制文本背景与常规图片
    QStyledItemDelegate::paint(painter, viewOption, index);

    if (m_treeView != nullptr)
    {
        const QAbstractItemModel *model = index.model();
        // 如果当前节点是叶子节点（即没有子项），则不需要渲染折叠/展开按钮
        if (!model->hasChildren(index))
        {
            return;
        }

        // 获取当前节点是否展开
        bool bExpanded = m_treeView->isExpanded(index);

        // 计算绘制在行最右端 30 像素偏移的位置
        int verticalOffset = (viewOption.rect.height() - m_pixmapHeight) / 2;
        QRect decorationRect(viewOption.rect.left() + viewOption.rect.width() - 30, 
                             viewOption.rect.top() + verticalOffset, 
                             m_pixmapWidth, m_pixmapHeight);

        // 根据展开折叠状态加载不同的图片渲染
        QPixmap pixmap = bExpanded ? QPixmap(m_expandIconName) : QPixmap(m_collapseIconName);
        painter->drawPixmap(decorationRect, pixmap);
    }
}

// 2. 劫持鼠标悬浮手型、ToolTip 与点击行为
bool ItemDelegate::editorEvent(QEvent *event, 
                               QAbstractItemModel *model, 
                               const QStyleOptionViewItem &option, 
                               const QModelIndex &index)
{
    if (m_treeView == nullptr || !model->hasChildren(index))
    {
        return QStyledItemDelegate::editorEvent(event, model, option, index);
    }

    // 计算与 paint() 完全相同的右侧按钮区域
    int verticalOffset = (option.rect.height() - m_pixmapHeight) / 2;
    QRect decorationRect(option.rect.left() + option.rect.width() - 30, 
                         option.rect.top() + verticalOffset, 
                         m_pixmapWidth, m_pixmapHeight);

    QMouseEvent *mouseEvent = static_cast<QMouseEvent*>(event);

    // 判定鼠标指针是否处于按钮的物理范围内
    if (decorationRect.contains(mouseEvent->pos()))
    {
        bool bExpanded = m_treeView->isExpanded(index);

        if (event->type() == QEvent::MouseButtonPress)
        {
            // 💡 关键：点击时直接通知发射折叠状态切换信号，或者在此处直接修改 treeView
            emit toggleExpanded(index);
        }
        else if (event->type() == QEvent::MouseMove)
        {
            // 悬浮时，改变光标样式为手型指向，并弹出 ToolTip 提示
            QApplication::setOverrideCursor(Qt::PointingHandCursor);
            QString tooltipText = bExpanded ? QStringLiteral("折叠") : QStringLiteral("展开");
            QToolTip::showText(mouseEvent->globalPos(), tooltipText);
        }
    }
    else
    {
        // 移出区域时，将光标还原为箭头
        QApplication::restoreOverrideCursor();
    }

    return QStyledItemDelegate::editorEvent(event, model, option, index);
}

二、 客户端绑定与信号处理 #

在初始化 QTreeView 的时候，除了使用 setItemDelegate 注入委托外，我们还要监听委托的 toggleExpanded 信号，调用 setExpanded 来真正修改节点的收缩折叠。

cpp
// 1. 初始化视图与模型
QTreeView *treeView = new QTreeView(this);
MyTreeModel *model = new MyTreeModel(this);
treeView->setModel(model);

// 2. 创建并绑定自定义委托
ItemDelegate *delegate = new ItemDelegate(this);
delegate->setView(treeView); // 传入视图以便获取/判定 Expanded 状态
treeView->setItemDelegate(delegate);

// 💡 3. 关联展开折叠信号
connect(delegate, &ItemDelegate::toggleExpanded, this, [=](const QModelIndex &index){
    bool nextState = !treeView->isExpanded(index);
    treeView->setExpanded(index, nextState); // 执行展开或收缩
});

三、 设计原理剖析 #

  • paint() 状态分支判定：委托的 paint 入参提供了一个 QStyleOptionViewItem，通过里面的 option.rect 可以精准获取到当前绘制的那一行的整体尺寸边界。我们在右端留出特定 Padding 绘制图标即可，这比起直接重载整个视图的 Paint 效率更高，代码也更清晰。
  • 与 Base-Class 分离逻辑：重写 paint 时，最开始调用 QStyledItemDelegate::paint() 能够将单元格原有的文本背景、选择高亮等基础绘制任务直接托管回系统底层，我们只需要在其表面通过“图层迭加”的方式在边缘追加一个小图标，省去了自算字体间距的繁琐步骤。
