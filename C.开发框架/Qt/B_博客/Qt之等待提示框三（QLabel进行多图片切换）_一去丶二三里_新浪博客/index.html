Qt之等待提示框三（QLabel进行多图片切换） #

在执行用户登录、后台线程数据验证或者其他耗时处理时，前台往往需要显示一个等待动画（例如旋转菊花），避免用户因无响应而反复操作，带来不良体验。

此前我们介绍过使用 QMovie 播放 gif 图片或者纯画笔代码绘制实现等待框的方法。今天我们再探讨另一种实现方式：通过 QLabel 配合 QTimer 定时切换图片帧来实现。

一、 等待提示组件头文件 (loading_widget.h) #

下面是自定义等待框小部件 LoadingWidget 的类声明：

cpp
#ifndef LOADING_WIDGET_H
#define LOADING_WIDGET_H

#include <QWidget>
#include <QLabel>
#include <QTimer>
#include <QHBoxLayout>
#include <QVBoxLayout>

class LoadingWidget : public QWidget
{
    Q_OBJECT

public:
    explicit LoadingWidget(QWidget *parent = nullptr);
    ~LoadingWidget();

    void setAccount(QString account);
    void start(bool is_start);

private slots:
    void updatePixmap();

private:
    void translateLanguage();

private:
    QLabel *loading_label;  // 轮播展示图片的 Label
    QLabel *account_label;  // 账号文本 Label
    QLabel *waiting_label;  // “正在登录...”等提示信息 Label

    QTimer *timer;          // 帧切换定时器
    int index;              // 当前图片索引
    QString account;        // 用户名/账号名称
};

#endif // LOADING_WIDGET_H

二、 核心实现文件 (loading_widget.cpp) #

cpp
#include "loading_widget.h"
#include <QPixmap>

// 假设我们有名为 Util::updateText 的公共辅助函数来将过长的字符串做省略显示
namespace Util {
    bool updateText(const QString &text, int width, QString &elidedText) {
        // 省略逻辑实现，如果字幅过长，可以用 "..." 代替
        if (text.length() > 20) {
            elidedText = text.left(17) + "...";
            return true;
        }
        return false;
    }
}

LoadingWidget::LoadingWidget(QWidget *parent)
    : QWidget(parent)
{
    index = 1;
    
    loading_label = new QLabel(this);
    account_label = new QLabel(this);
    waiting_label = new QLabel(this);
    
    // 初始化第一帧图片大小并设置
    QPixmap pixmap(QString(":/loginDialog/") + QString::number(index, 10));
    loading_label->setFixedSize(pixmap.size());
    loading_label->setPixmap(pixmap);
    loading_label->setScaledContents(true);
    
    // 右半边布局：账号 + 正在等待提示语
    QVBoxLayout *v_layout = new QVBoxLayout();
    v_layout->addWidget(account_label);
    v_layout->addWidget(waiting_label);
    v_layout->setSpacing(15);
    v_layout->setContentsMargins(0, 0, 0, 0);
    
    // 水平布局：左图 + 右文本，并添加两侧弹簧居中
    QHBoxLayout *h_layout = new QHBoxLayout();
    h_layout->addStretch();
    h_layout->addWidget(loading_label);
    h_layout->addLayout(v_layout);
    h_layout->addStretch();
    h_layout->setSpacing(20);
    h_layout->setContentsMargins(0, 0, 0, 30);
    
    // 定时器初始化：设置 100ms（每秒 10 帧）切换一次图片
    timer = new QTimer(this);
    timer->setInterval(100);
    connect(timer, &QTimer::timeout, this, &LoadingWidget::updatePixmap);
    
    this->setLayout(h_layout);
    this->translateLanguage();
}

LoadingWidget::~LoadingWidget()
{
}

void LoadingWidget::translateLanguage()
{
    waiting_label->setText(tr("正在登录，请稍候..."));
}

void LoadingWidget::setAccount(QString account)
{
    this->account = account;
}

// 启动或停止动画
void LoadingWidget::start(bool is_start)
{
    if (is_start) {
        // 对用户名做缩略排布处理，防止超长引起界面拉伸
        QString elided_text;
        bool is_elided = Util::updateText(account, 200, elided_text);
        if (!is_elided) {
            elided_text = account;
        }
        account_label->setText(tr("你好，") + elided_text);
        timer->start();
    } else {
        timer->stop();
        index = 1;
        // 重置为第一帧
        loading_label->setPixmap(QPixmap(QString(":/loginDialog/") + QString::number(index, 10)));
    }
}

// 💡 核心图片轮播更新
void LoadingWidget::updatePixmap()
{
    index++;
    // 如果总共有 8 张序列帧图片（命名为 1.png, 2.png ... 8.png）
    if (index > 8) {
        index = 1;
    }
    
    QPixmap pixmap(":/loginDialog/" + QString::number(index, 10));
    loading_label->setPixmap(pixmap);
}

三、 💡 效率与代码洁癖探讨 #

本篇的实现主要是用 QTimer 定时发出超时信号，在对应的槽函数中加载并切换图片。对此，有同行提出了如下优化方案：

优化建议：对于简单的状态更新（如帧切换），使用 new QTimer 连接信号槽会占用较多的内核开销。我们完全可以省去声明 QTimer 成员变量，而是直接通过调用 QObject::startTimer() 启动定时器，并在重写的 timerEvent() 槽函数中捕获定时器 ID 进行帧动画更新。

这样不仅可以使类声明更加干净（省去一个 QTimer 实例），在调用层面上效率也会略微提高。

优化版代码示范： #

cpp
// 在构造函数中直接启用内置定时器
// int timerId = this->startTimer(100);

// 重写 timerEvent 函数
void LoadingWidget::timerEvent(QTimerEvent *event)
{
    // 如果监控该 timerId
    index++;
    if (index > 8) {
        index = 1;
    }
    loading_label->setPixmap(QPixmap(QString(":/loginDialog/") + QString::number(index, 10)));
}

这种重载内核事件的形式使得代码在架构上更加紧凑，也是一种非常值得推荐的 Qt 代码编排习惯。
