QwtPlot之绘制统计图（二）柱体标值与零线标线绘制 #

在上一篇《QwtPlot之绘制统计图（一）》中，我们实现了基础的多色柱状图。但是在实际业务报表中，我们通常需要解决两个问题：

  1. 直接展示数值：用户需要在一眼扫过去时直接看到每个柱子代表的数值，而不用对比 Y 轴刻度线。
  2. 支持正负数值（盈亏分析）：柱子可能往上（盈利）或往下（亏损）生长，需要绘制一条明显的 0 点分界线。

本文将通过继承 QwtColumnSymbol 绘制柱体上方的数值文字，并利用 QwtPlotMarker 绘制零位基准线。

一、 自定义 X 轴“图标 + 规格”标尺 (DistroScaleDraw) #

我们重写 QwtScaleDraw 的 drawLabel 函数。在每一个 X 轴刻度下方，先绘制一个小图标，再在图标右侧写出直径标签。

cpp
#include <qwt_scale_draw.h>
#include <QPainter>

class DistroScaleDraw : public QwtScaleDraw
{
public:
    DistroScaleDraw(Qt::Orientation orientation, const QStringList &labels)
        : d_labels(labels)
    {
        setTickLength(QwtScaleDiv::MinorTick, 0);
        setTickLength(QwtScaleDiv::MediumTick, 0);
        setTickLength(QwtScaleDiv::MajorTick, 5);
        enableComponent(QwtScaleDraw::Backbone, false);
        
        double rotation = -10.0;
        setLabelRotation(rotation);
        setLabelAlignment(Qt::AlignCenter);
    }

    // 返回空串以便给自定义绘图留出足够的物理空间
    virtual QwtText label(double value) const override
    {
        return QString("                ");
    }

    // 自定义绘制标尺标签
    virtual void drawLabel(QPainter *painter, double val) const override
    {
        QFont font = this->label(val).font();
        QRectF rect = this->labelRect(font, val);
        QString text("");
        int index = static_cast<int>(val);
        if (index >= 0 && index < d_labels.size()) {
            text = d_labels[index];
        }

        QFontMetrics fm(font);
        int textWidth = fm.width(text);
        int offset_x = (rect.width() - textWidth - 13) / 2;
        
        QPointF point = this->labelPosition(val);
        int left = point.x() - rect.width() / 2 + offset_x;

        // 1. 绘制左侧的图标
        painter->drawPixmap(QPointF(left, point.y()), QPixmap(":/BarLevel.png"));
        
        // 2. 绘制右侧的文本
        painter->drawText(QRect(left + 13, point.y(), rect.width() - offset_x - 13, rect.height()), 
                          Qt::AlignLeft | Qt::AlignVCenter, text);
    }

private:
    QStringList d_labels;
};

二、 柱体及数值标注绘制器 (ColumnSymbol) #

默认的 QwtColumnSymbol 只能绘制柱子矩形。我们通过重写其 draw() 方法，利用 Painter 绘制柱体填充后，根据正负值计算高度偏移，在柱子顶部或底部绘制具体的数值字符串。

cpp
#include <qwt_column_symbol.h>
#include <QPainter>

class ColumnSymbol: public QwtColumnSymbol
{
public:
    ColumnSymbol(int hit, QColor color) : QwtColumnSymbol()
    {
        d_hit = hit;
        d_color = color;
    }

    // 重写柱体核心绘制过程
    virtual void draw(QPainter *painter, const QwtColumnRect &rect) const override
    {
        QRectF columnRect = rect.toRect();
        
        // 1. 填充柱体矩形
        painter->setBrush(d_color);
        painter->setPen(Qt::NoPen);
        painter->drawRect(columnRect);

        // 2. 绘制数值标注
        painter->setPen(Qt::black);
        int top = columnRect.top();
        int height = columnRect.height();
        Qt::Alignment alignment;
        
        if (d_hit >= 0)
        {
            // 正数（盈利）：数值绘制在柱体上方，预留 15 像素高度
            top -= 15;
            height += 15;
            alignment = Qt::AlignTop | Qt::AlignHCenter;
        }
        else
        {
            // 负数（亏损）：数值绘制在反向生长柱体的最下方
            height += 15;
            alignment = Qt::AlignBottom | Qt::AlignHCenter;
        }
        
        painter->drawText(QRect(columnRect.left(), top, columnRect.width(), height), 
                          alignment, QString::number(d_hit));
    }

private:
    int d_hit;
    QColor d_color;
};

三、 图表方向自适应与零线标线 (QwtPlotMarker) #

为了指明正负值的分界线，我们需要引入 QwtPlotMarker。在纵向图表中，零点分割线是一条水平基准线（y = 0）；在横向图表中，则是一条垂直基准线（x = 0）。

零线绘制与方向切换实现 #

cpp
#include "barchart.h"
#include <qwt_plot_marker.h>
#include <qwt_plot_canvas.h>
#include <qwt_plot_layout.h>
#include <qwt_legend.h>

BarChart::BarChart(QWidget *parent)
    : QwtPlot(parent), m_plotMarker(nullptr)
{
    // 初始化盈亏数据（包含正负值）
    const struct {
        QString distro;
        int hits;
        QColor color;
    } pageHits[] = {
        {QString::fromLocal8Bit("8"), 30, QColor(0, 160, 230)},
        {QString::fromLocal8Bit("10"), 50, QColor(115, 185, 35)},
        {QString::fromLocal8Bit("12"), 40, QColor(235, 30, 35)},
        {QString::fromLocal8Bit("14"), 80, QColor(160, 70, 164)},
        {QString::fromLocal8Bit("16"), -20, QColor(255, 130, 0)},
        {QString::fromLocal8Bit("18"), -40, QColor(125, 125, 125)},
        {QString::fromLocal8Bit("12-菱"), -60, QColor(125, 125, 255)},
        {QString::fromLocal8Bit("20"), 60, QColor(180, 255, 180)},
        {QString::fromLocal8Bit("22"), 20, QColor(135, 205, 250)},
        {QString::fromLocal8Bit("24"), -10, QColor(255, 175, 200)}
    };

    setAutoFillBackground(true);
    setPalette(QColor("Linen"));
    setTitle(QStringLiteral("盈亏分析表"));

    QwtPlotCanvas *canvas = new QwtPlotCanvas();
    canvas->setLineWidth(2);
    canvas->setFrameStyle(QFrame::Box | QFrame::Sunken);
    canvas->setBorderRadius(10);
    setCanvas(canvas);

    d_barChartItem = new DistroChartItem();
    QVector<double> samples;
    for (size_t i = 0; i < sizeof(pageHits) / sizeof(pageHits[0]); i++)
    {
        d_distros += pageHits[i].distro;
        samples += pageHits[i].hits;
        // 把实际数值传入内部存储，以便在绘制 Column 时渲染文本
        d_barChartItem->addDistro(pageHits[i].hits, pageHits[i].color);
    }
    d_barChartItem->setSamples(samples);
    d_barChartItem->attach(this);

    insertLegend(new QwtLegend());
    setOrientation(0); // 默认纵向
    setAutoReplot(false);
}

void BarChart::setOrientation(int value)
{
    Qt::Orientation orientation = (value == 0) ? Qt::Vertical : Qt::Horizontal;
    int xBottomAxisId = QwtPlot::xBottom;
    int yLeftAxisId = QwtPlot::yLeft;
    
    if (orientation == Qt::Horizontal) {
        qSwap(xBottomAxisId, yLeftAxisId);
    }
    
    d_barChartItem->setOrientation(orientation);

    // 设置轴标题与刻度范围
    setAxisTitle(xBottomAxisId, QStringLiteral("钢筋规格"));
    setAxisMaxMinor(xBottomAxisId, 3);
    setAxisScale(xBottomAxisId, 0, d_barChartItem->dataSize() - 1, 1.0);
    setAxisScaleDraw(xBottomAxisId, new DistroScaleDraw(orientation, d_distros));

    setAxisTitle(yLeftAxisId, QStringLiteral("钢筋量"));
    setAxisScale(yLeftAxisId, -100, 100); // 范围包含正负区间 [-100, 100]
    setAxisMaxMinor(yLeftAxisId, 5);
    setAxisScaleDraw(yLeftAxisId, new QwtScaleDraw());

    // 绘制 0 点基准分割线 (Zero Line Marker)
    if (m_plotMarker == nullptr)
    {
        m_plotMarker = new QwtPlotMarker();
        m_plotMarker->setLinePen(QPen(Qt::red, 1, Qt::DashLine)); // 红色虚线
        m_plotMarker->attach(this);
    }
    
    if (orientation == Qt::Horizontal)
    {
        // 横向柱状图下，绘制垂直零位线 (x = 0)
        m_plotMarker->setLabel(QwtText("x = 0"));
        m_plotMarker->setLabelAlignment(Qt::AlignRight | Qt::AlignTop);
        m_plotMarker->setLineStyle(QwtPlotMarker::VLine);
        m_plotMarker->setXValue(0.0);
    }
    else
    {
        // 纵向柱状图下，绘制水平零位线 (y = 0)
        m_plotMarker->setLabel(QwtText("y = 0"));
        m_plotMarker->setLabelAlignment(Qt::AlignRight | Qt::AlignTop);
        m_plotMarker->setLineStyle(QwtPlotMarker::HLine);
        m_plotMarker->setYValue(0.0);
    }

    plotLayout()->setCanvasMargin(5);
    replot();
}

通过以上方式，当柱状图因正负值越过零点基准时，红色的零点分界虚线能清晰区分，且文字会智能根据延伸朝向贴在柱头最外缘。
