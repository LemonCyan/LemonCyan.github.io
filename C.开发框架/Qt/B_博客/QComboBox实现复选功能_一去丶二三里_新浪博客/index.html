Qt之QComboBox下拉列表多选与复选框功能实现 #

在常规的桌面软件开发中，普通的下拉列表框 QComboBox 只能支持单选交互。然而，很多业务场景需要用户在下拉列表中勾选多个选项（例如：勾选多个导出的列字段、多选过滤器标签等）。

针对 QComboBox 实现多选/复选功能，常见有几种实现方案：

  1. 模型/视图委托法：自定义 QStandardItemModel 并使用 QItemDelegate 渲染 checkbox，控制逻辑略微复杂。
  2. 列表视图嵌入法（推荐）：使用 QListWidget 结合 QCheckBox 拼接，通过 QComboBox::setView 和 setModel 嵌入到下拉框，简单且易于定制。

为了让初学者能够直接复制并在项目中使用，下面提供一个开箱即用、封装好的 MultiComboBox 自定义多选控件实现。

一、 核心设计方案 #

我们继承标准的 QComboBox，在内部做如下重写：

  • 创建一个 QListWidget 作为下拉视图，每个 Item 都使用 setItemWidget() 挂载一个底层的 QCheckBox。
  • 将 QComboBox 的视图重新设置为我们自定义的 QListWidget，并将 Model 同步。
  • 绑定一个只读的 QLineEdit 作为展示文本条，当任何一个 CheckBox 勾选状态发生改变时，遍历收集所有被勾选的文本，使用分号（;）拼接后写回输入框。

二、 完整代码实现 #

1. 头文件 (MultiComboBox.h) #

cpp
#pragma once
#include <QComboBox>
#include <QListWidget>
#include <QLineEdit>
#include <QCheckBox>
#include <QStringList>

class MultiComboBox : public QComboBox
{
    Q_OBJECT
public:
    explicit MultiComboBox(QWidget *parent = nullptr);

    // 动态添加选项接口
    void addCheckItem(const QString &text, const QVariant &userData = QVariant());
    
    // 获取所有选中项的文本列表
    QStringList getSelectedTexts() const;
    
    // 获取所有选中项绑定的用户数据
    QList<QVariant> getSelectedData() const;

    // 清空所有选项
    void clearAll();

signals:
    // 当选中项发生改变时发射此信号
    void selectionChanged();

private slots:
    void onStateChanged(int state);
    void onTextChanged(const QString &text);

private:
    QListWidget *m_listWidget;
    QLineEdit   *m_lineEdit;
    QString      m_selectedText; // 存储当前拼接好的选中明文
    bool         m_isUpdating;   // 状态更新防护锁
};

2. 源文件 (MultiComboBox.cpp) #

cpp
#include "MultiComboBox.h"
#include <QDebug>

MultiComboBox::MultiComboBox(QWidget *parent)
    : QComboBox(parent)
    , m_isUpdating(false)
{
    m_listWidget = new QListWidget(this);
    m_lineEdit = new QLineEdit(this);
    m_lineEdit->setReadOnly(true);

    // 1. 设置 QComboBox 的模型与下拉视图
    this->setModel(m_listWidget->model());
    this->setView(m_listWidget);
    this->setLineEdit(m_lineEdit);

    // 2. 💡 避坑关键点：
    // 当下拉列表展开、用户点击空白处折叠或者失去焦点时，QComboBox 默认会试图将
    // 当前项的数据重写回 LineEdit，导致我们辛苦拼接好的文本被直接清空。
    // 解决办法：连接 textChanged 信号，如果是非用户手动状态的清空，强行恢复 m_selectedText。
    connect(m_lineEdit, &QLineEdit::textChanged, this, &MultiComboBox::onTextChanged);
}

void MultiComboBox::addCheckItem(const QString &text, const QVariant &userData)
{
    QListWidgetItem *pItem = new QListWidgetItem(m_listWidget);
    m_listWidget->addItem(pItem);
    pItem->setData(Qt::UserRole, userData);

    QCheckBox *pCheckBox = new QCheckBox(this);
    pCheckBox->setText(text);
    m_listWidget->setItemWidget(pItem, pCheckBox);

    // 绑定复选框状态改变信号
    connect(pCheckBox, &QCheckBox::stateChanged, this, &MultiComboBox::onStateChanged);
}

void MultiComboBox::onStateChanged(int state)
{
    Q_UNUSED(state);
    if (m_isUpdating) return;
    m_isUpdating = true;

    QStringList selectedList;
    int count = m_listWidget->count();
    
    // 遍历所有 item 的 CheckBox 状态
    for (int i = 0; i < count; ++i)
    {
        QListWidgetItem *pItem = m_listWidget->item(i);
        QWidget *pWidget = m_listWidget->itemWidget(pItem);
        QCheckBox *pCheckBox = qobject_cast<QCheckBox*>(pWidget);

        if (pCheckBox && pCheckBox->isChecked())
        {
            selectedList.append(pCheckBox->text());
        }
    }

    // 分号拼接
    m_selectedText = selectedList.join("; ");
    m_lineEdit->setText(m_selectedText);
    m_lineEdit->setToolTip(m_selectedText); // 超出宽度时鼠标悬停看全部

    m_isUpdating = false;
    emit selectionChanged();
}

void MultiComboBox::onTextChanged(const QString &text)
{
    // 如果外部或者底层视图回退试图清空输入框文本，我们直接强行复原
    if (!m_isUpdating && text != m_selectedText)
    {
        m_lineEdit->setText(m_selectedText);
    }
}

QStringList MultiComboBox::getSelectedTexts() const
{
    QStringList list;
    int count = m_listWidget->count();
    for (int i = 0; i < count; ++i)
    {
        QListWidgetItem *pItem = m_listWidget->item(i);
        QCheckBox *pCheckBox = qobject_cast<QCheckBox*>(m_listWidget->itemWidget(pItem));
        if (pCheckBox && pCheckBox->isChecked())
        {
            list.append(pCheckBox->text());
        }
    }
    return list;
}

QList<QVariant> MultiComboBox::getSelectedData() const
{
    QList<QVariant> dataList;
    int count = m_listWidget->count();
    for (int i = 0; i < count; ++i)
    {
        QListWidgetItem *pItem = m_listWidget->item(i);
        QCheckBox *pCheckBox = qobject_cast<QCheckBox*>(m_listWidget->itemWidget(pItem));
        if (pCheckBox && pCheckBox->isChecked())
        {
            dataList.append(pItem->data(Qt::UserRole));
        }
    }
    return dataList;
}

void MultiComboBox::clearAll()
{
    m_isUpdating = true;
    m_listWidget->clear();
    m_selectedText.clear();
    m_lineEdit->clear();
    m_isUpdating = false;
}

三、 使用方法示例 #

在你的主界面 Widget 中，你可以像使用普通 QComboBox 一样初始化它：

cpp
// 创建多选下拉框
MultiComboBox *multiCombo = new MultiComboBox(this);

// 插入选项并绑定业务 ID
multiCombo->addCheckItem("全部开发人员", 1001);
multiCombo->addCheckItem("测试团队", 1002);
multiCombo->addCheckItem("运维工程师", 1003);
multiCombo->addCheckItem("项目产品经理", 1004);

// 接收选中状态通知
connect(multiCombo, &MultiComboBox::selectionChanged, this, [=](){
    qDebug() << "Current Selected Text List:" << multiCombo->getSelectedTexts();
    qDebug() << "Current Selected Database IDs:" << multiCombo->getSelectedData();
});

通过本自定义类的封装，我们绕过了 Qt 原生 QComboBox 的回写缺陷，实现了界面清爽、且调用方式简单的多选下拉框控件。
