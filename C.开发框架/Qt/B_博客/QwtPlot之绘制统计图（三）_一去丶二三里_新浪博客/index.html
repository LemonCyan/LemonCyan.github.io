QwtPlot之绘制高级统计图（三） #

在日常项目开发中，QWT 库默认提供的图表组件通常只能满足最基础的绘图需求。针对复杂的多维度统计报表（例如“超采控制表”），我们必须重写 QWT 的核心绘图类，实现坐标轴标签自定义绘制、多色柱状图阈值高亮、数值标注防叠防遮挡等高级定制化需求。

下面详细讲解该超采控制图表的完整实现方案。

效果图前瞻 #

最终图表包含以下定制特性：

  1. X 轴混排标签：每个刻度标签同时显示“钢筋规格图标”和“直径文字描述”。
  2. 多色状态柱形：柱子根据计划量与实际采购量之差，动态呈现红色（超采 > 5%）、黄色（正常 ±5%）、绿色（结余 < 5%）。
  3. 数据标签智能避让：根据数值高度比例，自动判断数据文本绘制在柱状图顶部还是内部，避免文字相互叠加影像视觉。
  4. 图表自适应导出：支持完美导出为 PDF、SVG 以及 PNG/BMP 图片格式。

一、 X 轴刻度绘制：自定义“图标 + 文本” (DistroScaleDraw) #

QWT 的默认坐标轴只能绘制纯文本刻度。为了呈现“图标 + 文本”的效果，我们需要继承 QwtScaleDraw 并重写其 drawLabel 和 label 方法。

cpp
#include <qwt_scale_draw.h>
#include <QPainter>

class DistroScaleDraw : public QwtScaleDraw
{
public:
    DistroScaleDraw(QStringList labels, EnScaleDraw scaleDraw)
        : QwtScaleDraw(), m_scaleDraw(scaleDraw), d_labels(labels)
    {
        // 隐藏 X 轴刻度线（设长度为 0）
        setTickLength(QwtScaleDiv::MinorTick, 0);
        setTickLength(QwtScaleDiv::MediumTick, 0);
        setTickLength(QwtScaleDiv::MajorTick, 0);
        // 不显示主坐标轴底线 (Backbone)
        enableComponent(QwtScaleDraw::Backbone, false);
    }

    // 1. 设置占位文本（预留出图标与文字的绘制宽度）
    virtual QwtText label(double value) const override
    {
        // 返回包含 20 个空格的占位串以拉开刻度间距
        return QwtText("                   ");
    }

    // 2. 自定义绘制刻度标签
    virtual void drawLabel(QPainter *painter, double value) const override
    {
        QFont font = this->label(value).font();
        QRectF rect = this->labelRect(font, value);
        
        QString text("");
        int index = static_cast<int>(value);
        if (index >= 0 && index < d_labels.size()) {
            text = d_labels[index];
        }

        QString strBarLevel("");
        QString strDiameter("");
        if (!text.isEmpty()) {
            // 解析字符串格式 (如 "LEVEL_A;Φ12")
            QStringList barList = text.split(";");
            if (barList.count() > 1) {
                strBarLevel = barList.at(0);
                strDiameter = barList.at(1);
            }
        }

        QFontMetrics fm(font);
        int textWidth = fm.width(text);
        
        // 计算居中绘图坐标偏移
        int offset_x = (rect.width() - textWidth - 13) / 2;
        QPointF point = this->labelPosition(value);
        int draw_left = point.x() - rect.width() / 2 + offset_x;

        // 绘制钢筋规格的小图标
        painter->drawPixmap(QPointF(draw_left, point.y()), 
                            QPixmap(QString(":/Bar/barLevel%1").arg(strBarLevel)));
        
        // 绘制右侧的直径尺寸文本
        painter->drawText(QRect(draw_left + 13, point.y(), rect.width() - offset_x - 13, rect.height() + 5),
                          Qt::AlignLeft | Qt::AlignVCenter, strDiameter);
    }

private:
    EnScaleDraw m_scaleDraw;
    QStringList d_labels;
};

二、 柱状图定制：阈值着色与文本智能避让 (GJHOverdraftColumnSymbol) #

我们需要自定义每一个数据柱子的前景色、高度比例以及数值文本的标注方式。

cpp
#include <qwt_column_symbol.h>
#include <QPainter>

class GJHOverdraftColumnSymbol : public QwtColumnSymbol
{
public:
    GJHOverdraftColumnSymbol(double amount, double planAmount, double purchaseAmount, EnAmount enAmount)
        : QwtColumnSymbol(), m_amount(amount), m_planAmount(planAmount),
          m_purchaseAmount(purchaseAmount), m_enAmount(enAmount),
          m_textHeight(18), m_textSpacing(10), m_width(30)
    {
        // 四舍五入，消除精度带来的逻辑误差
        m_amount = qRound(m_amount);
        m_planAmount = qRound(m_planAmount);
        m_purchaseAmount = qRound(m_purchaseAmount);

        // 核心阈值判定逻辑
        m_color = QColor(0, 160, 230); // 默认蓝色
        if (m_enAmount == PurchaseAmount) {
            if (m_planAmount == 0) {
                m_color = (m_purchaseAmount > 0) ? Qt::red : Qt::yellow;
            } else {
                double diffRatio = qAbs(m_planAmount - m_purchaseAmount) / m_planAmount;
                if (diffRatio <= 0.05) {
                    m_color = Qt::yellow; // 波动在 ±5% 以内呈黄色
                } else {
                    m_color = (m_planAmount > m_purchaseAmount) ? Qt::green : Qt::red; // 结余绿色，超采红色
                }
            }
        }
        m_color.setAlpha(180); // 半透明画刷
    }

    // 重写具体的柱状绘制过程
    virtual void draw(QPainter *painter, const QwtColumnRect &rect) const override
    {
        QRectF rectF = rect.toRect();
        if (!rectF.isValid()) return;

        Qt::Alignment alignment = Qt::AlignTop | Qt::AlignHCenter;
        int nHeight = rectF.height();

        // 固定柱子宽度为 30 像素
        if (rectF.width() != m_width) {
            int nLeft = (rectF.width() - m_width) / 2;
            rectF = QRect(rectF.left() + nLeft, rectF.top(), m_width, rectF.height());
        }

        if (m_enAmount == PurchaseAmount) {
            int nPurchaseHeight = nHeight;
            if (m_purchaseAmount == 0) {
                nPurchaseHeight = 0;
            } else if (m_purchaseAmount < m_amount * 1.2) {
                nPurchaseHeight = m_purchaseAmount / (m_amount * 1.2 / nHeight);
            }
            int nSpacing = nHeight - nPurchaseHeight;
            QRectF planRect = QRect(rectF.left(), rectF.top() + nSpacing, rectF.width(), nPurchaseHeight);
            QRectF planTextRect = QRect(rectF.left() - m_textSpacing, rectF.top() + nSpacing - m_textHeight, 
                                        rectF.width() + 2 * m_textSpacing, nPurchaseHeight + m_textHeight);
            
            painter->setPen(m_color);
            painter->setBrush(m_color);
            painter->drawRect(planRect);
            
            // 绘制当前数值
            painter->setPen(Qt::black);
            painter->drawText(planTextRect, alignment, QString("%1").arg(m_purchaseAmount));
        }
        else {
            // 空心框表示总计划量底框
            painter->setBrush(Qt::NoBrush);
            painter->setPen(m_color);
            painter->drawRect(rectF);

            int nPlanHeight = 0;
            if (m_amount != 0) {
                nPlanHeight = m_planAmount / (m_amount / nHeight);
            }
            int nSpacing = nHeight - nPlanHeight;
            QRect planRect = QRect(rectF.left(), rectF.top() + nSpacing, rectF.width(), nHeight - nSpacing);
            QRect planTextRect = QRect(rectF.left() - m_textSpacing, rectF.top() + nSpacing - m_textHeight, 
                                       rectF.width() + 2 * m_textSpacing, nPlanHeight + m_textHeight);
            
            // 填充阶段量画刷
            painter->setBrush(m_color);
            painter->drawRect(planRect);
            painter->setPen(Qt::black);
            
            // 绘制顶部总量文字
            QRect amountTextRect(rectF.left() - m_textSpacing, rectF.top() - m_textHeight, 
                                 rectF.width() + 2 * m_textSpacing, nHeight + m_textHeight);
            painter->drawText(amountTextRect, alignment, QString("%1").arg(m_amount));

            // [智能避让] 如果阶段量柱子超过了柱体的一半高，把数值写入柱体内部，避免文字挤在外部重合
            if (m_planAmount > m_amount / 2) {
                planTextRect = planRect;
            }
            painter->drawText(planTextRect, alignment, QString("%1").arg(m_planAmount));
        }
    }

private:
    double m_amount, m_planAmount, m_purchaseAmount;
    EnAmount m_enAmount;
    QColor m_color;
    int m_textHeight, m_textSpacing, m_width;
};

class GJHPlotMultiBarChart : public QwtPlotMultiBarChart
{
public:
    GJHPlotMultiBarChart() : QwtPlotMultiBarChart() {}
    void setList(QList<BarAmount> barAmountList) { m_barAmountList = barAmountList; }

protected:
    // 回调钩子：动态替换每一个 Sample 项的 Column Symbol
    virtual QwtColumnSymbol *specialSymbol(int sampleIndex, int valueIndex) const override
    {
        BarAmount barAmount = m_barAmountList.at(sampleIndex);
        EnAmount enAmount = (valueIndex == 1) ? PurchaseAmount : PlanAmount;
        
        return new GJHOverdraftColumnSymbol(barAmount.amount, barAmount.planAmount, 
                                             barAmount.purchaseAmount, enAmount);
    }
private:
    QList<BarAmount> m_barAmountList;
};

GJHOverdraftControlChart::GJHOverdraftControlChart(QWidget *parent)
    : QwtPlot(parent)
{
    setTitle(tr("超采控制表"));
    setAxisTitle(QwtPlot::yLeft, tr("完成比例 (%)"));
    setAxisTitle(QwtPlot::xBottom, tr("钢筋规格"));

    // 自定义画布
    QwtPlotCanvas *canvas = new QwtPlotCanvas();
    canvas->setFrameStyle(QwtColumnSymbol::NoFrame);
    canvas->setBorderRadius(5);
    canvas->setCursor(Qt::ArrowCursor);
    setCanvas(canvas);

    // 绑定多柱状图数据项
    m_barChartItem = new GJHPlotMultiBarChart();
    m_barChartItem->setLayoutPolicy(QwtPlotMultiBarChart::AutoAdjustSamples);
    m_barChartItem->setSpacing(10);
    m_barChartItem->setMargin(0);
    m_barChartItem->attach(this);

    insertLegend(new QwtLegend());
    initColumnSymbol();
}

四、 报表自适应导出 PDF/图片 (QwtPlotRenderer) #

QWT 提供 QwtPlotRenderer 用来进行向量图与高质量 PDF 导出。由于默认导出可能导致图片比例变形，我们需要重载导出逻辑。

cpp
void GJHOverdraftControlChart::exportChart(int width, int height)
{
    QString savePath = getRecentDir();
    // 补齐系统路径分隔符
    if (!savePath.endsWith("\\") && !savePath.endsWith("/")) {
        savePath.append("\\");
        savePath = QDir::toNativeSeparators(savePath);
    }
    
    QString documentName = tr("超采控制表");
    QString filePath = savePath + documentName;

    // 获取当前支持的所有图片写出格式
    const QList<QByteArray> imageFormats = QImageWriter::supportedImageFormats();
    
    QStringList filterList;
    filterList += tr("PDF文档 (*.pdf)");
    filterList += tr("SVG文档 (*.svg)");
    filterList += tr("Postscript文档 (*.ps)");
    
    if (imageFormats.size() > 0) {
        QString imageFilter = tr("Images图片 (");
        for (int i = 0; i < imageFormats.size(); i++) {
            if (i > 0) imageFilter += " ";
            imageFilter += QString("*.%1").arg(QString(imageFormats.at(i)));
        }
        imageFilter += ")";
        filterList += imageFilter;
    }

    // 通过 QFileDialog 引导用户选择导出文件格式与文件名
    QString selectedFilter;
    QString fileName = QFileDialog::getSaveFileName(this, tr("导出图表为"), filePath, 
                                                    filterList.join(";;"), &selectedFilter);
    if (fileName.isEmpty()) return;

    QwtPlotRenderer renderer;
    // 重绘渲染并导出到指定尺寸 (毫米)
    renderer.renderTo(this, fileName, QSizeF(width, height));
}
