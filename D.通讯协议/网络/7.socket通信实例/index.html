server.c

include <stdio.h> #

include <unistd.h> #

include <errno.h> #

include <string.h> #

include <strings.h> #

include <stdlib.h> #

include <sys/types.h> #

include <sys/socket.h> #

include <netinet/in.h> #

include <netinet/ip.h> #

define SERVER_PORT 5001 #

define SERVER_IP "192.168.1.101" #

define QUIT "quit" #

int main(int argc,char *argv[]) { int fd = -1,newfd,ret; struct sockaddr_in sin; char buf[BUFSIZ];

//1.创建socket套接字 if ( (fd = socket(AF_INET,SOCK_STREAM,0)) == -1 ) { perror("socket"); exit(1); }

bzero(&sin,sizeof(sin)); sin.sin_family = AF_INET;确定域 sin.sin_port = htons(SERVER_PORT);

if 0 #

sin.sin_addr.s_addr = inet_addr(SERVER_IP);

endif #

//把本地字节序转换为网络字节序 if ( inet_pton(AF_INET,SERVER_IP,(void *)&sin.sin_addr.s_addr) != 1 ) { perror("inet_pton"); exit(1); }

//绑定 if ( bind(fd,(struct sockaddr *)&sin,sizeof(sin)) != 0 ) { perror("bind"); exit(1); }

if ( listen(fd,5) != 0 ) { perror("listen"); exit(1); }

//阻塞等待客户端的连接请求 if ( (newfd = accept(fd,NULL,NULL)) == -1 ) { perror("accept"); exit(1); }

while(1) { bzero(buf,BUFSIZ); do {

ret = read(newfd,buf,BUFSIZ-1); }while( ret < 0 && EINTR == errno);

if ( ret < 0 ) { perror("read"); exit(1); }

if ( !ret ) { break; }

printf("receive data:%s",buf);

if ( !strncasecmp(buf,QUIT,strlen(QUIT)) ) { printf("client is exiting!\n"); break; } }

close(newfd); close(fd);

return 0; }

if ( (fd = socket(AF_INET,SOCK_STREAM,0)) == -1 ) { perror("socket"); exit(1); }

bzero(&sin,sizeof(sin)); sin.sin_family = AF_INET; sin.sin_port = htons(SERVER_PORT);

if ( inet_pton(AF_INET,SERVER_IP,(void *)&sin.sin_addr.s_addr) != 1 ) { perror("inet_pton"); exit(1);

}

if ( connect(fd,(struct sockaddr *)&sin,sizeof(sin)) != 0 ) { perror("listen"); exit(1); }

while(1) { bzero(buf,BUFSIZ); if ( fgets(buf,BUFSIZ-1,stdin) == NULL ) { perror("fgets"); continue; } write(fd,buf,strlen(buf));

if ( !strncasecmp(buf,QUIT,strlen(QUIT))) { printf("client is exiting\n"); break; } } close(fd); return 0; }
