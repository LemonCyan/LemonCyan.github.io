ret = bind(listen_fd,(struct sockaddr*)&srv_addr,sizeof(srv_addr)); if (ret == -1) { perror("cannot bind server socket"); close(listen_fd); unlink(UNIX_DOMAIN); return 1; } ret = listen(listen_fd, 1); if (ret == -1) { perror("cannot listen the client connect request"); close(listen_fd); unlink(UNIX_DOMAIN); return 1; } len = sizeof(clt_addr); com_fd = accept(listen_fd, (struct sockaddr*)&clt_addr, &len);
