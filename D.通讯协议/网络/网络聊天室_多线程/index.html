include <stdlib.h> #

include <stdio.h> #

include <errno.h> #

include <string.h> #

include <netdb.h> #

include <sys/types.h> #

include <netinet/in.h> #

include <sys/socket.h> #

include <pthread.h> #

define SERVER_PORT 8888 #

#define MAX_ONLINE_NUM 512

struct ON_LINE { int fd; char name[32]; };

struct ON_LINE online[MAX_ONLINE_NUM];

void online_init() { int i;

for(i=0;i<MAX_ONLINE_NUM;i++) { online[i].fd = -1; } }

void add_user(int new_fd,char *name) { int i;

for(i=0;i<MAX_ONLINE_NUM;i++) { if(online[i].fd == -1) { online[i].fd = new_fd; strcpy(online[i].name,name); break; } }

}

void del_user(int fd,int index) { int i; char msg[BUFSIZ];

for(i=0;i<MAX_ONLINE_NUM;i++) { if(online[i].fd == fd) { online[i].fd = -1; break; } }

sprintf(msg,"%s left!\n",online[index].name);

for(i=0;i<MAX_ONLINE_NUM;i++) {

if(online[i].fd != -1) { send(online[i].fd,msg,strlen(msg),0); } } }

int find_index_by_fd(int fd) { int i;

for(i=0;i<MAX_ONLINE_NUM;i++) { if(online[i].fd == fd) { return i; } } return -1; }

void func(void arg) { int byte_count,new_fd,i,index; char buffer[BUFSIZ]; char msg[BUFSIZ+32];

new_fd = (int)arg; printf("new_fd = %d \n",new_fd); index = find_index_by_fd(new_fd);

while(1) { if( (byte_count=recv(new_fd,buffer, BUFSIZ,0)) == -1 )

{ perror("recv"); close(new_fd); del_user(new_fd,index); break; }

if( byte_count == 0 ) { printf("peer disconnect \n"); close(new_fd); del_user(new_fd,index); break; }

buffer[byte_count] = '\0'; printf("recv:%s\n",buffer);

sprintf(msg,"%s say %s",online[index].name,buffer);

for(i=0;i<MAX_ONLINE_NUM;i++) { if(online[i].fd != -1) { send(online[i].fd,msg,strlen(msg),0); }

}

}

}

int main(int argc, char *argv[])

{ int sockfd,new_fd,byte_count; struct sockaddr_in server_addr; struct sockaddr_in client_addr; socklen_t sin_size; char buff[BUFSIZ]; char name[32]; int len; pthread_t pid;

if( (sockfd=socket(AF_INET,SOCK_STREAM,0)) == -1 ) { perror("socket"); exit(1); }

printf("sockfd is %d \n",sockfd);

bzero(&server_addr,sizeof(struct sockaddr_in)); server_addr.sin_family=AF_INET; server_addr.sin_addr.s_addr=htonl(INADDR_ANY); server_addr.sin_port=htons(SERVER_PORT);

if( bind(sockfd,(struct sockaddr *)(&server_addr),sizeof(struct sockaddr)) ==-1 ) { perror("bind"); exit(1); }

if( listen(sockfd, 5) == -1 ) { perror("listen"); exit(1); }

online_init();

while(1) { sin_size = sizeof(struct sockaddr_in);

if( (new_fd = accept(sockfd,(struct sockaddr *)(&client_addr),&sin_size)) == -1 ) { perror("accept"); exit(1); }

len = recv(new_fd,name,32,0);

name[len] = '\0'; add_user(new_fd,name);

pthread_create(&pid, NULL,func, (void *)(&new_fd)); }

close(sockfd); close(new_fd);

exit(0); }
