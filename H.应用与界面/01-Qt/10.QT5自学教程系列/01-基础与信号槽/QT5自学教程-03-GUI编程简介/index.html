//头文件保护符，使用该功能能有效防止重复包含的发生
#ifndef MAINWINDOW_H
#define MAINWINDOW_H
//QMainWindow类提供了一个主窗口的应用程序。
#include <QMainWindow>
namespace Ui {
class MainWindow;
}
class MainWindow : public QMainWindow //以public方式从QMainWindow类派生
{
Q_OBJECT //Q_OBJECT宏必须出现在一个定义了它自己的信号和槽或使用了其他Qt元对象系统的类的私有部
public:
explicit MainWindow(QWidget *parent = 0); //QWidget的类是所有用户界面对象的基类
~MainWindow();
private:
Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H
#include "mainwindow.h"
#include "ui_mainwindow.h"
MainWindow::MainWindow(QWidget *parent) :
QMainWindow(parent),
ui(new Ui::MainWindow)
{
ui->setupUi(this); //为指定控件设定用户界面。
ui->pushButton->setText("Close"); //设置按键上显示的文本
}
MainWindow::~MainWindow()
{
delete ui;
}
//2014/7/21 wayne HDU
//问题描述：创建一个简单的gui例子
#include "mainwindow.h"
#include <QApplication>
int main(int argc, char *argv[])
{
QApplication a(argc, argv); //QApplication类管理了GUI应用程序的控制流和主要设置
MainWindow w;
w.show(); //显示项目（项目默认情况下可见）
//QApplication::exec()函数进入主事件循环并且等待直到exit（）函数被调用，然后返回exit（）被设置的值
//有必要调用这个函数来启动事件处理。主事件循环从窗口系统接收事件并且将它们分发给不同应用部件。通常情况下
在exec调用之前不能发生用户交互。
return a.exec();
}
