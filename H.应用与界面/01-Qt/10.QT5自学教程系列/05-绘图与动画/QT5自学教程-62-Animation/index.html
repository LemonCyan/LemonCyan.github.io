#ifndef MYITEM_H
#define MYITEM_H
#include <QGraphicsItem>
#include <QGraphicsScene>
#include <QPainter>
class MyItem : public QGraphicsItem
{
public:
MyItem();
QRectF boundingRect() const;
void paint(QPainter * painter, const QStyleOptionGraphicsItem * option, QWidget * widget = 0);
protected:
void advance(int phase);
private:
qreal angle; //旋转角度
qreal speed; //速度
void DoCollision(); //碰撞处理函数
};
#endif // MYITEM_H
#include "myitem.h"
MyItem::MyItem()
{
//设置开始角度
angle = qrand() % 360;
setRotation(angle); //设置旋转角度
//设置速度
speed = 5;
//设置起始位置
int startX = 0;
int startY = 0;
if(qrand() % 1)
{
startX = qrand() % 200;
startY = qrand() % 200;
}
else
{
startX = qrand() % -100;
startY = qrand() % -100;
}
//setPos()设置当前图形项的坐标为其父对象的坐标。如果图形项没有父对象，设置的坐标为场景的坐标
//mapToParent() 按点进行映射，从图形项的坐标系映射到其父对象的坐标系。如果图形项不存在父对象，点将
会被映射到当前的场景坐标系。
setPos(mapToParent(startX,startY));
}
//返回固定大小矩形
QRectF MyItem::boundingRect() const
{
return QRectF(0,0,20,20);
}
void MyItem::paint(QPainter *painter, const QStyleOptionGraphicsItem *option, QWidget *widget)
{
QRectF rec = boundingRect(); //新建矩形
QBrush brush(Qt::red); //新建画笔刷
//检测碰撞
//collidingItems() 返回碰撞项的列表
if(scene()->collidingItems(this).isEmpty())
{
//无碰撞
brush.setColor(Qt::green);
}
else
{
//发生碰撞
brush.setColor(Qt::red);
//重新图形对象参数
DoCollision();
}
//绘制图形
painter->fillRect(rec,brush);
painter->drawRect(rec);
}
void MyItem::advance(int phase)
{
if(!phase) return;
QPointF location = this->pos();
setPos(mapToParent(0,-speed));
}
void MyItem::DoCollision()
{
//改变旋转角度
if(qrand() % 1)
{
setRotation(rotation() + 180 + qrand() % 10);
}
else
{
setRotation(rotation() + 180 + qrand() % -10);
}
//如果图形位置处在边界，则对其进行设置
QPointF newPoint = mapToParent(-(boundingRect().width()), -(boundingRect().width()+2));
if(!scene()->sceneRect().contains(newPoint))
{
//将其移回边界
newPoint = mapToParent(0,0);
}
else
{
//设置新的位置
setPos(newPoint);
}

#ifndef PEOPLE_H

class People
{
//重载<<和>>运算符，将其声明为类的友元函数
friend QDataStream& operator<<(QDataStream& out,const People& elem);
friend QDataStream& operator>>(QDataStream& in, People& elem);
public:
People(QString name = "",int age = 0);
void print(); //打印成员函数
private:
QString name;
int age;
};
#endif // PEOPLE_H

#include <QDebug>

}

{

}
//重载<<运算符实现
QDataStream& operator<<(QDataStream &out, const People &elem)
{
out << elem.name << elem.age;
return out;
}
//重载>>运算符实现
QDataStream& operator>>(QDataStream &in, People &elem)

in >> elem.name >> elem.age;
