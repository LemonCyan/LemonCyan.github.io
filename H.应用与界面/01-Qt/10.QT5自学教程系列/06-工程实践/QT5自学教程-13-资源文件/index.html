在这一节中将介绍如何在Qt中读取资源文件中的文件内容，下面先来介绍一些新的概念。

A：Resource File顾名思义也就是资源文件的意思。Qt的资源系统是一个独立于平台的机制，用于存储应用程序的可执行二进制文件。如果你的应用程序总是需要一组特定的文件（图标，翻译文件等），并且你不想冒丢失文件的风险，那么这是非常有用的。

#include <QString>

#include <QFile>
//读文件函数
void ReadFile(const QString &filename)
{
//QFile类提供读写文件的接口。QFile是一个用于读写文本文件和二进制文件和资源文件的I/O 设备
QFile my_file(filename);//将QFile与相关文件关联
if(!my_file.open(QIODevice::ReadOnly | QIODevice::Text))//以只读和文本模式打开文件
{
qDebug() <<"Could not open file for Reading";
return;
}
//QTextStream提供一种读写文本文档边界的接口
QTextStream outText(&my_file); //将QTextStream与特定文件关联
qDebug() << outText.readAll(); //读出QTextStream对象中所有内容
my_file.close(); //关闭文件
}
//写文件函数
void WriteFile(const QString &filename)
{
QFile my_file(filename);
if(!my_file.open(QIODevice::WriteOnly | QIODevice::Text)) //以只写和文本模式打开文件
{
qDebug() <<"Could not open file for Writing";
return;
}
QTextStream inText(&my_file);
inText<<"Hello World!"; //向QTextStream中写入内容
my_file.flush(); //刷新文件，将buffer中所有数据写入文件，此语句不可少。
my_file.close();
}
