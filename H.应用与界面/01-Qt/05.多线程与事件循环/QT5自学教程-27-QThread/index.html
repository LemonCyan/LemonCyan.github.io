QT5自学教程 —— QThread #

在这一章将介绍QThread的知识，由于这部分内容比较复杂，所以下面将给出多个例子。因此，下面的介绍将首先，依旧是先厘清相关的概念。

A：Thread也就是线程的意思。线程是程序中一个单一的顺序控制流程.在单个程序中同时运行多个线程完成不同的工作，称为多线程。要弄清楚线程，须结合进程加以理解：

道、同步对象等）组成。一个应用程序可以有一个或多个进程，一个进程可以有一个或多个线程，其中一个是主线程是操作系统分时调度分配CPU时间的基本实体。一个线程可以执行程序的任意部分的代码，即使这部分代码被另一个线程并发地执行；一个进程的所有线程共享它的虚拟地址空间、全局变量和操作系统资源。之所以有线程这个概念，是因为以线程而不是进程为调度对象效率更高：由于创建新进程必须加载代码，而线程要执行的代码已经被映射到进程的地址空间，所以创建、执行线程的速度比进程更快。

A：Priority就是优先级。优先级是计算机分时操作系统在处理多个作业程序时，决定各个作业程序接受系统资源的优先等级的参数。打开任务管理其你可以查看或者更改一个进程的优先级，如下图所示：

cpp
#ifndef MYTHREAD_H
#define MYTHREAD_H
#include <QThread>
class MyThread : public QThread
{
Q_OBJECT
public:
explicit MyThread(QObject *parent = 0);
QString name;
bool stop; //标志位
signals:
public slots:
//这里的run是继承自QThread类的虚函数，在这里对该虚函数进行重写，创建属于我们自己命名类版本的run
//QThread中run是线程的奇点。调用start（）后，新创建的线程调用这个函数。默认实现调用exec（）。
void run();
};
#endif // MYTHREAD_H
#include "mythread.h"
#include <QDebug>
#include <QMutex>
MyThread::MyThread(QObject *parent) :
QThread(parent)
{
stop = false;
}
void MyThread::run()
{
qDebug() <<name<<" Running";
for(int i = 0;i != 1000;++i)
{
//QMutex类使得线程之间的访问串行化。
//QMutex的目的是保护的对象、数据结构或代码段，因此在同一时间只有一个线程能够对其进行访问
QMutex mutex;
mutex.lock();//此时只有当前线程有权限访问相关数据，其他线程被锁定
if(stop)
{
mutex.unlock(); //接触其他线程的锁定限制
break;
}
mutex.unlock(); //接触其他线程的锁定限制
qDebug() << name <<i;
}

例4（线程在gui应用中的简单应用，本例子在gui界面中控制线程的进行和终止，并且显示线程执行中信息的变
#ifndef MYTHREAD_H
#define MYTHREAD_H
#include <QThread>
class MyThread : public QThread
{
Q_OBJECT
public:
explicit MyThread(QObject *parent = 0);
bool Stop; //标识符，用来标识是否终止线程
signals:
//用来发射变化的数值的信号
void NumberChanged(int);
public slots:
void run(); //重写继承函数run
};
#endif // MYTHREAD_H
#include "mythread.h"
#include <QMutex>
MyThread::MyThread(QObject *parent) :
QThread(parent)
{
Stop = false;
}
void MyThread::run()
{
for(int i = 0;i != 10000 ; ++i)
{
QMutex mutex;
mutex.lock();
if(Stop)
{
mutex.unlock();
break;
}
mutex.unlock();
emit NumberChanged(i);//发射信号
msleep(100);//强制当前线程休眠100毫秒
}
#ifndef DIALOG_H
#define DIALOG_H
#include <QDialog>
#include "mythread.h"
namespace Ui {
class Dialog;
}
class Dialog : public QDialog
{
Q_OBJECT
public:
explicit Dialog(QWidget *parent = 0);
~Dialog();
private slots:
//这个槽用来接受thread发出的信号
void onNumberChanged(int number);
//触发按键对应的槽
void on_pushButton_clicked();
void on_pushButton_2_clicked();
private:
Ui::Dialog *ui;
MyThread *thread; //声明线程
};
#endif // DIALOG_H
#include "dialog.h"
#include "ui_dialog.h"
Dialog::Dialog(QWidget *parent) :
QDialog(parent),
ui(new Ui::Dialog)
{
ui->setupUi(this);
thread = new MyThread(this);//定义线程
//连接对应的信号和槽
connect(thread,SIGNAL(NumberChanged(int)),this,SLOT(onNumberChanged(int)));
}
Dialog::~Dialog()
{
delete ui;
}
//依据线程中发出的信号改变当前label中的显示数字
void Dialog::onNumberChanged(int number)
{
ui->label->setText(QString::number(number));
}
void Dialog::on_pushButton_clicked()
{
//Start 开始线程
thread->start();
}
void Dialog::on_pushButton_2_clicked()
{
//Stop 终止线程
thread->Stop = true;
}
