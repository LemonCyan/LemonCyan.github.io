一、两个基础结构体 #

c
struct device_node {
    const char *name;             /* 节点名（不含 @unit-address） */
    phandle phandle;              /* 被 &引用 时用的编号 */
    const char *full_name;        /* 带 @unit-address 的全名 */
    struct property *properties;  /* 属性链表头 */
    struct property *deadprops;   /* 被删除的属性挂这里 */
    struct device_node *parent;
    struct device_node *child;
    struct device_node *sibling;
    /* ... */
};

struct property {
    char *name;
    int length;                   /* 值的字节数 */
    void *value;                  /* 值本身，大端 */
    struct property *next;
};

节点是树（parent/child/sibling 三指针），属性是链表（properties/next）。所有遍历 API 本质上就是在这两种结构上走。

二、找节点 #

2.1 从树上按条件搜 #

函数 说明
of_find_node_by_name(from, name) 按节点名搜，from 为 NULL 时从根开始
of_find_node_by_type(from, type) 按 device_type 属性搜（如 "memory"）
of_find_compatible_node(from, type, compat) 按 compatible 搜，最常用
of_find_node_by_path(path) 按绝对路径搜，如 "/soc/serial@50000000"
of_find_matching_node_and_match(from, matches, &match) 按 of_device_id 表搜，可回传命中的那一项
of_get_child_by_name(np, name) 在某个节点的直接子节点里找
of_get_parent(np) 取父节点
of_get_next_child(np, prev) 遍历子节点（prev 传 NULL 取第一个） 
c
/* 典型写法：从根开始找第一个 compatible 命中的节点 */
struct device_node *np;

np = of_find_compatible_node(NULL, NULL, "samsung,s3c2410-uart");
if (!np) {
    pr_err("uart node not found\n");
    return -ENODEV;
}
/* 用完要 of_node_put(np)，否则引用计数泄漏 */
of_node_put(np);

遍历所有子节点：

c
struct device_node *child;

for_each_child_of_node(parent, child) {
    pr_info("child: %pOFn\n", child);
}

%pOFn / %pOF 是专门打印 device_node 的格式符，比 %s 加 full_name 更安全。

2.2 找属性 #

c
struct property *of_find_property(const struct device_node *np,
                                  const char *name, int *lenp);

返回属性指针，lenp 带回值的字节数。只关心「有没有」时用它最省：

c
if (of_find_property(np, "interrupt-controller", NULL))
    pr_info("this node is an interrupt controller\n");

三、读属性值 #

3.1 整数：单值与数组 #

函数族 读什么
of_property_read_u8/u16/u32/u64(np, name, &out) 单值
of_property_read_u8_array/u16/u32/u64_array(np, name, out, count) 数组
of_property_read_u32_index(np, name, index, &out) 数组里第 index 个

返回值约定要记牢：

返回 含义
0 成功
-EINVAL 属性不存在，或长度不对
-ENODATA 属性存在但没有值
-EOVERFLOW 属性值比目标类型大 
c
u32 reg_base;
int ret;

ret = of_property_read_u32(np, "reg-base", &reg_base);
if (ret) {
    dev_err(dev, "read reg-base failed: %d\n", ret);
    return ret;
}

3.2 字符串 #

c
const char *name;
of_property_read_string(np, "clock-names", &name);   /* 只取第一个字符串 */

3.3 其它常用判定 #

函数 用途
of_property_count_elems_of_size(np, name, elem_size) 数一数属性里有几个元素，例如数中断个数
of_property_match_string(np, "clock-names", "ipg") 返回该名字在字符串数组里的下标
of_n_addr_cells(np) / of_n_size_cells(np) 取本节点 #address-cells / #size-cells
of_device_is_compatible(np, "x,y") 判断某个节点是否含指定 compatible
of_device_is_available(np) 判断 status 是否为 okay/缺省 
c
/* 数中断个数：elem_size 传 4 表示按 u32 计 */
int n = of_property_count_elems_of_size(np, "interrupts", 4);

四、地址与资源：从 dts 到 ioremap #

4.1 三条路的区别 #

方式 拿到什么 是否已映射
platform_get_resource(pdev, IORESOURCE_MEM, i) struct resource * 否
of_address_to_resource(np, i, &res) 同上，但只用 device_node，不依赖 platform_device
of_iomap(np, i) 直接返回映射后的虚拟地址 是

of_iomap 等价于 of_address_to_resource + ioremap，省掉了中间那一步，在只看得到 device_node 的场合（比如 pinctrl、clk 这类子系统驱动）很好用：

c
void __iomem *base;

base = of_iomap(np, 0);
if (!base) {
    pr_err("ioremap failed\n");
    return -ENOMEM;
}
writel(0x1, base + 0x10);
iounmap(base);

4.2 struct resource 与标志位 #

c
struct resource {
    resource_size_t start;
    resource_size_t end;
    const char *name;
    unsigned long flags;
    struct resource *parent, *sibling, *child;
};
标志 值 含义
IORESOURCE_IO 0x00000100 IO 端口
IORESOURCE_MEM 0x00000200 内存映射寄存器
IORESOURCE_REG 0x00000300 寄存器
IORESOURCE_IRQ 0x00000400 中断
IORESOURCE_DMA 0x00000800 DMA 通道
IORESOURCE_BUS 0x00001000 总线号

4.3 大端与地址转换 #

of_get_address(np, index, &size, &flags) 返回的是 const __be32 *（大端原始值），需要自己 be32_to_cpup 转换。跨总线时用 of_translate_address(np, in_addr) 走一遍 ranges 映射，得到真正的 CPU 物理地址。

五、中断相关 #

函数 用途
of_irq_count(np) 数中断个数（内部反复调 of_irq_parse_one）
irq_of_parse_and_map(np, index) 解析第 index 个中断并建立映射，返回 virq
of_irq_get(np, index) 新接口，同上
platform_get_irq(pdev, i) platform 驱动最常用的一条 
c
int irq = platform_get_irq(pdev, 0);
if (irq < 0)
    return irq;

ret = devm_request_irq(&pdev->dev, irq, my_handler, 0, "mydev", dev_id);

devm_request_irq 的第四个参数 flags 传 0 时，触发方式完全按设备树里的 interrupts 第 [3] 个 cell 来。手写 IRQF_TRIGGER_RISING 反而会覆盖设备树配置。

六、头文件分工 #

#include 错了会报一堆隐式声明，这张表可以直接查：

头文件 提供什么
<linux/of.h> 基础：查找节点/属性、读值、of_device_id
<linux/of_address.h> of_address_to_resource、of_iomap、of_get_address
<linux/of_device.h> of_device_alloc、of_find_device_by_node
<linux/of_irq.h> irq_of_parse_and_map、of_irq_get
<linux/of_gpio.h> of_get_named_gpio 等 GPIO 相关
<linux/of_platform.h> of_platform_populate、of_platform_device_create
<linux/of_dma.h> / of_net.h / of_pci.h / of_graph.h 各子系统专用
<linux/of_reserved_mem.h> 保留内存（CMA、framebuffer）

七、一个完整的取资源模板 #

把上面几节拼起来，就是一个 platform 驱动 probe 开头该有的样子：

c
static int my_probe(struct platform_device *pdev)
{
    struct device *dev = &pdev->dev;
    struct device_node *np = dev->of_node;
    struct resource *res;
    void __iomem *base;
    int irq, ret;

    if (!np) {
        dev_err(dev, "no device tree node\n");
        return -EINVAL;
    }

    /* 1. 寄存器 */
    res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    base = devm_ioremap_resource(dev, res);
    if (IS_ERR(base))
        return PTR_ERR(base);

    /* 2. 中断 */
    irq = platform_get_irq(pdev, 0);
    if (irq < 0)
        return irq;

    ret = devm_request_irq(dev, irq, my_handler, 0, dev_name(dev), priv);
    if (ret)
        return ret;

    /* 3. 私有属性 */
    ret = of_property_read_u32(np, "my,delay-us", &priv->delay);
    if (ret)
        priv->delay = 100;      /* 属性可选，给默认值 */

    return 0;
}

八、易错点 #

现象 原因
编译报 implicit declaration of of_xxx 头文件没 include 对，查第六节的表
读到垃圾值 属性长度与读取类型不匹配（比如 u32 读了 8 字节）
of_property_read_string 拿到空 属性是字符串数组，得用 of_property_match_string 或下标访问
引用计数泄漏 of_find_* 拿到节点后忘了 of_node_put
中断触发方式不对 request_irq 的 flags 里手写了触发位，覆盖了 dts
地址不对 忘了 ranges 转换，或用 of_get_address 没做 be32_to_cpup
