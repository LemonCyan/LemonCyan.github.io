一、总览：一条从汇编到 probe 的链路 #

text
head.S                      r1 = machine nr, r2 = atags/dtb 物理地址
  |
  v
start_kernel
  setup_arch(&command_line)                      arch/arm/kernel/setup.c
    mdesc = setup_machine_fdt(__atags_pointer)   arch/arm/kernel/devtree.c
      early_init_dt_verify(phys_to_virt(dt_phys))
      early_init_dt_scan()                       -> bootargs / memory / cells
        of_scan_flat_dt(early_init_dt_scan_chosen)
        of_scan_flat_dt(early_init_dt_scan_root)
        of_scan_flat_dt(early_init_dt_scan_memory)
      of_flat_dt_match_machine()                 -> 选出 machine_desc
    arm_memblock_init(mdesc)
      early_init_fdt_reserve_self()              -> 保护 dtb 自身不被覆盖
  ...
  unflatten_device_tree()                        -> 生成 device_node 树
  rest_init -> kernel_init -> kernel_init_freeable
    do_basic_setup() -> do_initcalls()
      arch_initcall_sync(of_platform_default_populate_init)
        of_platform_populate()                   -> device_node 变 platform_device

理解这条链，就理解了「设备树什么时候生效」：前半段在内存管理初始化之前，只能读平坦的 dtb；后半段在 initcall 阶段，才用得上完整的 device_node 树。

关键在于「平坦」两个字。dtb 在内存里就是一段字节流，没有指针、没有结构体、不能随机访问。所以前期只能用 of_scan_flat_dt 从头顺序走一遍，边走边回调。等 paging_init 之后内存分配器能用了，才把这坨字节流「展开」（unflatten）成互相指针连接的 device_node 树。

二、汇编层的三条约定 #

进入内核 C 代码前，head.S 必须留下三个寄存器：

寄存器 含义
r0 固定为 0
r1 machine number（老式板级编号，如 SMDK2440 是 168）
r2 ATAGS 首地址 或 dtb 物理地址，二选一

此时 MMU 关闭、D-cache 关闭。C 代码接手后，r1/r2 会被保存到两个全局变量：

c
/* arch/arm/kernel/head-common.S */
__machine_arch_type   <- r1
__atags_pointer       <- r2

所以「内核怎么知道 dtb 在哪」的答案很简单：U-Boot 把它塞进了 r2。类型上它是个物理地址，setup_machine_fdt 拿到手第一件事就是 phys_to_virt()。

三、U-Boot 侧：r2 到底传了什么 #

bootm 走 boot_jump_linux()（arch/arm/lib/bootm.c）：

c
static void boot_jump_linux(bootm_headers_t *images, int flag)
{
    unsigned long machid = gd->bd->bi_arch_number;
    void (*kernel_entry)(int zero, int arch, uint params);
    unsigned long r2;
    int fake = (flag & BOOTM_STATE_OS_FAKE_GO);

    kernel_entry = (void (*)(int, int, uint))images->ep;

    s = getenv("machid");
    if (s) {
        machid = simple_strtoul(s, NULL, 16);
        debug("## Using machid from environment: 0x%lx\n", machid);
    } else {
        debug("## Using machid from image: 0x%lx\n", machid);
    }

    if (IMAGE_ENABLE_OF_LIBFDT && images->ft_len)
        r2 = (unsigned long)images->ft_addr;    /* 有 dtb：传 dtb 地址 */
    else
        r2 = gd->bd->bi_boot_params;            /* 无 dtb：传 atags */

    kernel_entry(0, machid, r2);
}

关键判断就是 images->ft_len：bootm kernel_addr ramdisk_addr dtb_addr 传了第三个参数时它非 0，r2 就指向 dtb。

这也是为什么启动命令漏掉 dtb 参数，内核就退回 ATAGS 老路。现象很有欺骗性：内核能启动，串口正常，但 /proc/device-tree 是空的，所有依赖设备树的驱动全部 probe 不进来。查了一下午 dts，问题其实在 bootargs 那一行少写了一个参数。

四、选出 machine_desc：compatible 是怎么打分的 #

4.1 老方式：靠 machine number #

c
#define MACHINE_START(_type, _name)                     \
static const struct machine_desc __mach_desc_##_type    \
    __used                                              \
    __attribute__((__section__(".arch.info.init"))) = { \
    .nr        = MACH_TYPE_##_type,                     \
    .name      = _name,

.nr 就是 U-Boot 传的 machid（Mini2440 = MACH_TYPE_SMDK2440 = 168），链接脚本用 __arch_info_begin / __arch_info_end 界定这个段，setup_machine_tags() 拿 machid 一比对就找到板子。

改成设备树后，这段逻辑变成：

c
/* arch/arm/mach-s3c24xx/mach-smdk2440-dt.c */
static const char *const s3c2440_dt_compat[] __initconst = {
        "samsung,s3c2440",
        "samsung,mini2440",
        NULL
};

DT_MACHINE_START(S3C2440_DT, "Samsung S3C2440 (Flattened Device Tree)")
        .dt_compat      = s3c2440_dt_compat,
        .map_io         = s3c2440_dt_map_io,
        .init_irq       = irqchip_init,
        .init_machine   = s3c2440_dt_machine_init,
MACHINE_END

4.2 新方式：compatible 打分 #

setup_machine_fdt() 本体不长，值得整段看：

c
const struct machine_desc * __init setup_machine_fdt(unsigned int dt_phys)
{
    const struct machine_desc *mdesc, *mdesc_best = NULL;

#ifdef CONFIG_ARCH_MULTIPLATFORM
    DT_MACHINE_START(GENERIC_DT, "Generic DT based system")
    MACHINE_END
    mdesc_best = &__mach_desc_GENERIC_DT;
#endif

    /* dt_phys 是 bootloader 传的物理地址，先转虚拟地址再扫 */
    if (!dt_phys || !early_init_dt_scan(phys_to_virt(dt_phys)))
        return NULL;

    mdesc = of_flat_dt_match_machine(mdesc_best, arch_get_next_mach);
    if (!mdesc) {
        /* 打印根节点 compatible 列表，然后 dump_machine_table() 不返回 */
        ...
        dump_machine_table(); /* does not return */
    }

    /* 把老式 machine number 对齐到选中的 mdesc，后面还有人读它 */
    __machine_arch_type = mdesc->nr;
    return mdesc;
}

注意 CONFIG_ARCH_MULTIPLATFORM 那段：多平台内核先塞一个 GENERIC_DT 兜底，作用是万一没匹配上，不至于直接 panic，先让通用逻辑跑起来。

of_flat_dt_match_machine() 遍历 .arch.info.init 段里所有 machine_desc：

c
const void * __init of_flat_dt_match_machine(const void *default_match,
                                const void * (*get_next_compat)(const char * const**))
{
    const void *data = NULL;
    const void *best_data = default_match;
    const char *const *compat;
    unsigned long dt_root;
    unsigned int best_score = ~1, score = 0;

    dt_root = of_get_flat_dt_root();
    while ((data = get_next_compat(&compat))) {
        score = of_flat_dt_match(dt_root, compat);
        /* 只接受正分，且比当前最优更小 */
        if (score > 0 && score < best_score) {
            best_data = data;
            best_score = score;
        }
    }
    if (!best_data) {
        pr_err("\n unrecognized device tree list:\n[ ");
        /* 把根节点 compatible 逐条打印出来 */
        ...
        return NULL;
    }
    pr_info("Machine model: %s\n", of_flat_dt_get_machine_name());
    return best_data;
}

计分方向是这里最容易记反的一条：score 不是「匹配度越高越大」，而是根节点 compatible 列表里的命中序号（从 1 起）。根节点写 compatible = "samsung,s3c2440", "samsung,mini2440";，某个 dt_compat 命中第一个串得 1 分，命中第二个得 2 分。score < best_score 取最小值，等价于「越靠前越具体，越优先」。

unrecognized device tree list 这条报错几乎只有一个原因：dts 根节点写的 compatible，和内核里任何一个 DT_MACHINE_START 的 dt_compat 都对不上。改 dts 或改内核，二选一。

arch_get_next_mach() 是遍历器，靠链接脚本里的两个符号界定范围：

c
/* arch/arm/kernel/vmlinux.lds.S */
__arch_info_begin = .;
*(.arch.info.init)
__arch_info_end = .;

c
static const void * __init arch_get_next_mach(const char *const **match)
{
    static const struct machine_desc *mdesc = __arch_info_begin;
    const struct machine_desc *m = mdesc;

    if (m >= __arch_info_end)
        return NULL;
    mdesc++;                    /* 指针后移，下次取下一个 */
    *match = m->dt_compat;
    return m;
}

. 是链接器的 location counter。这段等于说「把所有目标文件里的 .arch.info.init 段首尾相接放在这里」，于是所有 MACHINE_START / DT_MACHINE_START 定义的 machine_desc 变成内存里一个连续的数组，用两个符号夹住就能当数组遍历。static 变量在函数里只初始化一次，所以这个遍历器是有状态的、不能并发调用。前期单核单线程，没问题。

4.3 machine_desc 里到底有哪些字段 #

选了半天的东西长这样。移植时真正要填的只有 .dt_compat 和几个 init_* 回调：

字段 类型 作用 什么时候必须填
nr unsigned int 老式架构编号 设备树方案下也要给（__machine_arch_type 会用它），填个不冲突的值即可
name const char * 板名，machine_name 取自这里 必填
atag_offset unsigned long ATAGS 在内存里的偏移（相对 RAM 起始） 只走 ATAGS 的板子
dt_compat const char *const * 设备树匹配表，NULL 结尾 走设备树必填
nr_irqs unsigned int 中断号预留数量 老式板级
dma_zone_size phys_addr_t DMA 区大小，仅 CONFIG_ZONE_DMA 有 ISA DMA 需求时
video_start / video_end unsigned int 显存范围 有 framebuffer 的板子
reserve_lp0/1/2 位域 禁止进入 lp0/lp1/lp2 低功耗态 硬件不支持对应低功耗态时
reboot_mode enum reboot_mode 默认重启方式 一般 REBOOT_HARD
smp / smp_init struct smp_operations * / 函数指针 SMP 启动回调 多核平台
fixup void (*)(struct tag *, char **, struct meminfo *) 修补 ATAGS / bootargs 只走 ATAGS 的板子
init_meminfo void (*)(void) 改写 meminfo 内存布局要修正时
reserve void (*)(void) 预留内存块 有共享内存/显存要占位时
map_io void (*)(void) 静态 I/O 映射 设备树方案一般留空（iotable_init 已经在 dts 里描述）
init_early void (*)(void) setup_arch 早期钩子 需要在 memblock 之前动硬件的场合
init_irq void (*)(void) 中断控制器初始化 设备树方案填 irqchip_init
init_time void (*)(void) 定时器初始化 老式；设备树方案由 clocksource 驱动接手
init_machine void (*)(void) 板级设备注册 有 dts 表达不了的板级特例时
init_late void (*)(void) 晚期钩子 少用
handle_irq void (*)(struct pt_regs *) 中断入口，CONFIG_MULTI_IRQ_HANDLER 多中断控制器平台
restart void (*)(enum reboot_mode, const char *) 重启实现，赋给 arm_pm_restart 需要 reboot 正常工作的板子

五、平坦 dtb 上怎么走：of_scan_flat_dt 的状态机 #

这一节是整条链的技术核心。前面所有 early_init_dt_scan_* 都只是回调，真正读字节的是 of_scan_flat_dt()：

c
int __init of_scan_flat_dt(int (*it)(unsigned long node,
                                     const char *uname, int depth,
                                     void *data),
                           void *data)
{
    unsigned long p = ((unsigned long)initial_boot_params) +
        be32_to_cpu(initial_boot_params->off_dt_struct);
    int rc = 0;
    int depth = -1;

    do {
        u32 tag = be32_to_cpup((__be32 *)p);
        const char *pathp;

        p += 4;                 /* 跳过 tag */
        if (tag == OF_DT_END_NODE) {
            depth--;
            continue;
        }
        if (tag == OF_DT_NOP)
            continue;
        if (tag == OF_DT_END)
            break;
        if (tag == OF_DT_PROP) {
            u32 sz = be32_to_cpup((__be32 *)p);
            p += 8;             /* 跳过 len 和 nameoff */
            if (be32_to_cpu(initial_boot_params->version) < 0x10)
                p = ALIGN(p, sz >= 8 ? 8 : 4);
            p += sz;
            p = ALIGN(p, 4);
            continue;           /* 属性不回调，直接跳过 */
        }
        if (tag != OF_DT_BEGIN_NODE) {
            pr_err("Invalid tag %x in flat device tree!\n", tag);
            return -EINVAL;
        }
        depth++;
        pathp = (char *)p;
        p = ALIGN(p + strlen(pathp) + 1, 4);
        if (*pathp == '/')
            pathp = kbasename(pathp);
        rc = it(p, pathp, depth, data);
        if (rc != 0)
            break;
    } while (1);

    return rc;
}

四个易错点，全是实测踩出来的：

细节 为什么这么写 不看会怎样
depth 初值 -1 根节点前面没有 BEGIN_NODE 的父级，第一个 BEGIN_NODE 自增后正好是 0 初值写成 0，根节点的 depth 就变成 1，early_init_dt_scan_chosen 的 depth != 1 判断直接把根节点当成 chosen 的兄弟节点
回调收到的是属性区起始地址，不是节点起始地址 it(p, ...) 里 p 已经跳过节点名并对齐 回调里再按「节点 + 4」算偏移，会读到属性名中间
version < 0x10 时按 8 字节对齐 老版 dtspec 规定 8 字节以上属性值按 8 字节对齐 现代 dtb 都是 17，这段不生效；但读 15 及以下的旧 dtb 会全错位
*pathp == '/' 时取 kbasename 老版本存整条路径 /soc/serial@50000000，新版本只存单元名 不处理的话 strcmp(uname, "chosen") 永远不成立

kbasename 是个内核里的小工具（lib/string.c），返回路径最后一段。所以 /root/my_root 进来，回调收到的是 my_root。

返回值语义：回调返回非 0 就中断扫描。这就是为什么 early_init_dt_scan_chosen 找到节点后 return 1 的原因：chosen 只有一个，找到就收工。而 early_init_dt_scan_root 也是 return 1，根节点当然只有一个。

六、early_init_dt_scan：一开始只读三样东西 #

early_init_dt_scan() 是入口，先验 magic，再连扫三遍：

c
struct fdt_header *initial_boot_params;

bool __init early_init_dt_scan(void *params)
{
    if (!params)
        return false;

    initial_boot_params = params;

    if (be32_to_cpu(initial_boot_params->magic) != OF_DT_HEADER) {
        initial_boot_params = NULL;
        return false;
    }

    of_scan_flat_dt(early_init_dt_scan_chosen, boot_command_line);
    of_scan_flat_dt(early_init_dt_scan_root, NULL);
    of_scan_flat_dt(early_init_dt_scan_memory, NULL);
    return true;
}

三遍扫描，每次从头走一遍字节流。为什么不合成一遍？因为有依赖：解析 /memory 的 reg 需要知道根节点的 #address-cells / #size-cells，而这两个属性在根节点上，位置在 memory 之前也可能在后，顺序无法保证。分开扫，依赖关系就显式了。

扫描项 回调 结果落到
/chosen 的 bootargs early_init_dt_scan_chosen boot_command_line
根节点 #address-cells / #size-cells early_init_dt_scan_root dt_root_addr_cells / dt_root_size_cells
/memory 的 reg early_init_dt_scan_memory meminfo（交给 arm_add_memory）

先知道 /chosen 的 bootargs，才有 boot_command_line；先知道根节点 cells 宽度，才能正确解析 /memory 的 reg。

6.1 chosen：只收 bootargs 和 initrd #

c
int __init early_init_dt_scan_chosen(unsigned long node, const char *uname,
                                     int depth, void *data)
{
    unsigned long l;
    char *p;

    pr_debug("search \"chosen\", depth: %d, uname: %s\n", depth, uname);

    /* depth 必须是 1（根节点下），且节点名匹配；data 是 boot_command_line */
    if (depth != 1 || !data ||
        (strcmp(uname, "chosen") != 0 && strcmp(uname, "chosen@0") != 0))
        return 0;

    early_init_dt_check_for_initrd(node);

    p = of_get_flat_dt_prop(node, "bootargs", &l);
    if (p != NULL && l > 0)
        strlcpy(data, p, min((int)l, COMMAND_LINE_SIZE));

    pr_debug("Command line is: %s\n", (char*)data);
    return 1;
}

chosen 不代表任何硬件，只是固件和内核之间传数据的信箱。早期阶段它只被读两样：bootargs 和 initrd 地址。dts 里一般留空，由 bootloader 启动时填充：

dts
chosen {
    bootargs = "root=/dev/nfs rw nfsroot=192.168.1.1 console=ttyS0,115200";
    linux,initrd-start = <0x85500000>;
    linux,initrd-end   = <0x855a3212>;
};

initrd 的处理单独一个函数：

c
static void __init early_init_dt_check_for_initrd(unsigned long node)
{
    u64 start, end;
    unsigned long len;
    __be32 *prop;

    prop = of_get_flat_dt_prop(node, "linux,initrd-start", &len);
    if (!prop)
        return;
    start = of_read_number(prop, len/4);

    prop = of_get_flat_dt_prop(node, "linux,initrd-end", &len);
    if (!prop)
        return;
    end = of_read_number(prop, len/4);

    initrd_start = (unsigned long)__va(start);
    initrd_end   = (unsigned long)__va(end);
    /* 这条是关键：告诉后面「initrd 允许落在内核镜像下方」 */
    initrd_below_start_ok = 1;
}

initrd_below_start_ok = 1 是设备树带来的改动。ATAGS 时代，initrd 必须在内核之上，否则 reserve_initrd() 会认为参数错乱直接拒绝。设备树把地址写进了 chosen，内核无法靠位置推断合法性，只能信它，于是置位跳过检查。

6.2 根节点：只取两个 cells 宽度 #

c
int __init early_init_dt_scan_root(unsigned long node, const char *uname,
                                   int depth, void *data)
{
    __be32 *prop;

    if (depth != 0)
        return 0;

    dt_root_size_cells = OF_ROOT_NODE_SIZE_CELLS_DEFAULT;   /* 1 */
    dt_root_addr_cells = OF_ROOT_NODE_ADDR_CELLS_DEFAULT;   /* 1 */

    prop = of_get_flat_dt_prop(node, "#size-cells", NULL);
    if (prop)
        dt_root_size_cells = be32_to_cpup(prop);

    prop = of_get_flat_dt_prop(node, "#address-cells", NULL);
    if (prop)
        dt_root_addr_cells = be32_to_cpup(prop);

    return 1;
}

两个全局变量，加两个默认值宏（都是 1）。默认值是 1 而不是报错，这是个历史包袱：老 dtb 不写这两个属性，全靠默认值撑。写新 dts 一定要显式写出来，否则 64 位平台上会被当成 32 位解析，reg 全部错位。

6.3 memory：usable-memory 优先于 reg #

c
int __init early_init_dt_scan_memory(unsigned long node, const char *uname,
                                     int depth, void *data)
{
    char *type = of_get_flat_dt_prop(node, "device_type", NULL);
    __be32 *reg, *endp;
    unsigned long l;

    if (type == NULL) {
        /* 没有 device_type 的老 dtb，只认 depth==1 且名字恰好为 memory@0 的节点 */
        if (depth != 1 || strcmp(uname, "memory@0") != 0)
            return 0;
    } else if (strcmp(type, "memory") != 0)
        return 0;

    /* 先看 linux,usable-memory，没有再退回 reg */
    reg = of_get_flat_dt_prop(node, "linux,usable-memory", &l);
    if (reg == NULL)
        reg = of_get_flat_dt_prop(node, "reg", &l);
    if (reg == NULL)
        return 0;

    endp = reg + (l / sizeof(__be32));
    while ((endp - reg) >= (dt_root_addr_cells + dt_root_size_cells)) {
        u64 base, size;

        base = dt_mem_next_cell(dt_root_addr_cells, &reg);
        size = dt_mem_next_cell(dt_root_size_cells, &reg);
        if (size == 0)
            continue;

        early_init_dt_add_memory_arch(base, size);
    }
    return 0;
}

两个值得记的点：

一、linux,usable-memory 优先。 这个属性是「内核实际能用的内存」，可以和物理上真实存在的范围不同。典型场景：板子上焊了 1GB，但 bootloader 或安全固件只允许内核用其中 768MB。这时 reg 写 1GB 描述硬件事实，linux,usable-memory 写 768MB 描述可用范围。选错会导致内核试图管理不该碰的物理页，随机 panic。

二、memory@0 这个老兜底。 现代 dtb 都写 device_type = "memory"，这个分支永远不走。但读 2011 年前后的 dtb 会遇到，现象是「memory 节点明明在，meminfo 却是 0」，内核报 No memory 直接挂。

6.4 内存信息现在有三种来源 #

来源 谁写的 现状
ATAG_MEM U-Boot 的 bi_boot_params 老方案，仍在用
command line 的 mem= bootargs 调试用，覆盖前两者
dtb 的 /memory 节点 dts 推荐

三者可以共存，优先级从高到低是 mem= > dtb > ATAG。调试时用 mem=256M 临时砍内存，是排查「高地址内存有问题」最快的一招。

七、读属性的三层封装：of_fdt_get_property 与字符串块 #

节点地址到手之后，读属性还得再走一层。三个函数套娃，每层只管一件事：

c
void *__init of_get_flat_dt_prop(unsigned long node, const char *name,
                                 unsigned long *size)
{
    return of_fdt_get_property(initial_boot_params, node, name, size);
}

第一层只是补上全局的 initial_boot_params，让调用方不用到处传 blob。

c
void *of_fdt_get_property(struct fdt_header *blob, unsigned long node,
                          const char *name, unsigned long *size)
{
    unsigned long p = node;

    do {
        u32 tag = be32_to_cpup((__be32 *)p);
        u32 sz, noff;
        const char *nstr;

        p += 4;
        if (tag == OF_DT_NOP)
            continue;
        if (tag != OF_DT_PROP)
            return NULL;            /* 属性区结束，没找到 */

        sz   = be32_to_cpup((__be32 *)p);
        noff = be32_to_cpup((__be32 *)(p + 4));
        p += 8;

        if (be32_to_cpu(blob->version) < 0x10)
            p = ALIGN(p, sz >= 8 ? 8 : 4);

        nstr = of_fdt_get_string(blob, noff);
        if (nstr == NULL) {
            pr_warning("Can't find property index name !\n");
            return NULL;
        }

        if (strcmp(name, nstr) == 0) {
            if (size)
                *size = sz;
            return (void *)p;       /* 返回属性值地址 */
        }

        p += sz;
        p = ALIGN(p, 4);
    } while (1);
}

第二层是真正的遍历。注意 return NULL 的两个位置语义完全不同：tag != OF_DT_PROP 是正常结束（这个节点的属性读完了，没这个属性），nstr == NULL 是数据损坏（属性名偏移越界）。但两者都返回 NULL，调用方分不出来。这是老代码的一个设计缺陷，紧着用。

第三层就是字符串块解引用：

c
char *of_fdt_get_string(struct fdt_header *blob, u32 offset)
{
    return ((char *)blob) + be32_to_cpu(blob->off_dt_strings) + offset;
}

一行。属性名的本体不在结构块里，而是集中存在字符串块中，节点属性里只存一个 nameoff 偏移。这么设计是因为属性名重复率极高（reg、compatible、status 到处都有），去重能省掉整个 dtb 体积的两三成。

还有一个小函数 of_read_number()，负责把 N 个 big-endian cell 拼成一个 u64：

c
static inline u64 of_read_number(const __be32 *cell, int size)
{
    u64 r = 0;
    while (size--)
        r = (r << 32) | be32_to_cpu(*(cell++));
    return r;
}

size 是 cell 个数，不是字节数。所以调用处都是 len/4。0x85500000 这样的值在 dtb 里是 85 50 00 00 四个字节，be32_to_cpu 之后才是 0x85500000。dtb 全程大端，这是 FDT 规范定死的，与 CPU 字节序无关。

八、dtb 自身不能被覆盖 #

dtb 被 U-Boot 下载到某块内存（Mini2440 常用 0x30001000，紧贴内核 0x30008000 下方），内核初始化内存管理时有可能把这块当空闲内存用掉，所以：

c
start_kernel
  setup_arch
    arm_memblock_init(mdesc)
      early_init_fdt_reserve_self();

内部执行：

c
early_init_dt_reserve_memory_arch(__pa(initial_boot_params),
                                  fdt_totalsize(initial_boot_params), 0);

即把 dtb 占用区 memblock_reserve 掉。另有 early_init_fdt_scan_reserved_mem() 处理 dts 里的 /memreserve/ 条目。

这块要小心的是地址重叠。dtb 放在内核镜像下方 28KB 处时，如果 dtb 因为加了节点膨胀到超过这个间距，bootm 解压内核就会把 dtb 覆盖掉。现象是内核报 bad magic 或者 Invalid device tree blob header，但 dtb 文件本身是好的。查法：bootm 前后各 md 一次 dtb 地址的头 16 字节，对比 magic 有没有变。

九、unflatten：平坦 dtb 变成 device_node 树 #

unflatten_device_tree() 是分界线。跑到这里说明内存分配器已经能用，可以把字节流「物化」成结构体了。

c
void __init unflatten_device_tree(void)
{
    __unflatten_device_tree(initial_boot_params, &of_allnodes,
                            early_init_dt_alloc_memory_arch);
    of_alias_scan(early_init_dt_alloc_memory_arch);
}

9.1 为什么要扫两遍 #

核心函数 __unflatten_device_tree()：

c
static void __unflatten_device_tree(struct fdt_header *blob,
                                    struct device_node **mynodes,
                                    void * (*dt_alloc)(u64 size, u64 align))
{
    unsigned long size;
    void *start, *mem;
    struct device_node **allnextp = mynodes;

    if (!blob) {
        pr_debug("No device tree pointer\n");
        return;
    }

    if (be32_to_cpu(blob->magic) != OF_DT_HEADER) {
        pr_err("Invalid device tree blob header\n");
        return;
    }

    start = ((void *)blob) + be32_to_cpu(blob->off_dt_struct);

    /* 第一遍：mem=0、allnextpp=NULL，只累加尺寸，不落地 */
    size = (unsigned long)unflatten_dt_node(blob, 0, &start, NULL, NULL, 0);
    size = ALIGN(size, 4);

    /* 一次性申请，多给 4 字节放哨兵 */
    mem = dt_alloc(size + 4, __alignof__(struct device_node));
    memset(mem, 0, size);
    *(__be32 *)(mem + size) = cpu_to_be32(0xdeadbeef);

    /* 第二遍：真正填结构体 */
    start = ((void *)blob) + be32_to_cpu(blob->off_dt_struct);
    unflatten_dt_node(blob, mem, &start, NULL, &allnextp, 0);

    if (be32_to_cpup(start) != OF_DT_END)
        pr_warning("Weird tag at end of tree: %08x\n", be32_to_cpup(start));
    if (be32_to_cpup(mem + size) != 0xdeadbeef)
        pr_warning("End of tree marker overwritten: %08x\n", be32_to_cpup(mem + size));
    *allnextp = NULL;
}

两遍扫描的理由很直接：这棵树要多少内存，只有走完才知道。节点数、属性数、full_name 长度全都依赖内容，事先算不出来。所以第一遍把 mem 传 0、allnextpp 传 NULL，函数内部所有「写」都被跳过，只累加偏移量，返回值就是要的字节数。第二遍带着真的内存地址再走一次，这次才写。

0xdeadbeef 那个哨兵是越界探针：申请 size+4，最后 4 字节写个魔数。第二遍写完回头看一眼，如果魔数被改了，说明写超了。这招比 address sanitizer 早二十年，成本几乎为零。

9.2 unflatten_dt_node 逐段看 #

这是全链最绕的一个函数。它的核心难点在于：dtb 里节点只存名字，不存路径，而 device_node 需要 full_name（完整路径）。所以它得一边递归一边把父路径传下去，还要精确算出需要多大空间。

c
static void * unflatten_dt_node(struct fdt_header *blob,
                                void *mem, void **p,
                                struct device_node *dad,
                                struct device_node ***allnextpp,
                                unsigned long fpsize)
{
    struct device_node *np;
    struct property *pp, **prev_pp = NULL;
    char *pathp;
    u32 tag;
    unsigned int l, allocl;
    int has_name = 0;
    int new_format = 0;

    tag = be32_to_cpup(*p);
    if (tag != OF_DT_BEGIN_NODE) {
        pr_err("Weird tag at start of node: %x\n", tag);
        return mem;
    }
    *p += 4;
    pathp = *p;
    l = allocl = strlen(pathp) + 1;
    *p = PTR_ALIGN(*p + l, 4);

    if ((*pathp) != '/') {
        new_format = 1;
        if (fpsize == 0) {
            /* 根节点：没有父路径可拼 */
            fpsize = 1;
            allocl = 2;
            l = 1;
            *pathp = '\0';
        } else {
            /* 子节点：父路径长度 + 本节点名长度 */
            fpsize += l;
            allocl = fpsize;
        }
    }

    np = unflatten_dt_alloc(&mem, sizeof(struct device_node) + allocl,
                            __alignof__(struct device_node));
    ...

几个关键设计：

设计 实现 为什么这么干
full_name 内联在结构体尾部 np->full_name = fn = ((char *)np) + sizeof(*np); 一次分配、一次 free（这内存压根不 free），少一个指针追踪，也省一份 malloc 元数据
fpsize 逐层累加 父节点的 fpsize 作为参数传给递归调用 子节点不知父路径有多长，只能靠参数传
第一次调用 fpsize == 0 把 pathp 首个字符改成 '\0' 根节点没有名字，写空串；l = 1 刚好放下那个 '\0'
new_format 判断 *pathp == '/'（老格式存整路径）还是单元名（新格式） 版本 16 之前存路径，之后存单元名
allnextpp 双指针 **allnextpp = np; *allnextpp = &np->allnext; 尾插法。*allnextpp 始终指向「最后一个节点的 allnext 字段的地址」，下次直接写进去

父子关系那一段最容易被忽略：

c
    if (dad != NULL) {
        np->parent = dad;
        if (dad->next == NULL)   /* 第一个孩子 */
            dad->child = np;
        else
            dad->next->sibling = np;
        dad->next = np;
    }

代码临时把 dad->next 当成「最后一个孩子」指针在用。这么干是因为 device_node 里没有专门的 last_child 字段，而 unflatten 阶段 next 字段（「同类型节点的下一个」）还没被用到，先借来当临时变量，建完树自然就不用它了。省一个字段的代价是可读性，读代码时很容易被 dad->next 误导成「遍历同类型节点」。

属性遍历也有一处补丁逻辑。dtb 里节点不一定有 name 属性（新格式不存），但内核很多地方依赖 np->name，所以 has_name 为 0 时现场构造一个：

c
    if (!has_name) {
        char *p1 = pathp, *ps = pathp, *pa = NULL;
        int sz;

        while (*p1) {
            if ((*p1) == '@')
                pa = p1;
            if ((*p1) == '/')
                ps = p1 + 1;
            p1++;
        }
        if (pa < ps)
            pa = p1;
        sz = (pa - ps) + 1;
        pp = unflatten_dt_alloc(&mem, sizeof(struct property) + sz,
                                __alignof__(struct property));
        if (allnextpp) {
            pp->name = "name";
            pp->length = sz;
            pp->value = pp + 1;
            *prev_pp = pp; prev_pp = &pp->next;
            memcpy(pp->value, ps, sz - 1);
            ((char *)pp->value)[sz - 1] = 0;
        }
    }

ps 越过最后一个 / 指向节点名开头，pa 指向 @，sz = pa - ps + 1，正好是去掉 @unit-address 之后的名字长度。所以 serial@50000000 的 np->name 是 serial，@ 后缀丢掉。这也是为什么驱动侧匹配用 of_node_name_eq() 而不是直接 strcmp。

phandle 有三种写法，都认：

c
    if ((strcmp(pname, "phandle") == 0) || (strcmp(pname, "linux,phandle") == 0)) {
        if (np->phandle == 0)
            np->phandle = be32_to_cpup((__be32*)*p);
    }
    if (strcmp(pname, "ibm,phandle") == 0)
        np->phandle = be32_to_cpup((__be32 *)*p);

phandle 是标准写法，linux,phandle 是 Linux 早期私货，ibm,phandle 是 PowerPC 遗留。np->phandle == 0 那个判断保证标准写法优先。所以读代码时「为什么不是直接覆盖」的答案是：同一个节点可能同时带两个属性，标准那个先到就不让私货覆盖。

9.3 分配器就是个 bump allocator #

c
static void *unflatten_dt_alloc(void **mem, unsigned long size,
                                unsigned long align)
{
    void *res;

    *mem = PTR_ALIGN(*mem, align);
    res = *mem;
    *mem += size;
    return res;
}

十六行不到。因为整棵树是一次性分配的，所有子分配只需要「对齐、取地址、指针前移」，不需要记录、不需要 free。第一次调用（mem = 0）时它返回的是偏移量，正好就是第一遍要算的总大小。

9.4 device_node 的全貌 #

上面内联的 full_name、临时借用的 next，都在这两个结构体里：

c
struct device_node {
    const char *name;               /* 节点名，不含 @地址 */
    const char *type;               /* device_type 属性值 */
    phandle phandle;
    const char *full_name;          /* 带 @unit-address 的完整路径 */
    struct property *properties;    /* 属性链表 */
    struct property *deadprops;     /* 已被删除的属性 */
    struct device_node *parent;
    struct device_node *child;      /* 第一个孩子 */
    struct device_node *sibling;    /* 下一个兄弟 */
    struct device_node *next;       /* 同类型节点的下一个（也临时当过 last-child） */
    struct device_node *allnext;    /* of_allnodes 链表的下一个 */
    struct proc_dir_entry *pde;     /* /proc/device-tree 条目 */
    struct kref kref;
    unsigned long _flags;
    void *data;
#if defined(CONFIG_SPARC)
    const char *path_component_name;
    unsigned int unique_id;
    struct of_irq_controller *irq_trans;
#endif
};

struct property {
    char *name;             /* 指向字符串块，不单独分配 */
    int length;
    void *value;            /* 指向 dtb 里的属性值，不拷贝 */
    struct property *next;
    unsigned long _flags;
    unsigned int unique_id;
};

两个「不拷贝」值得单独说：property->name 和 property->value 都直接指向 dtb 本体，没有复制。所以整个 unflatten 之后，dtb 那块内存仍然必须保持有效。这就是第八章 early_init_fdt_reserve_self() 存在的根本原因。dtb 被覆盖 = 整棵设备树的属性值全部变成垃圾。

deadprops 是动态修改设备树的产物：of_remove_property() 不真删（因为内存不能 free），而是把属性摘下来挂到 deadprops 链表上。

顺带把三个全局变量填了：of_chosen、of_stdout、of_aliases。linux,stdout-path 决定 console= 不写时默认往哪个串口输出。早期启动日志跑到哪个口，就看它。

alias 的解析逻辑有点绕，原文注释写得比代码还难懂：

dts
aliases {
    serial0 = &uart0;
    serial1 = &uart1;
    ethernet0 = &eth0;
};

属性名 serial0 里，serial 是 stem（词干），0 是 id。解析时：

c
    /* 跳过 name / phandle 这类非 alias 属性 */
    if (!strcmp(pp->name, "name") ||
        !strcmp(pp->name, "phandle") ||
        !strcmp(pp->name, "linux,phandle"))
        continue;

    np = of_find_node_by_path(pp->value);
    if (!np)
        continue;

    /* 从名字尾部往前剥数字，end 最终指向数字开头 */
    while (isdigit(*(end-1)) && end > start)
        end--;

    len = end - start;              /* 词干长度 */

    if (kstrtoint(end, 10, &id) < 0)
        continue;

    /* 一次分配 alias_prop + 词干字符串 */
    ap = dt_alloc(sizeof(*ap) + len + 1, 4);
    memset(ap, 0, sizeof(*ap) + len + 1);
    ap->device_node = np;
    ap->stem = (char *)ap + sizeof(*ap);
    memcpy(ap->stem, start, len);
    ap->stem[len] = '\0';
    of_alias_add(ap, np, id, start, len);

while (isdigit(*(end-1)) && end > start) end--; 这一行是全函数最难读的：end 初值是 start + strlen(start)，指向 '\0'。往前退一格是最后那个数字 '0'，是数字就继续退，退到字母 'l' 停下，此时 end 指向 '0'，len = end - start = 6（serial 正好 6 个字符）。然后 kstrtoint(end, 10, &id) 从 '0' 开始解析得到 id = 0。

of_alias_add() 把结果按 stem 和 id 排序插进全局 aliases_lookup 链表。之后驱动调 of_alias_get_id(np, "serial") 就能拿到 0，这是 serial0 / ttyS0 编号对应关系的来源。没有 aliases 节点，串口编号就不稳定。同型号两块板子，内核可能把它枚举成 ttyS0 和 ttyS3，脚本里写死的 /dev/ttyS0 直接失效。

9.6 运行时怎么直接看这棵树 #

bash
ls /sys/firmware/devicetree/base/
cat /sys/firmware/devicetree/base/model
cat /proc/device-tree/mykey/interrupts | hexdump -C

/sys/firmware/devicetree/base/ 和 /proc/device-tree/ 是同一棵树的两个挂载点（5.x 之后前者是官方的，后者是兼容软链）。目录即节点，文件即属性，文件名就是属性名。属性值是裸字节，所以看整数要 hexdump。这也是最省工具的调试手段，改完 dtb 不用重编内核，scp 上去重启就能验。

十、initcall 阶段：device_node 变成 platform_device #

10.1 initcall 的八个等级 #

rest_init → kernel_init → kernel_init_freeable → do_basic_setup → do_initcalls：

c
static const char *initcall_level_names[] __initdata = {
    "pure", "core", "postcore", "arch", "subsys", "fs", "device", "late",
};

每级由链接脚本里的 __initcallN_start / __initcallN_end 界定（System.map 里能直接看到）。设备树的「节点转设备」动作排在 arch 级同步：

c
/* drivers/of/platform.c */
arch_initcall_sync(of_platform_default_populate_init);

排这么早是有讲究的：设备必须在驱动之前到位。arch 级之后就轮到 subsys（各子系统初始化）和 device（驱动 probe），此时 /sys/bus/platform/devices 里必须已经有设备条目，否则驱动的 driver_register 找不到可绑定的对象。

10.2 哪些节点会变成 platform_device #

of_platform_populate() 只认三类：

情况 是否转换
根节点的子节点，且含 compatible 是（根节点自身即便有 compatible 也跳过）
含 "simple-bus" / "simple-mfd" / "isa" / "arm,amba-bus" 的节点的子节点（子节点须有 compatible） 是，递归展开
i2c / spi 控制器下的子节点 否，不转 platform_device
其它（只有 reg 没有 compatible） 否

最后一条最容易被误解：i2c 总线的子设备最终由父节点的 platform_driver 在 probe 里创建成 i2c_client，spi 同理成 spi_device。所以「设备树里写了就是从设备」这句话只对 platform 总线成立。

顺带一个常被问的问题：为什么 simple-bus 里的子节点会递归展开，而 i2c@... 下的不会？因为 of_default_bus_match_table 里只有那四类 compatible。想让某个总线也递归，要么改这张表，要么在总线驱动的 probe 里自己调 of_platform_populate()，i2c 核心走的就是后者。

十一、of_device_alloc：资源是怎么凑出来的 #

c
struct platform_device *of_device_alloc(struct device_node *np,

{
        struct platform_device *dev;

        dev = platform_device_alloc("", PLATFORM_DEVID_NONE);

        /* 数一数这个节点有几段 reg */
        while (of_address_to_resource(np, num_reg, &temp_res) == 0)
                num_reg++;

        /* 数一数这个节点有几个中断 */
        num_irq = of_irq_count(np);

        if (num_irq || num_reg) {
                res = kcalloc(num_irq + num_reg, sizeof(*res), GFP_KERNEL);
                dev->num_resources = num_reg + num_irq;
                dev->resource = res;

                for (i = 0; i < num_reg; i++, res++)
                        rc = of_address_to_resource(np, i, res);

                if (of_irq_to_resource_table(np, res, num_irq) != num_irq)
                        pr_debug("not all legacy IRQ resources mapped\n");
        }

        dev->dev.of_node = of_node_get(np);
        dev->dev.fwnode = &np->fwnode;
        dev->dev.parent = parent ? : &platform_bus;
        return dev;
}

关键在开头那个 while：用「试到失败为止」的方式数 reg 段数。of_address_to_resource() 返回非 0 就说明第 num_reg 段不存在，此时 num_reg 正好是总数。没有专门的 of_address_count()，因为「试探调用一次」比「再实现一个遍历函数」更省。

于是 dts 里一句 reg = <0x30000000 0x4000000>; 加两条 interrupts，到了驱动侧就是 platform_get_resource(pdev, IORESOURCE_MEM, 0) 和 platform_get_irq(pdev, 0)。

资源数组的排列顺序是先所有 mem，再所有 irq，这一点必须记住。所以 platform_get_resource(pdev, IORESOURCE_MEM, 1) 取的是第二段 reg，而 platform_get_irq(pdev, 0) 内部按 IORESOURCE_IRQ 类型过滤，两者不冲突。但如果驱动直接按下标遍历 pdev->resource[]，就必须自己判类型。

十二、版本差异与读老资料的坑 #

设备树这块代码在 3.10 ~ 5.x 之间动过好几轮，老资料里的函数名很多已经不存在了：

3.10 ~ 3.14 5.x / 6.x 影响
struct boot_param_header struct fdt_header（3.10 改名） 字段名也从 dt_strings_size/dt_struct_size 改成 size_dt_strings/size_dt_struct
of_fdt_get_property() 在 drivers/of/fdt.c 迁入 lib/fdt.c，Linux 内部函数改名 fdt_getprop_namelen() 老文章里的调用链对不上
initial_boot_params 类型 struct boot_param_header * const void * 直接取字段的代码全部编译不过
of_scan_flat_dt() 保留，另加 of_scan_flat_dt_subnodes() 老函数签名没变，但新增了子节点专扫版本
setup_machine_fdt 直接返回 mdesc 多了 early_init_dt_verify 校验 magic 3.14 之后 magic 校验提前到 setup_machine_fdt 里
显式调用 of_platform_populate(NULL, of_default_bus_match_table, ...) of_platform_default_populate_init 自动执行 板级代码里不用再手动调
struct device 里只有 of_node 多了 fwnode 抽象层（ACPI/DT 统一） dev->of_node 仍可用，但内部统一走 fwnode
of_platform_populate() 是主入口 被 of_platform_default_populate() 取代 前者仍存在，语义变了（默认只从根节点开始）
initcall 7 级 8 级（多 early 级） System.map 里符号名不同
memory@0 兜底分支 已删除 现代内核不认没有 device_type 的 memory 节点

还有一条与版本无关、但读任何年代的资料都会碰到的：unflatten_dt_node 在 3.14 和 4.x 之间被拆成了 populate_node + populate_properties 两个函数。所以你看 3.x 的资料是一大坨，看 5.x 的资料是两截，内容等价，别当成两个阶段。
