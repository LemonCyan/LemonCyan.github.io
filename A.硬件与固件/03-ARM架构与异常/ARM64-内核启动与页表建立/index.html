零、先立住四个地址宏 #

不把下面这几个量分清，后面的汇编一行都读不通。

c
/* arch/arm64/include/asm/memory.h */
#define VA_BITS        (CONFIG_ARM64_VA_BITS)        /* 可选 36 / 39 / 42 / 47 / 48 */
#define PAGE_OFFSET    (UL(0xffffffffffffffff) << (VA_BITS - 1))

/* arch/arm64/kernel/head.S 与 vmlinux.lds.S */
#define KERNEL_START   _text
#define __PHYS_OFFSET  (KERNEL_START - TEXT_OFFSET)

. = PAGE_OFFSET + TEXT_OFFSET;
.head.text : { _text = .; HEAD_TEXT }
宏 含义 48 位 VA 时的值
VA_BITS 虚拟地址位宽 48
PAGE_OFFSET 线性映射区的起始虚拟地址（内核空间的基准） 0xffff8000_00000000
TEXT_OFFSET 内核 image 在起始处的保留偏移 0x80000（512 KB）
KERNEL_START 等于 _text，链接期的内核入口虚拟地址 0xffff8000_00080000
__PHYS_OFFSET 内核 image 的物理地址（名字有误导，见下） 0x00080000

两个必须澄清的点：

  1. __PHYS_OFFSET 名字里的 PHYS 是历史包袱。它在链接期是个虚拟地址常量，MMU 关着时靠 adrp 取值，得到的就是物理地址。它也不等于「内核物理基址」——从 __PHYS_OFFSET 到 KERNEL_START 之间那 512KB 是 TEXT_OFFSET 保留区，真正的 image 从后面才开始。
  2. TEXT_OFFSET 为什么在 32 位是 0x8000（32KB）、64 位反而涨到 0x80000（512KB）：32 位时代这块地方放进程 0 的 PGD 和 tag 参数，需求小；64 位页表一张就是 4KB、启动期要连着放好几个，又要求 2MB 对齐，512KB 才够周转。

还有个少见的配置项值得提一句：

makefile
ifeq ($(CONFIG_ARM64_RANDOMIZE_TEXT_OFFSET), y)
TEXT_OFFSET := $(shell awk 'BEGIN {srand(); printf "0x%03x000\n", int(512 * rand())}')
else
TEXT_OFFSET := 0x00080000
endif

它不是为了安全（真随机化用的是 KASLR），而是测试 bootloader 到底会不会读 image header 里的 text_offset 字段。大家都硬编码 0x80000，导致这个特性常年无人验证——把它随机一下，不对的 bootloader 立刻暴露。

一、bootloader 交给内核的是什么 #

进 stext 之前，bootloader 必须做到：

  1. 初始化 RAM，并把内存信息告诉内核；
  2. 准备好设备树 blob，把它的物理地址放进 x0；
  3. 可选地解压内核；
  4. 移交控制权。

交接时的硬约定：

text
MMU     = off
D-cache = off
I-cache = on 或 off 都行
x0 = FDT blob 的物理地址
x1 ~ x3 = 0        （非零是留给将来用的，内核会校验）

「D-cache off」不是只关 L1，而是各个 level 的 data cache 与 unified cache 全关。MMU 关着为什么必须关 D-cache？因为没有页表描述符，就没法表达 memory type / shareability / cacheability。硬件并不禁止「MMU off + D-cache on」这种组合，此时它强制把所有内存当作 fixed 属性：

text
memory type = normal, Non-shareable
inner / outer cache 策略 = Write-Back, Read-Write Allocate

完整协议见 Documentation/arm64/booting.txt。

二、stext 入口四连 #

armasm
ENTRY(stext)
    bl  preserve_boot_args
    bl  el2_setup              // 降到 EL1，w20 记录从哪个 EL 来的
    adrp    x24, __PHYS_OFFSET
    bl  set_cpu_boot_mode_flag
    bl  __vet_fdt
    ...
ENDPROC(stext)

2.1 preserve_boot_args：把 x0~x3 存下来 #

armasm
preserve_boot_args:
    mov  x21, x0                 // 先把 dtb 物理地址挪走，腾出 x0
    adr_l   x0, boot_args
    stp  x21, x1, [x0]           // boot_args[0] = dtb, [1] = x1
    stp  x2, x3, [x0, #16]       // boot_args[2] = x2,  [3] = x3
    dmb  sy
    add  x1, x0, #0x20           // 4 × 8 = 32 字节
    b    __inval_cache_range

三个细节：

  • 为什么用 adr_l 而不是 adrp：adrp 以 4KB 为粒度（低 12 位恒为 0），boot_args 不是页对齐的，直接用 adrp 会丢低位。adr_l = adrp + 低 12 位补偿。
  • 为什么要 dmb sy：MMU 和 D-cache 都关着，写 RAM 绕过 cache，而各级 cache 里可能残留 bootloader 时期读过的旧行。ARM ARM 原文明确说：除了 DC ZVA 之外的 data cache 指令，与访问 Device 内存、Inner Non-cacheable 内存的 load/store 之间可以任意重排，除非中间有 DMB 或 DSB。
  • 为什么要 invalidate：写完再刷一次，保证下次读走内存而不是命中旧 cacheline。

2.2 el2_setup：从 EL2 降到 EL1 #

它分两段。第一段判断当前在哪一级、定 endianness、开定时器、配 GICv3：

armasm
ENTRY(el2_setup)
    mrs x0, CurrentEL              // 从 PSTATE 读当前 EL
    cmp x0, #CurrentEL_EL2
    b.ne    1f
    mrs x0, sctlr_el2              // ---- 在 EL2 ----
CPU_BE( orr x0, x0, #(1 << 25) )   // 置 EE
CPU_LE( bic x0, x0, #(1 << 25) )   // 清 EE
    msr sctlr_el2, x0
    b   2f
1:  mrs x0, sctlr_el1              // ---- 已在 EL1 ----
CPU_BE( orr x0, x0, #(3 << 24) )   // 置 EE 和 E0E
CPU_LE( bic x0, x0, #(3 << 24) )   // 清 EE 和 E0E
    msr sctlr_el1, x0
    mov w20, #BOOT_CPU_MODE_EL1
    isb
    ret
2:  mov x0, #(1 << 31)             // HCR_EL2.RW = 1
    msr hcr_el2, x0
    mrs x0, cnthctl_el2
    orr x0, x0, #3                 // 允许 EL1 访问物理 counter/timer
    msr cnthctl_el2, x0
    msr cntvoff_el2, xzr           // 虚拟 counter 偏移清零
    mrs x0, id_aa64pfr0_el1
    ubfx    x0, x0, #24, #4        // GIC 支持位
    cmp x0, #1
    b.ne    3f
    mrs_s   x0, ICC_SRE_EL2
    orr x0, x0, #ICC_SRE_EL2_SRE   // 走 system register 访问 GIC
    orr x0, x0, #ICC_SRE_EL2_ENABLE
    msr_s   ICC_SRE_EL2, x0
    isb
    msr_s   ICH_HCR_EL2, xzr       // 关掉虚拟中断
3:

三处容易看漏：

  • sctlr_el2.EE（bit25）不只管 EL2 自己的数据访问，还管 EL1/EL0 的 stage2 翻译的 endianness；sctlr_el1 的 EE 管 EL1 数据访问与 EL1/EL0 stage1，E0E 专管 EL0。
  • 改 system control register 之后必须 isb：既确认前面的 MSR 已生效，又让后面的指令重新取指。
  • hcr_el2.bit31 是 RW（Register Width），复位值是 implementation defined，所以必须显式置 1 才能保证 EL1 跑在 AArch64 上。

第二段是「伪装成 EL1」的部分：

armasm
    mrs x0, midr_el1
    mrs x1, mpidr_el1
    msr vpidr_el2, x0              // 让 EL1 读到真实 MIDR
    msr vmpidr_el2, x1

    mov x0, #0x0800                // bit11 是 RES1，必须写 1
CPU_BE( movk x0, #0x33d0, lsl #16 )    // → 0x33d00800
CPU_LE( movk x0, #0x30d0, lsl #16 )    // → 0x30d00800
    msr sctlr_el1, x0

    mov x0, #0x33ff
    msr cptr_el2, x0               // 关掉到 EL2 的协处理器 trap
#ifdef CONFIG_COMPAT
    msr hstr_el2, xzr              // 关掉到 EL2 的 CP15 trap
#endif

    mrs x0, pmcr_el0
    ubfx    x0, x0, #11, #5        // PMCR_EL0[15:11] = 计数器个数
    msr mdcr_el2, x0               // 允许 EL1 用全部 PMU counter
    msr vttbr_el2, xzr             // 清 stage-2 页表基址

    adrp    x0, __hyp_stub_vectors
    add x0, x0, #:lo12:__hyp_stub_vectors
    msr vbar_el2, x0

    mov x0, #(PSR_F_BIT | PSR_I_BIT | PSR_A_BIT | PSR_D_BIT | PSR_MODE_EL1h)
    msr spsr_el2, x0
    msr elr_el2, lr
    mov w20, #BOOT_CPU_MODE_EL2
    eret

eret 本质上是模拟一次异常返回：SPSR_EL2 灌进 PSTATE、ELR_EL2 灌进 PC，CPU 降回 EL1。这里 ELR_EL2 填的是函数返回地址 lr，所以 eret 之后就像从 el2_setup 正常返回一样，只不过已经在 EL1 了。

为什么只有 el2_setup 没有 el3_setup：SOC 上电进 EL3 时，接管的是 Secure Platform Firmware（secure monitor），它初始化硬件、加载 trusted OS，然后把控制权交给 non-secure world（多半是 EL2，不支持虚拟化就是 EL1）。Linux 永远看不到 secure world，自然不需要处理 EL3。

2.3 set_cpu_boot_mode_flag：全系统必须一致 #

armasm
ENTRY(set_cpu_boot_mode_flag)
    adr_l   x1, __boot_cpu_mode
    cmp w20, #BOOT_CPU_MODE_EL2
    b.ne    1f
    add x1, x1, #4                 // 是 EL2 → 写第二个 long
1:  str w20, [x1]
    dmb sy
    dc  ivac, x1                   // invalidate 这一行，防止残留旧值
    ret
ENDPROC(set_cpu_boot_mode_flag)

ENTRY(__boot_cpu_mode)
    .long   BOOT_CPU_MODE_EL2
    .long   BOOT_CPU_MODE_EL1

__boot_cpu_mode 初始值是 {EL2, EL1}。每个 CPU 都跑一遍这个函数，一旦有两个 CPU 报的模式不一样，数组就变成 {EL2, EL1} 之外的值——内核据此判定「全系统 CPU 必须从同一 EL 启动」，混用是不允许的。全 EL2 说明支持虚拟化，KVM 才起得来。

2.4 __vet_fdt：三条合法性检查 #

armasm
__vet_fdt:
    tst x21, #0x7                  // ① 8 字节对齐
    b.ne    1f
    cmp x21, x24                   // ② >= __PHYS_OFFSET
    b.lt    1f
    mov x0, #(1 << 29)             // 512 MB
    add x0, x0, x24
    cmp x21, x0                    // ③ < __PHYS_OFFSET + 512MB
    b.ge    1f
    ret
1:  mov x21, #0                    // 不合法就清零，后续按「无 FDT」处理
    ret

三、__create_page_tables：一次建三张映射 #

3.1 页表放在哪、有多大 #

位置在 bss 段之后，连续排列：

. = ALIGN(PAGE_SIZE);
idmap_pg_dir = .;
. += IDMAP_DIR_SIZE;
swapper_pg_dir = .;
. += SWAPPER_DIR_SIZE;

早先的方案是放在 PHYS_OFFSET 到 PHYS_OFFSET + TEXT_OFFSET（0x80000）之间，沿用 ARM32 习惯。废弃的原因是：这段可能正被 bootloader 用来传参，而当时 memory block 子系统还没 ready，没法用 memblock_reserve 标记保护，建页表会直接覆盖掉有用数据。搬到 bss 之后就没这个问题了。

大小由「section map 用几级」决定：

c
#if ARM64_SWAPPER_USES_SECTION_MAPS
#define SWAPPER_PGTABLE_LEVELS  (CONFIG_PGTABLE_LEVELS - 1)
#define IDMAP_PGTABLE_LEVELS    (ARM64_HW_PGTABLE_LEVELS(PHYS_MASK_SHIFT) - 1)
#else
#define SWAPPER_PGTABLE_LEVELS  (CONFIG_PGTABLE_LEVELS)
#define IDMAP_PGTABLE_LEVELS    (ARM64_HW_PGTABLE_LEVELS(PHYS_MASK_SHIFT))
#endif
#define SWAPPER_DIR_SIZE    (SWAPPER_PGTABLE_LEVELS * PAGE_SIZE)
#define IDMAP_DIR_SIZE      (IDMAP_PGTABLE_LEVELS * PAGE_SIZE)

48 位 VA + 4KB 页时地址切成 9+9+9+9+12，每级 512 项 × 8 字节 = 4KB = 一页。用 section map 建三级（PGD/PUD/PMD，PMD 里直接放 2MB block 描述符，不需要 PT），于是：

text
idmap_pg_dir   3 页
swapper_pg_dir 3 页
合计 6 页，连续摆放

3.2 建表前的两步准备 #

armasm
__create_page_tables:
    adrp    x25, idmap_pg_dir       // idmap 页表基址
    adrp    x26, swapper_pg_dir     // 内核空间页表基址
    mov x27, lr

    mov x0, x25
    add x1, x26, #SWAPPER_DIR_SIZE
    bl  __inval_cache_range         // 先把这 6 页的 cacheline 无效掉

    mov x0, x25
    add x6, x26, #SWAPPER_DIR_SIZE
1:  stp xzr, xzr, [x0], #16
    stp xzr, xzr, [x0], #16
    stp xzr, xzr, [x0], #16
    stp xzr, xzr, [x0], #16
    cmp x0, x6
    b.lo    1b                      // 全部清零

为什么这两步都不能省：

  • invalidate：boot 协议只保证 kernel image 那一段 clean 到 PoC，页表不在 image 里，其 cacheline 可能是脏的或旧的。
  • 清零 = 全部 invalid：描述符 bit0 就是有效位，写 0 天然就是「无效描述符」，不需要另外构造。

3.3 三个建表宏 #

create_table_entry：填一个中间级描述符

armasm
.macro create_table_entry, tbl, virt, shift, ptrs, tmp1, tmp2
    lsr \tmp1, \virt, #\shift        // 取 index
    and \tmp1, \tmp1, #\ptrs - 1      // mask（PGD/PUD 都是 0x1ff）
    add \tmp2, \tbl, #PAGE_SIZE       // 下一级表 = 当前页 + 4KB（硬编码！）
    orr \tmp2, \tmp2, #PMD_TYPE_TABLE // bit0=1 有效, bit1=1 table
    str \tmp2, [\tbl, \tmp1, lsl #3]  // 每项 8 字节
    add \tbl, \tbl, #PAGE_SIZE        // 指针移到下一级
.endm

注意 add \tmp2, \tbl, #PAGE_SIZE 这一行：下一级页表的地址是硬编码成「当前页 + 一页」的。它能成立，完全依赖链接脚本里那 6 页连续摆放。这是理解整个布局的关键耦合，一般文章只贴宏不解释这层。

create_pgd_entry：把中间级一路串下去，留最后一级

armasm
.macro create_pgd_entry, tbl, virt, tmp1, tmp2
    create_table_entry \tbl, \virt, PGDIR_SHIFT, PTRS_PER_PGD, \tmp1, \tmp2
    create_table_entry \tbl, \virt, TABLE_SHIFT,  PTRS_PER_PTE, \tmp1, \tmp2
.endm

4.15 里第二行的 TABLE_SHIFT 改名叫 SWAPPER_TABLE_SHIFT（即 PUD_SHIFT）。只填 PGD + PUD 两个描述符，就能覆盖 1GB 虚拟地址范围，剩下的 PMD 填多少项由 kernel image 实际大小决定。

create_block_map：填最后一级的 block 描述符

armasm
.macro create_block_map, tbl, flags, phys, start, end
    lsr \phys,  \phys,  #BLOCK_SHIFT
    lsr \start, \start, #BLOCK_SHIFT
    and \start, \start, #PTRS_PER_PTE - 1      // 起始 index
    orr \phys,  \flags, \phys, lsl #BLOCK_SHIFT
    lsr \end,   \end,   #BLOCK_SHIFT
    and \end,   \end,   #PTRS_PER_PTE - 1      // 结束 index
9999: str \phys, [\tbl, \start, lsl #3]
    add \start, \start, #1
    add \phys,  \phys,  #BLOCK_SIZE
    cmp \start, \end
    b.ls    9999b                              // 闭区间，end 那一项也要填
.endm

4.15 里 BLOCK_SHIFT / BLOCK_SIZE 改名为 SWAPPER_BLOCK_SHIFT / SWAPPER_BLOCK_SIZE。原文给出的 C 版等价实现写的是 while (start != end)，与汇编的 b.ls（闭区间）语义不一致，以汇编为准。

3.5 映射 ①：identity mapping #

ARM ARM 的原文理由是：

If the PA of the software that enables or disables a particular stage of address translation differs from its VA, speculative instruction fetching can cause complications. ARM strongly recommends that the PA and VA of any software that enables or disables a stage of address translation are identical…

翻译过来就是：开关 MMU 的那段代码，虚拟地址必须等于物理地址，否则流水线的推测取指会跑到一个「翻译前后含义不同」的地方去。

armasm
    ldr x7, =MM_MMUFLAGS
    mov x0, x25                     // idmap_pg_dir
    adrp    x3, KERNEL_START
    create_pgd_entry x0, x3, x5, x6
    mov x5, x3                      // __pa(KERNEL_START)
    adr_l   x6, KERNEL_END
    create_block_map x0, x7, x3, x5, x6

4.1.10 里 identity map 覆盖的是整个 kernel image。到 4.15 缩到了只映射 .idmap.text 这一段：

#define IDMAP_TEXT                                        \
    . = ALIGN(SZ_4K);                                     \
    VMLINUX_SYMBOL(__idmap_text_start) = .;               \
    *(.idmap.text)                                        \
    VMLINUX_SYMBOL(__idmap_text_end) = .;

System.map 实测（48 位 VA、4.15），这段区间是 ffff200008fbc000 ~ ffff200008fbc408，里面装的正是那些「必须在 identity 前提下跑」的函数：

3.6 映射 ②：内核空间 #

4.1.10——映射到线性区

armasm
    mov x0, x26                     // swapper_pg_dir（= 进程 0 的地址空间）
    mov x5, #PAGE_OFFSET            // 48 位 VA → 0xffff8000_00000000
    create_pgd_entry x0, x5, x3, x6
    ldr x6, =KERNEL_END
    mov x3, x24                     // __PHYS_OFFSET
    create_block_map x0, x7, x3, x5, x6

4.15——映射到 VMALLOC 区

armasm
    adrp    x0, swapper_pg_dir
    mov_q   x5, KIMAGE_VADDR + TEXT_OFFSET    // 编译期的 __va(_text)
    add x5, x5, x23                           // KASLR 位移，默认 0
    create_pgd_entry x0, x5, x3, x6
    adrp    x6, _end
    adrp    x3, _text
    sub x6, x6, x3                            // _end - _text
    add x6, x6, x5                            // 运行期 __va(_end)
    create_block_map x0, x7, x3, x5, x6

这里有个值得注意的选型规则：

符号 用 adrp 还是 adr_l 原因
KERNEL_START adrp 链接脚本保证它 4KB 对齐
KERNEL_END adr_l 没有任何协议保证它页对齐，adrp 会丢低位

只映射「image 起始（含 TEXT_OFFSET 保留区）→ _end」，覆盖 text/rodata/data/bss 和各种段。不可能把 256TB 都映射上。

3.7 映射 ③：DTB blob #

armasm
    mov x3, x21                          // FDT 物理地址
    and x3, x3, #~((1 << 21) - 1)        // 按 2MB 向下对齐
    mov x6, #PAGE_OFFSET
    sub x5, x3, x24                      // 相对 __PHYS_OFFSET 的偏移
    tst x5, #~((1 << 29) - 1)            // 超出 512MB 了吗
    csel    x21, xzr, x21, ne            // 超了就把 FDT 指针清零
    b.ne    1f
    add x5, x5, x6                       // blob 的虚拟地址
    add x6, x5, #1 << 21                 // 只映射 2MB
    sub x6, x6, #1
    create_block_map x0, x7, x3, x5, x6
1:

为什么只建 PMD、不建 PGD/PUD：协议已经把 blob 限制在「kernel space 起始 512MB 内 + 8 字节对齐」，它必然落在内核空间映射已经建好的 PGD/PUD 覆盖范围内，直接复用即可。

csel x21, xzr, x21, ne 这个写法值得学：条件选择替代分支，省一次跳转，也不破坏流水线。

3.8 收尾 #

armasm
    mov x0, x25
    add x1, x26, #SWAPPER_DIR_SIZE
    dmb sy
    bl  __inval_cache_range
    mov lr, x27

建表过程是在 cache off 下写内存的，cacheline 可能被推测执行过的 load 读过，所以再 invalid 一次收尾。

四、__cpu_setup：算好开 MMU 需要的寄存器值 #

armasm
ENTRY(__cpu_setup)
    ic  iallu              // 无效全部 I-cache（含 BTB）
    tlbi vmalle1is         // 无效全部 inner-shareable PE 的 TLB
    dsb ish                // 保证上面两条在开 MMU 前做完

    mov x0, #3 << 20
    msr cpacr_el1, x0      // FPEN[21:20] = 11，放开 FP/ASIMD
    msr mdscr_el1, xzr
    ...

tlbi vmalle1is 拆开读：vm-all + e1 + is（inner shareable）。也就是说它作用于所有 inner-shareable PE。

这里有个合理的质疑：__cpu_setup 是每个 CPU 各跑一次的，从核启动时再去刷所有 PE 的 TLB，代价过大。更合理的做法是只刷本核（去掉 is 后缀）。原文作者提了这一点，没有定论。

后面算两组寄存器值。先看 SCTLR：

armasm
    adr x5, crval
    ldp w5, w6, [x5]
    mrs x0, sctlr_el1
    bic x0, x0, x5         // 清掉该清的
    orr x0, x0, x6         // 置上该置的

armasm
crval:
    .word 0xfcffffff       // 需要清 0 的位
    .word 0x34d5d91d       // 需要置 1 的位

0xfcffffff 这个掩码没有包含 EE/E0E，所以 el2_setup 里根据大小端定下来的 endianness 会被保留下来。几个关键位：

位 名字 作用
[0] M MMU 总开关
[2] C D-cache（也包括 stage1 页表自身的访问）
[12] I I-cache
[9] UMA EL0 能否访问 PSTATE.{D,A,I,F}
[15] UCT EL0 能否读 CTR_EL0
[16] WFI EL0 能否执行 WFI
[18] nTWE EL0 能否执行 WFE
[26] UCI EL0 能否执行 DC/IC

再算 TCR：

armasm
    ldr x10, =TCR_TxSZ(VA_BITS) | TCR_CACHE_FLAGS | TCR_SMP_FLAGS | \
               TCR_TG_FLAGS | TCR_ASID16 | TCR_TBI0
    tcr_set_idmap_t0sz x10, x9
    mrs x9, ID_AA64MMFR0_EL1
    bfi x10, x9, #32, #3       // IPS = PARange，物理地址位宽
    msr tcr_el1, x10
    ret

TCR_EL1 主要域：

位 名字 含义
[5:0] T0SZ VA_BITS = 64 - T0SZ
[21:16] T1SZ 同上，管 TTBR1
[15:14] / [31:30] TG0 / TG1 00=4K、01=64K、10=16K
[13:12] / [29:28] SH0 / SH1 页表 walk 的 shareability
[11:10] / [27:26] ORGN0 / ORGN1 outer cache 属性
[9:8] / [25:24] IRGN0 / IRGN1 inner cache 属性
[36] AS ASID 宽度（0=8 位，1=16 位）
[37] / [38] TBI0 / TBI1 Top Byte Ignored，高 8 位可挪作他用
[34:32] IPS 物理地址位宽（来自 ID_AA64MMFR0_EL1.PARange）

注意 ⇔__cpu_setup 在这里还没有写 mair_el1。这不是遗漏：4.1.10 的这段汇编只负责算 SCTLR 与 TCR，MAIR 的设置点在别处（或按配置默认）。写驱动时手动改内存属性，记得 MAIR 才是属性表本体。

返回值约定也很有意思。stext 尾部是：

armasm
    ldr x27, =__mmap_switched
    adr_l   lr, __enable_mmu
    b   __cpu_setup

用 b 而不是 bl，所以 __cpu_setup 里的 ret 会直接落到 __enable_mmu，而不是回到 stext。x0 就是它算好的 SCTLR_EL1 值。

4.1 内存类型的五个索引 #

c
#define MT_DEVICE_nGnRnE  0
#define MT_DEVICE_nGnRE   1
#define MT_DEVICE_GRE     2
#define MT_NORMAL_NC      3
#define MT_NORMAL         4

这五个数只是 MAIR_EL1 的索引。真正的属性表在 MAIR_EL1 里，64 位切成 8 条，描述符的 AttrIndx[2:0] 指哪条就是哪条。

  • device 类型：总是 non-cacheable + outer shareable，再加三个附加特性——G/nG（能否合并访问）、R/nR（能否重排）、E/nE（写 ack 是否必须来自最终目的地而不是 write buffer）。
  • normal 类型：cacheable 与否、write-through / write-back、read/write allocate、transient 与否，inner 和 outer 还能分开设。
  • ARMv8 是 weak-ordered 模型，normal 内存允许重排/合并/重复；device 内存有副作用（FIFO 地址固定但每次读出不同），禁止推测性数据访问。

五、__enable_mmu 与 __mmap_switched #

armasm
__enable_mmu:
    ldr x5, =vectors
    msr vbar_el1, x5        // ① 向量表基址
    msr ttbr0_el1, x25      // ② TTBR0 ← idmap_pg_dir
    msr ttbr1_el1, x26      //    TTBR1 ← swapper_pg_dir
    isb
    msr sctlr_el1, x0       // ③ M 位置 1，MMU on
    isb
    br  x27                 // ④ 直接跳走，不设 lr

四个入参：x0 = SCTLR 值、x25 = idmap 页表、x26 = swapper 页表、x27 = __mmap_switched。

为什么 idmap 必须挂 TTBR0：物理地址的高 16 位全是 0，落在 VA 的低段，而低段由 TTBR0 索引。等内核起来之后，TTBR0 会被换成各进程的用户地址空间，TTBR1 始终指向内核那份 swapper_pg_dir。

两条 isb 是干嘛的：开 MMU 前后的指令存在乱序风险——绿色（MMU off）时期的 load 可能在总线上发出时 MMU 已经开了；橘色（MMU on）时期的指令可能提前执行完。两种解法：

  1. identity mapping：A~B 的虚拟映射到 A~B 物理，不管走 VA 还是 PA 都命中同一块内存；
  2. 指令屏障：isb 把执行切成三段。

这里两种都用了：idmap 保证取指不断，isb 保证 msr sctlr_el1 前后边界清晰。

armasm
__mmap_switched:
    adr_l   x6, __bss_start
    adr_l   x7, __bss_stop
1:  cmp x6, x7
    b.hs    2f
    str xzr, [x6], #8           // 清 BSS
    b   1b
2:  adr_l   sp, initial_sp, x4
    str_l   x21, __fdt_pointer, x5    // 保存 FDT 物理地址
    str_l   x24, memstart_addr, x6    // 保存 PHYS_OFFSET
    mov x29, #0
    b   start_kernel

armasm
.set initial_sp, init_thread_union + THREAD_START_SP

adr_l sp, initial_sp 这一句很关键：从这一刻起，初始化代码归属 idle 进程（pid 0），有了 thread_info、task_struct 和自己的内核栈。而 memstart_addr 是后面 __virt_to_phys / __phys_to_virt 做线性换算的偏移基准。

MMU 打开之后，48 位 VA 空间是这么分的：

text
低段 0x0000_0000_0000_0000 ~ 0x0000_FFFF_FFFF_FFFF   用户空间（TTBR0）
高段 0xFFFF_0000_0000_0000 ~ 0xFFFF_FFFF_FFFF_FFFF   内核空间（TTBR1）
中间那一大段                                       无效，访问即 fault

MMU 打开之前内核能跑，是因为 stext 这一路的代码全是位置无关码（PIC）。

六、UEFI stub：64 字节头部的双重身份 #

armasm
#ifdef CONFIG_EFI
efi_head:
    add x13, x18, #0x16      // 这条指令的机器码字节恰好构成 MZ 签名
    b   stext
#else
    b   stext
    .long   0
#endif
    .quad   _kernel_offset_le
    .quad   _kernel_size_le
    .quad   _kernel_flags_le
    .quad   0
    .quad   0
    .quad   0
    .byte   0x41             // "ARMd"
    .byte   0x52
    .byte   0x4d
    .byte   0x64
#ifdef CONFIG_EFI
    .long   pe_header - efi_head    // offset 0x3c，指向 PE header
#else
    .word   0
#endif

这是整个启动流程里最「黑客」的一处：

  • 不开 EFI 时：头 8 字节是 b stext，最后 4 字节是 0，普通 bootloader 直接跳进来就行。
  • 开 EFI 时：这 64 字节被 UEFI 解释成 MS-DOS MZ header。add x13, x18, #0x16 这条指令的机器码字节序列恰好以 0x4d5a（"MZ"）开头，指令本身没有任何实际语义，纯粹是拿机器码冒充签名。
  • 中间三个 .quad 在 UEFI 眼里是 MS-DOS header 的无用区，在普通 bootloader 眼里是「加载偏移 / 加载长度 / 标志位」。
  • offset 0x3c 那个 4 字节指针指向 PE header——UEFI boot manager 看到 MZ 之后会按这个偏移继续解析。

接着是 PE / COFF / optional header：

optional_header:
    .short  0x20b           // PE32+ magic
    .byte   0x02            // MajorLinkerVersion
    .byte   0x14            // MinorLinkerVersion

0x20b 就是 PE32+，UEFI application 必须是 PE32+，不能是 PE32。真实的 PE 文件在 MS-DOS stub 区会打印「This program cannot be run in DOS mode」，Linux 内核不提供这段，只伪装头部。

原文只讲到 PE 结构布局为止，efi_stub_entry 的实现、ExitBootServices() 的调用时序、efi_config_table 的传递都没有展开。要写「boot services 退出」必须另找材料。

7.1 地址布局宏重排 #

c
#define VA_BITS         (CONFIG_ARM64_VA_BITS)
#define VA_START        (UL(0xffffffffffffffff) - (UL(1) << VA_BITS) + 1)
#define PAGE_OFFSET     (UL(0xffffffffffffffff) - (UL(1) << (VA_BITS - 1)) + 1)

#define KIMAGE_VADDR    (MODULES_END)
#define VMALLOC_START   (MODULES_END)
#define VMALLOC_END     (PAGE_OFFSET - PUD_SIZE - VMEMMAP_SIZE - SZ_64K)
#define MODULES_END     (MODULES_VADDR + MODULES_VSIZE)
#define MODULES_VADDR   (VA_START + KASAN_SHADOW_SIZE)
#define MODULES_VSIZE   (SZ_128M)
#define VMEMMAP_START   (PAGE_OFFSET - VMEMMAP_SIZE)
#define PCI_IO_END      (VMEMMAP_START - SZ_2M)
#define PCI_IO_START    (PCI_IO_END - PCI_IO_SIZE)
#define FIXADDR_TOP     (PCI_IO_START - SZ_2M)

注意第一行：KIMAGE_VADDR == VMALLOC_START == MODULES_END。内核运行地址从「线性映射区」搬到了「VMALLOC 区」。

按宏展开推算（48 位 VA）：VA_START = 0xffff000000000000，KASAN_SHADOW_SIZE = 2^45，再加 128MB → KIMAGE_VADDR ≈ 0xffff200008000000。上一章 System.map 里 __idmap_text_start = ffff200008fbc000 正落在其后，可以互相印证。

顺便说，4.1.10 的 PAGE_OFFSET 写作 (ULmax << (VA_BITS-1))，4.15 改成 ULmax - (1 << (VA_BITS-1)) + 1。48 位下两者都等于 0xffff800000000000，只是表达式换了个写法。

7.2 virt_to_phys 的两分支 #

c
#define __is_lm_address(addr)  (!!((addr) & BIT(VA_BITS - 1)))
#define __lm_to_phys(addr)     (((addr) & ~PAGE_OFFSET) + PHYS_OFFSET)
#define __kimg_to_phys(addr)   ((addr) - kimage_voffset)

#define __virt_to_phys_nodebug(x) ({              \
    phys_addr_t __x = (phys_addr_t)(x);           \
    __is_lm_address(__x) ? __lm_to_phys(__x) :    \
                           __kimg_to_phys(__x);   \
})

判定规则简单到有点意外：线性映射区的大小恰好是内核虚拟地址空间的一半，所以只看 bit(VA_BITS-1) 这一位——

  • 该位为 1 → 在线性映射区，用 PAGE_OFFSET 取偏移；
  • 该位为 0 → 在 kernel image 区（VMALLOC 段），用 kimage_voffset 换算。

7.3 kimage_voffset 是怎么来的 #

armasm
ENTRY(kimage_vaddr)
    .quad   _text - TEXT_OFFSET

__primary_switched:
    ldr_l   x4, kimage_vaddr     // 编译期的 image 虚拟地址
    sub x4, x4, x0               // x0 = __PHYS_OFFSET（物理）
    str_l   x4, kimage_voffset, x5
    b   start_kernel

armasm
__primary_switch:
    bl  __enable_mmu
    ldr x8, =__primary_switched
    adrp    x0, __PHYS_OFFSET    // 此刻 MMU 已开！
    br  x8

adrp x0, __PHYS_OFFSET 这行看着像 bug——MMU 都开了，adrp 拿到的是虚拟地址，怎么当物理地址用？

答案是：__primary_switch 本身位于 .idmap.text，那段被 identity mapping 覆盖，虚拟地址在数值上就等于物理地址。这也是 4.15 把 idmap 缩到极小范围后仍然必须包含 __primary_switch 的原因。

7.4 memstart_addr 的选取 #

c
void __init arm64_memblock_init(void)
{
    const s64 linear_region_size = -(s64)PAGE_OFFSET;

    BUILD_BUG_ON(linear_region_size != BIT(VA_BITS - 1));      // ①

    memstart_addr = round_down(memblock_start_of_DRAM(),       // ②
                               ARM64_MEMSTART_ALIGN);          //    1G 对齐

    memblock_remove(max_t(u64, memstart_addr + linear_region_size,   // ③
                          __pa_symbol(_end)), ULLONG_MAX);
    if (memstart_addr + linear_region_size < memblock_end_of_DRAM()) {
        memstart_addr = round_up(memblock_end_of_DRAM() - linear_region_size,
                                 ARM64_MEMSTART_ALIGN);        // ④
        memblock_remove(0, memstart_addr);                     // ⑤
    }
}
  • ① 那个 BUILD_BUG_ON 是编译期锁死「线性区恰好占内核虚拟空间一半」这个前提——它是 __is_lm_address 用单 bit 判定的依据。
  • ② 物理内存起点按 1GB 对齐。
  • ④⑤ 处理「物理内存太大，线性区装不下」的情形：改成从 RAM 末尾倒推基址，并把 [0, memstart_addr] 从 memblock 里摘掉。

八、fixmap：内存管理子系统还没起来时怎么办 #

内核这时已经跑在虚拟地址上了，但 DTB 的地址是物理的。要访问它就得建映射，而建映射需要内存——偏偏 ioremap 依赖的 vmalloc 还没 ready。于是内核预留一段编译期就定死的虚拟地址做固定映射。

c
enum fixed_addresses {
    FIX_HOLE,
#define FIX_FDT_SIZE        (MAX_FDT_SIZE + SZ_2M)
    FIX_FDT_END,
    FIX_FDT = FIX_FDT_END + FIX_FDT_SIZE / PAGE_SIZE - 1,   // ①
    FIX_EARLYCON_MEM_BASE,                                   // ②
    FIX_TEXT_POKE0,
    __end_of_permanent_fixed_addresses,
#define NR_FIX_BTMAPS       (SZ_256K / PAGE_SIZE)            // 64
#define FIX_BTMAPS_SLOTS    7
#define TOTAL_FIX_BTMAPS    (NR_FIX_BTMAPS * FIX_BTMAPS_SLOTS)   // 448
    FIX_BTMAP_END   = __end_of_permanent_fixed_addresses,
    FIX_BTMAP_BEGIN = FIX_BTMAP_END + TOTAL_FIX_BTMAPS - 1,  // ③
    FIX_PTE, FIX_PMD, FIX_PUD, FIX_PGD,
    __end_of_fixed_addresses
};
#define FIXADDR_SIZE    (__end_of_permanent_fixed_addresses << PAGE_SHIFT)
#define FIXADDR_START   (FIXADDR_TOP - FIXADDR_SIZE)

fixmap 空间平分成两半：permanent（建好就不变，特定模块长期用）与 temporary（用完即解）。

枚举方向要特别注意：

c
#define __fix_to_virt(x)  (FIXADDR_TOP - ((x) << PAGE_SHIFT))

index 越大，地址越低。所以 ①FIX_FDT 落在 FIX_FDT_END 下方 4MB 处；③FIX_BTMAP_BEGIN 在 FIX_BTMAP_END 之上（数值更大）但地址更低，区间 [FIX_BTMAP_BEGIN, FIX_BTMAP_END] 才说得通。

条目 大小 用途
FIX_FDT 4MB（MAX_FDT_SIZE + SZ_2M） 映射 DTB。放枚举最前面是因为 section mapping 要求虚拟地址 2MB 对齐，而 FIXADDR_TOP 本身 2MB 对齐，FIXADDR_TOP - (FIX_FDT << 12) 天然容易对齐
FIX_EARLYCON_MEM_BASE 1 页 early console 的串口寄存器就那么几个
FIX_BTMAP_* 7 slot × 256KB = 448 页 early_ioremap()

建表只建到 PMD 一级：

c
static pte_t bm_pte[PTRS_PER_PTE] __page_aligned_bss;
static pmd_t bm_pmd[PTRS_PER_PMD] __page_aligned_bss __maybe_unused;
static pud_t bm_pud[PTRS_PER_PUD] __page_aligned_bss __maybe_unused;

void __init early_fixmap_init(void)
{
    pgd_t *pgd; pud_t *pud; pmd_t *pmd;
    unsigned long addr = FIXADDR_START;

    pgd = pgd_offset_k(addr);
    if (pgd_none(*pgd))
        __pgd_populate(pgd, __pa_symbol(bm_pud), PUD_TYPE_TABLE);
    pud = fixmap_pud(addr);
    if (pud_none(*pud))
        __pud_populate(pud, __pa_symbol(bm_pmd), PMD_TYPE_TABLE);
    pmd = fixmap_pmd(addr);
    __pmd_populate(pmd, __pa_symbol(bm_pte), PMD_TYPE_TABLE);

    BUILD_BUG_ON((__fix_to_virt(FIX_BTMAP_BEGIN) >> PMD_SHIFT)
              != (__fix_to_virt(FIX_BTMAP_END)   >> PMD_SHIFT));
    ...
}

PGD 复用 swapper_pg_dir，只静态分配 PUD/PMD/PTE 三张表。于是整个 fixmap 的有效范围被限制在 2MB 以内——超出就必须退化成 section mapping。

8.1 映射 DTB #

c
void *__init __fixmap_remap_fdt(phys_addr_t dt_phys, int *size, pgprot_t prot)
{
    const u64 dt_virt_base = __fix_to_virt(FIX_FDT);
    int offset; void *dt_virt;

    BUILD_BUG_ON(dt_virt_base % SZ_2M);                    // 必须 2MB 对齐
    BUILD_BUG_ON(__fix_to_virt(FIX_FDT_END) >> SWAPPER_TABLE_SHIFT !=
                 __fix_to_virt(FIX_BTMAP_BEGIN) >> SWAPPER_TABLE_SHIFT);

    offset  = dt_phys % SWAPPER_BLOCK_SIZE;
    dt_virt = (void *)dt_virt_base + offset;

    create_mapping_noalloc(round_down(dt_phys, SWAPPER_BLOCK_SIZE),
                           dt_virt_base, SWAPPER_BLOCK_SIZE, prot);

    *size = fdt_totalsize(dt_virt);
    if (offset + *size > SWAPPER_BLOCK_SIZE)
        create_mapping_noalloc(round_down(dt_phys, SWAPPER_BLOCK_SIZE), dt_virt_base,
                               round_up(offset + *size, SWAPPER_BLOCK_SIZE), prot);
    return dt_virt;
}

三处设计意图：

  1. 第二个 BUILD_BUG_ON 保证 FDT 区与 BTMAP 区落在同一张 PMD 表里，省掉再分配一张 PMD 页。
  2. 第一次只映射 2MB（对齐到 SWAPPER_BLOCK_SIZE）。
  3. DTB 尾部越过 2MB 时，紧接着再补一段 2MB 对齐的映射。

8.2 early ioremap #

3 记录启动模式 set_cpu_boot_mode_flag EL1 无（写 __boot_cpu_mode + dc ivac）
4 校验 FDT __vet_fdt EL1 无（8 字节对齐 + 512MB 范围）
5 建启动页表 __create_page_tables EL1，MMU off idmap 3 页 + swapper 3 页；① identity ② kernel image（PGD+PUD+PMD block）③ DTB（只建 PMD，2MB）；前后各 invalidate 一次
6 CPU 初始化 __cpu_setup（b 进入） EL1 无页表动作；ic iallu + tlbi vmalle1is + dsb ish；cpacr_el1 = 3<<20；算 SCTLR 与 TCR
7 开 MMU __enable_mmu EL1 vbar_el1；TTBR0←idmap、TTBR1←swapper；isb；msr sctlr_el1（M=1）；isb；br x27
8 进 C 世界 __mmap_switched / __primary_switched EL1，MMU on 清 BSS；设 sp；存 __fdt_pointer、memstart_addr；4.15 另算 kimage_voffset
9 从核启动 secondary_startup 等（都在 .idmap.text） EL1 复用 BSP 建好的页表
10 start_kernel → setup_arch arm64_memblock_init / early_fixmap_init / __fixmap_remap_fdt / paging_init EL1，MMU on 定 memstart_addr；fixmap 建 PUD/PMD/PTE（PGD 复用）；FDT 2MB section 映射；线性区按 memstart_addr 建映射

十、速查表 #

现象 / 疑问 答案
为什么页表要放 bss 之后 原来放在 PHYS_OFFSET~+TEXT_OFFSET，那里可能被 bootloader 用来传参，而 memblock 还没 ready 无法保护
为什么恰好 6 页 create_table_entry 把「下一级表 = 当前页 + 一页」写死了，靠链接脚本连续摆放兜底
identity map 要多大 4.1.10 = 整个 kernel image；4.15 = 只 .idmap.text
为什么开 MMU 那段代码必须 VA==PA ARM ARM 明确推荐；否则推测取指会跑到翻译前后含义不同的地方
adrp 还是 adr_l 符号 4KB 对齐才可用 adrp；_end 无保证必须用 adr_l
__primary_switch 里 adrp 拿到虚拟地址怎么当物理用 该函数在 .idmap.text，VA 数值上等于 PA
4.15 里 kernel 跑在哪 KIMAGE_VADDR = MODULES_END，在 VMALLOC 区，不在线性区
virt_to_phys 怎么知道走哪条路 看 bit(VA_BITS-1)：1 是线性区，0 是 image 区
fixmap 为什么必须有 解析 DTB 要映射，建映射要内存，而 vmalloc 还没起来
UEFI 头部的 MZ 哪来的 add x13,x18,#0x16 的机器码字节恰好构成 0x4d5a，无实际语义

小结 #

  • 入口只做四件事：存参数、降到 EL1、记启动模式、校验 FDT。
  • 页表分三张：identity（开关 MMU 时用）、kernel image（跑内核用）、DTB（解析设备树用）。
  • 页表的摆放位置和彼此间距都是硬编码约定，改一个就得改另一个。
  • 开 MMU 前后那一小段是整条链上最脆的地方：靠 identity mapping + isb 双保险。
  • 4.15 之后 kernel image 搬进 VMALLOC，引出 kimage_voffset 和 __is_lm_address 的单 bit 判定——看老资料对不上号，多半是这里。
  • fixmap 是「内存管理还没起来」时的过渡方案，本质是编译期定死的一段虚拟地址。
