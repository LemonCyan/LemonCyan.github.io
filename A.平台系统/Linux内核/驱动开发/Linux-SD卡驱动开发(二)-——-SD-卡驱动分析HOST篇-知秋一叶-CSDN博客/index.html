Linux SD卡驱动开发（二）—— SD卡控制器HOST驱动深度剖析 #

在嵌入式 Linux 系统中，MMC（MultiMediaCard）/SD/SDIO 存储设备采用统一的 MMC 子系统进行管理。整个子系统从逻辑上被划分为三层：

graph TD
    A[CARD 区块层: mmc_block] -->|提供块设备接口| B[CORE 核心层: mmc_core]
    B -->|构建命令与协议| C[HOST 主机控制器层: s3cmci]
    C -->|直接操作硬件寄存器| D[SD/MMC 物理卡接口]
  1. CARD（区块层）：负责将 SD 卡实现为块设备（Block Device），为用户层暴露 /dev/mmcblk0 等读写节点。
  2. CORE（核心层）：子系统的中枢。它封装了具体的 SD/MMC 协议规范，完成了卡识别、状态管理，并向上层块设备驱动和下层主机控制器驱动提供接口。
  3. HOST（主机控制器层）：与具体芯片架构强相关的硬件驱动层（对于 S3C2440 平台，对应的文件为 s3cmci.c）。驱动工程师的主要工作就是在此层实现特定平台控制器的初始化、时钟波特率配置以及命令/数据的物理收发。

本文以 S3C2440/S3C2410 控制器为例，结合内核源码（Linux-3.14）深度剖析 HOST 层驱动的初始化和核心控制流程。

一、 核心数据结构关系 #

在 HOST 驱动中，有两个最关键的结构体描述控制器：

  • struct mmc_host：由内核核心层（mmc_core）定义，是一个通用的控制设备抽象（类似于基类），封装了所有平台控制器所需的通用属性（如支持的最大块大小、支持的数据总线位宽、操作方法集等）。
  • struct s3cmci_host：在 s3cmci.h 中定义的硬件私有结构体，代表具体的 S3C2440 SDI 硬件实例。它内部包含了通用 mmc_host 的指针，以及与其硬件绑定的 GPIO 引脚、时钟源、DMA 通道、I/O 映射虚拟基地址等。
c
struct mmc_host {
    const struct mmc_host_ops *ops;      // SD卡主控制器的操作函数集
    const struct mmc_bus_ops  *bus_ops;  // 总线操作函数集
    struct mmc_ios             ios;      // 配置时钟、总线位宽、电压等
    struct mmc_card           *card;     // 挂载在此控制器上的卡实例
    // ...
};

struct s3cmci_host {
    struct platform_device    *pdev;
    struct s3c24xx_mci_pdata  *pdata;
    struct mmc_host           *mmc;      // 指向通用 mmc_host
    void __iomem              *base;     // 物理寄存器 I/O 映射后的虚拟基地址
    int                        irq;      // SDI 控制器中断号
    int                        irq_cd;   // 插拔检测中断号
    int                        dma;      // DMA 通道号
    unsigned long              clk_rate; // 源时钟频率
    // ...
};

二、 平台驱动绑定与初始化 #

HOST 驱动常被注册为平台驱动（Platform Driver），当平台设备（Platform Device）的名字与驱动中的 .name（此处为 "s3c-sdi"）匹配时，便会触发 probe 函数。

在新版内核中，使用了辅助宏 module_platform_driver(s3cmci_driver); 代替了传统的 module_init 和 module_exit，以减少模板代码（Boilerplate Code）：

c
static struct platform_driver s3cmci_driver = {
    .driver = {
        .name  = "s3c-sdi",
        .owner = THIS_MODULE,
    },
    .id_table = s3cmci_driver_ids,
    .probe    = s3cmci_probe,
    .remove   = s3cmci_remove,
    .shutdown = s3cmci_shutdown,
};

三、 probe 函数的五大核心任务 #

当平台总线完成匹配，s3cmci_probe() 被调用。它承担了硬件控制器就绪的核心准备工作：

c
static int s3cmci_probe(struct platform_device *pdev)
{
    struct s3cmci_host *host;
    struct mmc_host *mmc;
    int ret, is2440;

    is2440 = platform_get_device_id(pdev)->driver_data;

    // 1. 分配 mmc_host 控制器内存
    // mmc_alloc_host 会同时在末尾为私有结构体 s3cmci_host 预留空间
    mmc = mmc_alloc_host(sizeof(struct s3cmci_host), &pdev->dev);
    if (!mmc) {
        return -ENOMEM;
    }

    host = mmc_priv(mmc); // 获取 s3cmci_host 的首地址
    host->mmc = mmc;
    host->pdev = pdev;
    host->is2440 = is2440;
    host->pdata = pdev->dev.platform_data;

    // 2. GPIO 资源的申请与配置 (GPE5 到 GPE10 分别对应 SDI 命令、时钟及 4 条数据线)
    for (int i = S3C2410_GPE(5); i <= S3C2410_GPE(10); i++) {
        ret = gpio_request(i, dev_name(&pdev->dev));
        if (ret) {
            goto probe_free_host;
        }
    }

    // 3. I/O 物理地址空间映射
    host->mem = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    host->mem = request_mem_region(host->mem->start, resource_size(host->mem), pdev->name);
    host->base = ioremap(host->mem->start, resource_size(host->mem));

    // 4. 申请中断及使能时钟
    host->irq = platform_get_irq(pdev, 0);
    request_irq(host->irq, s3cmci_irq, 0, DRIVER_NAME, host);
    disable_irq(host->irq); // 物理中断暂时屏蔽，等系统一切就绪再打开

    // 如果板级数据配置了插拔检测引脚，则申请 CD 中断
    if (!host->pdata->no_detect) {
        gpio_request(host->pdata->gpio_detect, "s3cmci detect");
        host->irq_cd = gpio_to_irq(host->pdata->gpio_detect);
        request_irq(host->irq_cd, s3cmci_irq_cd, IRQF_TRIGGER_RISING | IRQF_TRIGGER_FALLING, DRIVER_NAME, host);
    }

    // 5. 时钟使能与配置
    host->clk = clk_get(&pdev->dev, "sdi");
    clk_enable(host->clk);
    host->clk_rate = clk_get_rate(host->clk);

    // 6. 将操作函数集绑定并向核心层注册 mmc 主机
    mmc->ops = &s3cmci_ops;
    mmc->ocr_avail = MMC_VDD_32_33 | MMC_VDD_33_34;
    mmc->f_min = host->clk_rate / (host->clk_div * 256);
    mmc->f_max = host->clk_rate / host->clk_div;

    ret = mmc_add_host(mmc); // 向 mmc_core 层正式注册控制器
    if (ret) {
        goto free_cpufreq;
    }

    platform_set_drvdata(pdev, mmc);
    return 0;
}

四、 控制器的行为函数集 (struct mmc_host_ops) #

与字符设备的 file_operations 类似，mmc_host_ops 描述了该控制器硬件的底层通信操作，是核心层控制主控硬件的唯一桥梁。

c
static struct mmc_host_ops s3cmci_ops = {
    .request         = s3cmci_request,         // 发起物理命令/数据传输请求的入口（最核心）
    .set_ios         = s3cmci_set_ios,         // 配置时钟频率、总线线宽、供电电压时调用
    .get_ro          = s3cmci_get_ro,          // 查询卡是否写保护
    .get_cd          = s3cmci_card_present,    // 检测当前卡是否在位插入
    .enable_sdio_irq = s3cmci_enable_sdio_irq, // 使能 SDIO 中断
};

1. s3cmci_request 的处理流 #

当核心层（mmc_core）需要向物理卡发送操作命令或者准备进行块读写时，会调用底层的 .request 方法。

  • 该函数是非阻塞的，它将传输请求对象（struct mmc_request *mrq）挂载到私有主机结构体上。
  • 根据 mrq 的数据段参数配置 DMA 或者是启动 PIO（软中断 tasklet 机制）发送接收数据。
  • 物理命令发送完毕或数据读写完成后，通过中断返回并触发 mmc_request_done(mmc, mrq)，以告知核心层本次 I/O 事务已圆满结束。
