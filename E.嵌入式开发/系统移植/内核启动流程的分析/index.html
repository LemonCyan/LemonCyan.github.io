{//内核启动流程 arch/arm/kernel/head.S ENTRY(stext)

  1. 关闭所有的中断，进入SVC模式
  2. 获取处理器ID，检查处理器类型
  3. bl __vet_atags //检查给内核传递的参数（tag列表，或者设备树） 如果是atag列表传参，检测列表是否是以ATAG_CODE开始
  4. 创建页表 {//5. 开启mmu ldr r13, =__mmap_switched @ address to jump to after {//b __enable_mmu b __turn_mmu_on mcr p15, 0, r0, c1, c0, 0 @ write control reg mrc p15, 0, r3, c0, c0, 0 @ read id reg mov r3, r13 mov pc, r3 __mmap_switched: b start_kernel } {//start_kernel 调用了一系列的初始化函数，对内核进行初始化 //打开系统所有的中断 local_irq_enable(); //初始化控制台 ，，初始化控制台之后就可以调用printk函数进行打印 console_init(); rest_init(); } {//rest_init(); 创建了一个内核线程kernel_init kernel_thread(kernel_init, NULL, CLONE_FS | CLONE_SIGHAND); {//kernel_init {//kernel_init_freeable(); if (strncmp(root_device_name, "/dev/", 5) == 0)

root_device_name += 5; mount_root(); if (ROOT_DEV == Root_NFS) if (mount_nfs_root()) return;

printk(KERN_ERR "VFS: Unable to mount root fs via NFS, trying floppy.\n"); } 执行init进程 run_init_process(execute_command); if (!try_to_run_init_process("/sbin/init") || !try_to_run_init_process("/etc/init") || !try_to_run_init_process("/bin/init") || !try_to_run_init_process("/bin/sh")) return 0; } } } }

{//uImage的生成过程 {//1. make Image产生非压缩的镜像 image 1）内核的各个模块经过编译链接，在内核源代码的顶层的目录下生成vmlinux文件（ELF格式的镜 像） 2）使用objcopy命令把vmlinux转换为二进制格式镜像arch/arm/boot/Image } {//2. make zImage产生一个压缩的内核镜像 1）使用gzip对非压缩内核二进制镜像arch/arm/boot/Image进行压缩，生成 arch/arm/boot/compressed/piggy.gzip 2）其中在compressed目录下的3个文件，经过编译链接，在该目录下产生vmlinux文件（elf的镜像） 3）用objcopy命令将该目录下的vmlinux文件转换为zImage并放在arch/arm/boot/zImage }

{//3. mkimage 由zImage生成uImage

uImage是在zImage的基础上添加64字节的uImage头部信息，需要借助工具mkimage来生成， mkimage在uboot的tools中 } }
