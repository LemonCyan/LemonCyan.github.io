解析完 open、close、read、write 四个函数之后，终于到了 ioctl()。

read/write 只能传数据流，表达不了"设置波特率到 115200""让蜂鸣器以 1 kHz 鸣叫"这类带参数的离散控制语义。ioctl 就是为这件事存在的：它是设备驱动程序中对设备的 I/O 通道进行管理的函数。

一、ioctl 是什么 #

1.1 两个原型 #

用户态：

c
#include <sys/ioctl.h>

int ioctl(int fd, unsigned long cmd, ...);
参数 含义
fd open() 返回的文件描述符
cmd 控制命令码
... 可选参数，通常是一个指针（void *argp），有无及含义由 cmd 决定

驱动态（file_operations 成员）：

c
int (*ioctl)(struct inode *inode, struct file *filp, unsigned int cmd, unsigned long arg);
参数 含义
inode / filp 对应应用层传入的 fd，与 open 方法的参数一致
cmd 由用户空间不经修改地直接传给驱动
arg 可选，含义由 cmd 决定

现代内核里通用字符设备应当实现 unlocked_ioctl（long (*)(struct file *, unsigned int, unsigned long)），ioctl 成员属于 BKL 时代的旧接口，新驱动不要再用。compat_ioctl 用于 64 位内核兼容 32 位应用。

1.2 驱动里长什么样 #

实现上就是一个 switch { case } 结构，每个 case 对应一个命令码，做相应操作。怎么实现操作是每个程序员自己的事（设备都是特定的）；关键在于怎样组织命令码——因为在 ioctl 中，命令码是唯一联系用户程序命令和驱动支持的途径。

1.3 为什么需要 ioctl #

不用 ioctl 也能实现对设备 I/O 通道的控制。例如在驱动实现 write 时检查是否有特殊约定的数据流通过，如果有，后面就跟着控制命令（这在 socket 编程中很常见）。

但这样做会导致代码分工不明、程序结构混乱，程序员自己也会头昏眼花。所以用 ioctl 专门承载控制功能。

要记住：用户程序所做的只是通过命令码（cmd）告诉驱动它想做什么，至于怎么解释和执行这些命令，是驱动的事。

二、命令码：四段结构与 _IO 宏族 #

2.1 四段结构 #

命令码的组织是有讲究的。一定要做到命令和设备一一对应，否则会把正确的命令发给错误的设备、把错误的命令发给正确的设备……这些错误都会导致不可预料的事情，等程序员发现奇怪现象再来调试，会非常困难。

Linux 内核中命令码是这样定义的：

位域 设备类型（type） 序列号（nr） 方向（dir） 数据尺寸（size）
宽度 8 bit 8 bit 2 bit 14 bit

这样一个命令就变成了一个整数形式的命令码。

位域 位宽 位移 掩码 作用
nr 8 0 0x0000_00FF 序数，同一设备内区分不同命令
type 8 8 0x0000_00FF 幻数，标明设备类型
size 14 16 0x0000_3FFF 用户数据的大小
dir 2 30 0x0000_0003 传输方向

2.2 基础宏（include/asm/ioctl.h） #

c
#define _IOC_NRBITS      8        // 序数字段的字位宽度
#define _IOC_TYPEBITS    8        // 幻数字段的字位宽度
#define _IOC_SIZEBITS    14       // 大小字段的字位宽度
#define _IOC_DIRBITS     2        // 方向字段的字位宽度

#define _IOC_NRMASK     ((1 << _IOC_NRBITS)  - 1)   // 0x000000FF
#define _IOC_TYPEMASK   ((1 << _IOC_TYPEBITS)- 1)   // 0x000000FF
#define _IOC_SIZEMASK   ((1 << _IOC_SIZEBITS)- 1)   // 0x00003FFF
#define _IOC_DIRMASK    ((1 << _IOC_DIRBITS) - 1)   // 0x00000003

#define _IOC_NRSHIFT    0
#define _IOC_TYPESHIFT  (_IOC_NRSHIFT  + _IOC_NRBITS)    // 8
#define _IOC_SIZESHIFT  (_IOC_TYPESHIFT + _IOC_TYPEBITS) // 16
#define _IOC_DIRSHIFT   (_IOC_SIZESHIFT + _IOC_SIZEBITS) // 30

#define _IOC_NONE   0U   // 没有数据传输
#define _IOC_WRITE  1U   // 向设备写入数据，驱动必须从用户空间读入数据
#define _IOC_READ   2U   // 从设备读取数据，驱动必须向用户空间写入数据

#define _IOC(dir, type, nr, size) \
       (((dir)  << _IOC_DIRSHIFT)  | \
        ((type) << _IOC_TYPESHIFT) | \
        ((nr)   << _IOC_NRSHIFT)   | \
        ((size) << _IOC_SIZESHIFT))

2.3 构造命令：四个宏 #

宏 用途 方向
_IO(type, nr) 构造无参数的命令编号 无
_IOR(type, nr, size) 构造从驱动程序中读数据的命令编号 驱动 → 用户
_IOW(type, nr, size) 用于向驱动程序写数据命令 用户 → 驱动
_IOWR(type, nr, size) 用于双向传输 双向 
c
#define _IO(type, nr)          _IOC(_IOC_NONE, (type), (nr), 0)
#define _IOR(type, nr, size)   _IOC(_IOC_READ, (type), (nr), sizeof(size))
#define _IOW(type, nr, size)   _IOC(_IOC_WRITE, (type), (nr), sizeof(size))
#define _IOWR(type, nr, size)  _IOC(_IOC_READ | _IOC_WRITE, (type), (nr), sizeof(size))

方向是从驱动视角看的，这是最容易记反的地方：_IOR 表示驱动读设备把数据给用户；_IOW 表示驱动从用户取数据写进设备。

2.4 解析命令：四个解码宏 #

c
#define _IOC_DIR(nr)   (((nr) >> _IOC_DIRSHIFT)  & _IOC_DIRMASK)
#define _IOC_TYPE(nr)  (((nr) >> _IOC_TYPESHIFT) & _IOC_TYPEMASK)
#define _IOC_NR(nr)    (((nr) >> _IOC_NRSHIFT)   & _IOC_NRMASK)
#define _IOC_SIZE(nr)  (((nr) >> _IOC_SIZESHIFT) & _IOC_SIZEMASK)

另有几个便利宏等同使用：

c
#define IOC_IN        (_IOC_WRITE << _IOC_DIRSHIFT)
#define IOC_OUT       (_IOC_READ  << _IOC_DIRSHIFT)
#define IOC_INOUT     ((_IOC_WRITE | _IOC_READ) << _IOC_DIRSHIFT)
#define IOCSIZE_MASK  (_IOC_SIZEMASK << _IOC_SIZESHIFT)
#define IOCSIZE_SHIFT (_IOC_SIZESHIFT)

2.5 幻数（MAGIC_NUMBER） #

必须提一下"幻数"：幻数是一个字母，数据长度也是 8，用一个特定字母来标明设备类型，这和用一个数字是一样的，只是更加利于记忆和理解。

c
#define MAGIC_NUMBER  'k'      // 一个字母作为设备类型标识

不同的驱动必须用不同的幻数，否则两个驱动收到对方的命令码时会误判。

2.6 cmd 参数如何得出 #

cmd 在用户程序端由上述宏根据设备类型、序列号、传送方向、数据尺寸生成，这个整数通过系统调用传递到内核中的驱动；驱动再用解码宏从这个整数中得到各字段信息，然后通过 switch { case } 进行相应操作。

用户态：  #define BEEP_ON  _IO('k', 0)     →  生成整数 0x00006b00
                │  ioctl(fd, BEEP_ON)
                ▼
驱动：    switch(cmd) { case BEEP_ON: ... }   ← 同一个宏展开成同一个整数
                │
                └─ 需要时用 _IOC_TYPE/_IOC_NR/_IOC_DIR/_IOC_SIZE 拆字段

两端必须共用同一份头文件定义，否则命令码就对不上。

三、实现 ioctl 的三个技术环节 #

3.1 返回值 #

ioctl 的实现是根据命令执行的一个 switch 语句。当命令不能匹配任何一个设备所支持的命令时，通常返回 -EINVAL（非法参数）。

3.2 参数使用与校验 #

用户态 int ioctl(int fd, unsigned long cmd, ...) 里的 ... 就是传递的参数，再通过驱动原型里的 unsigned long arg 传入。

  • 如果 arg 是一个整数，可以直接使用；
  • 如果是指针，必须确保这个用户地址是有效的，使用之前要正确检查。

哪些需要检查、哪些不需要：

函数 是否需要手动 access_ok
copy_from_user / copy_to_user 内部已检查，不需要
get_user / put_user 内部已检查，不需要
__get_user / __put_user 需要手动检查

检查函数：

c
static inline int access_ok(int type, const void *addr, unsigned long size);
参数 含义
type VERIFY_READ 或 VERIFY_WRITE，表明要读用户内存还是写用户内存
addr 要操作的用户内存地址
size 操作的长度。若 ioctl 要从用户空间读一个整数，size 就等于 sizeof(int)

返回值是一个布尔值：1 表示成功（存取没问题），0 表示失败，此时 ioctl 应返回 -EFAULT。

新内核中 access_ok() 只剩两个参数（去掉了 type），改为 access_ok(addr, size)，方向检查由后续的 copy_*_user 完成。

3.3 命令操作 #

c
switch (cmd) {
case FIRST_CMD:
    /* ... */
    break;
default:
    return -EINVAL;
}

四、实例一：memdev（完整版） #

这是最完整的样板：头文件定义命令、驱动侧做类型校验与参数校验、应用侧调用。

4.1 memdev.h —— 命令定义 #

c
#ifndef _MEMDEV_H_
#define _MEMDEV_H_

/* mem 设备描述结构体 */
struct mem_dev {
    char *data;
    unsigned long size;
};

/* 定义幻数 */
#define MEMDEV_IOC_MAGIC   'k'

/* 定义命令 */
#define MEMDEV_IOCPRINT    _IO(MEMDEV_IOC_MAGIC, 1)
#define MEMDEV_IOCGETDATA  _IOR(MEMDEV_IOC_MAGIC, 2, int)
#define MEMDEV_IOCSETDATA  _IOW(MEMDEV_IOC_MAGIC, 3, int)
#define MEMDEV_IOC_MAXNR   3

#endif /* _MEMDEV_H_ */

4.2 memdev.c —— 驱动 #

c
static int mem_major = MEMDEV_MAJOR;
module_param(mem_major, int, S_IRUGO);

struct mem_dev *mem_devp;      /* 设备结构体指针 */
struct cdev cdev;

/* 文件打开函数 */
int mem_open(struct inode *inode, struct file *filp)
{
    struct mem_dev *dev;
    /* 获取次设备号 */
    int num = MINOR(inode->i_rdev);

    if (num >= MEMDEV_NR_DEVS)
        return -ENODEV;

    dev = &mem_devp[num];

    /* 将设备描述结构指针赋值给文件私有数据指针 */
    filp->private_data = dev;
    return 0;
}

/* 文件释放函数 */
int mem_release(struct inode *inode, struct file *filp)
{
    return 0;
}

/* IO 操作 */
int memdev_ioctl(struct inode *inode, struct file *filp, unsigned int cmd, unsigned long arg)
{
    int err = 0;
    int ret = 0;
    int ioarg = 0;

    /* 检测命令的有效性 */
    if (_IOC_TYPE(cmd) != MEMDEV_IOC_MAGIC)
        return -EINVAL;
    if (_IOC_NR(cmd) > MEMDEV_IOC_MAXNR)
        return -EINVAL;

    /* 根据命令类型，检测参数空间是否可以访问 */
    if (_IOC_DIR(cmd) & _IOC_READ)
        err = !access_ok(VERIFY_WRITE, (void *)arg, _IOC_SIZE(cmd));
    else if (_IOC_DIR(cmd) & _IOC_WRITE)
        err = !access_ok(VERIFY_READ, (void *)arg, _IOC_SIZE(cmd));
    if (err)
        return -EFAULT;

    /* 根据命令，执行相应的操作 */
    switch (cmd) {
    case MEMDEV_IOCPRINT:                       /* 打印当前设备信息 */
        printk("<--- CMD MEMDEV_IOCPRINT Done --->\n\n");
        break;

    case MEMDEV_IOCGETDATA:                     /* 获取参数 */
        ioarg = 1101;
        ret = __put_user(ioarg, (int *)arg);
        break;

    case MEMDEV_IOCSETDATA:                     /* 设置参数 */
        ret = __get_user(ioarg, (int *)arg);
        printk("<--- In Kernel MEMDEV_IOCSETDATA ioarg = %d --->\n\n", ioarg);
        break;

    default:
        return -EINVAL;
    }
    return ret;
}

/* 文件操作结构体 */
static const struct file_operations mem_fops = {
    .owner   = THIS_MODULE,
    .open    = mem_open,
    .release = mem_release,
    .ioctl   = memdev_ioctl,      /* 新内核应为 .unlocked_ioctl */
};

/* 设备驱动模块加载函数 */
static int memdev_init(void)
{
    int result;
    int i;
    dev_t devno = MKDEV(mem_major, 0);

    /* 静态申请设备号 */
    if (mem_major)
        result = register_chrdev_region(devno, 2, "memdev");
    else {  /* 动态分配设备号 */
        result = alloc_chrdev_region(&devno, 0, 2, "memdev");
        mem_major = MAJOR(devno);
    }
    if (result < 0)
        return result;

    /* 初始化 cdev 结构 */
    cdev_init(&cdev, &mem_fops);
    cdev.owner = THIS_MODULE;
    cdev.ops = &mem_fops;

    /* 注册字符设备 */
    cdev_add(&cdev, MKDEV(mem_major, 0), MEMDEV_NR_DEVS);

    /* 为设备描述结构分配内存 */
    mem_devp = kmalloc(MEMDEV_NR_DEVS * sizeof(struct mem_dev), GFP_KERNEL);
    if (!mem_devp) {  /* 申请失败 */
        result = -ENOMEM;
        goto fail_malloc;
    }
    memset(mem_devp, 0, sizeof(struct mem_dev));

    /* 为设备分配内存 */
    for (i = 0; i < MEMDEV_NR_DEVS; i++) {
        mem_devp[i].size = MEMDEV_SIZE;
        mem_devp[i].data = kmalloc(MEMDEV_SIZE, GFP_KERNEL);
        memset(mem_devp[i].data, 0, MEMDEV_SIZE);
    }

    return 0;

fail_malloc:
    unregister_chrdev_region(devno, 2);
    return result;
}

/* 模块卸载函数 */
static void memdev_exit(void)
{
    cdev_del(&cdev);                                  /* 注销设备 */
    kfree(mem_devp);                                  /* 释放设备结构体内存 */
    unregister_chrdev_region(MKDEV(mem_major, 0), 2); /* 释放设备号 */
}

MODULE_AUTHOR("David Xie");
MODULE_LICENSE("GPL");
module_init(memdev_init);
module_exit(memdev_exit);

素材原文 fail_malloc 分支里释放的是 unregister_chrdev_region(devno, 1)，而注册时 count 是 2，两者不一致会泄漏一个设备号。本篇已改为 2，与注册对称。

4.3 app-ioctl.c —— 应用程序 #

c
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <sys/ioctl.h>
#include "memdev.h"      /* 包含命令定义 */

int main(void)
{
    int fd = 0;
    int cmd;
    int arg = 0;

    /* 打开设备文件 */
    fd = open("/dev/memdev0", O_RDWR);
    if (fd < 0) {
        printf("Open Dev Mem0 Error!\n");
        return -1;
    }

    /* 调用命令 MEMDEV_IOCPRINT */
    printf("<--- Call MEMDEV_IOCPRINT --->\n");
    cmd = MEMDEV_IOCPRINT;
    if (ioctl(fd, cmd, &arg) < 0) {
        printf("Call cmd MEMDEV_IOCPRINT fail\n");
        return -1;
    }

    /* 调用命令 MEMDEV_IOCSETDATA */
    printf("<--- Call MEMDEV_IOCSETDATA --->\n");
    cmd = MEMDEV_IOCSETDATA;
    arg = 2007;
    if (ioctl(fd, cmd, &arg) < 0) {
        printf("Call cmd MEMDEV_IOCSETDATA fail\n");
        return -1;
    }

    /* 调用命令 MEMDEV_IOCGETDATA */
    printf("<--- Call MEMDEV_IOCGETDATA --->\n");
    cmd = MEMDEV_IOCGETDATA;
    if (ioctl(fd, cmd, &arg) < 0) {
        printf("Call cmd MEMDEV_IOCGETDATA fail\n");
        return -1;
    }

    printf("<--- In User Space MEMDEV_IOCGETDATA Get Data is %d --->\n\n", arg);

    close(fd);
    return 0;
}

这个例子里，MEMDEV_IOCSETDATA 把用户态的 2007 送进内核并 printk 出来；MEMDEV_IOCGETDATA 把内核里的 1101 送回用户态并打印。一进一出，把两个方向都验证到了。

五、实例二：beep（精简版） #

真实设备上更常见的是这种极简用法——命令不带数据，直接触发动作。

5.1 驱动侧 #

c
#define MAGIC_NUMBER  'k'
#define BEEP_ON       _IO(MAGIC_NUMBER, 0)
#define BEEP_OFF      _IO(MAGIC_NUMBER, 1)
#define BEEP_FREQ     _IO(MAGIC_NUMBER, 2)

#define BEEP_IN_FREQ  100000

static void beep_freq(unsigned long arg)
{
    writel(BEEP_IN_FREQ / arg,      timer_base + TCNTB0);
    writel(BEEP_IN_FREQ / (2 * arg), timer_base + TCMPB0);
}

static long beep_ioctl(struct file *filep, unsigned int cmd, unsigned long arg)
{
    switch (cmd) {
    case BEEP_ON:
        fs4412_beep_on();
        break;
    case BEEP_OFF:
        fs4412_beep_off();
        break;
    case BEEP_FREQ:
        beep_freq(arg);
        break;
    default:
        return -EINVAL;
    }
    return 0;
}

这里 BEEP_FREQ 用 _IO 而不是 _IOW，因为频率值是直接当整数传（ioctl(fd, BEEP_FREQ, 1000)），不是通过指针。用 _IOW 反而会误导读者以为要传指针。

5.2 应用侧 #

c
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <sys/ioctl.h>
#include <unistd.h>

#define MAGIC_NUMBER  'k'
#define BEEP_ON       _IO(MAGIC_NUMBER, 0)
#define BEEP_OFF      _IO(MAGIC_NUMBER, 1)
#define BEEP_FREQ     _IO(MAGIC_NUMBER, 2)

int main(void)
{
    int fd;

    fd = open("/dev/beep", O_RDWR);
    if (fd < 0) {
        perror("open fail \n");
        return -1;
    }

    ioctl(fd, BEEP_ON);
    sleep(6);
    ioctl(fd, BEEP_OFF);

    close(fd);
    return 0;
}

六、写 ioctl 的检查清单 #

检查项 要求
幻数唯一 每个驱动一个字母，不能与同系统其他驱动撞
两端头文件一致 用户态与驱动共用同一份 xxx.h 中的命令定义
方向别记反 _IOR = 驱动读给用户，_IOW = 用户写进驱动
指针参数必校验 用 __get_user/__put_user 前先 access_ok
校验 type 与 nr _IOC_TYPE(cmd) != MAGIC 或 _IOC_NR(cmd) > MAXNR 直接 -EINVAL
默认分支返回 default 必须 return -EINVAL，不能静默成功
用 unlocked_ioctl 新驱动不要用旧的 .ioctl 成员
