V4L2 是 Linux 内核的视频设备框架，为摄像头、视频输出、overlay 等设备提供一套统一的 ioctl 接口。用上它之后，同一份采集程序能驱动 UVC 摄像头、SoC 自带的 MIPI CSI 控制器、TV 采集卡，差别只体现在支持的格式和能力位上。

一、V4L2 管什么 #

V4L2（video for Linux 2）在 include/uapi/linux/videodev2.h 里定义了一整套数据结构和 ioctl 命令。

老文章常写 include/linux/videodev.h，那个头文件在 2.6.38 已随 V4L1 一起删除。现在只有 videodev2.h，路径在 uapi/ 下（用户态可见的 API 都在 uapi 里）。

它面向三类接口：

接口 干什么 典型设备
video capture 视频捕获 摄像头、采集卡
video output 视频输出，把图像序列编码成模拟视频信号送出去 电视输出芯片
video overlay 把采集的数据直接送到显示设备，不经过 CPU 搬运 老式带 overlay 的显示控制器

本篇只讲 capture。设备节点是 /dev/video0、/dev/video1……，本质上就是字符设备，open / close / ioctl / mmap 四件套。

采集数据有两条路：

方式 机制 适用
read 内核把一帧拷到用户 buffer 静态图片、低频采集
mmap + streaming 内核维护缓冲队列，映射到用户空间 连续视频

read 每帧都要跨空间拷一次，1080p 一帧 3MB，拷一次几毫秒，帧率上不去。mmap 只是建立页表映射，用户态直接读内核那几块 buffer，零拷贝。连续采集一律用 mmap。

设备支持哪种，由 VIDIOC_QUERYCAP 报告的 V4L2_CAP_READWRITE / V4L2_CAP_STREAMING 决定。

二、一次采集的完整流程 #

九步，顺序固定：

text
① open("/dev/video0")                      打开设备
② VIDIOC_QUERYCAP                          查能力，确认支持 capture + streaming
③ VIDIOC_ENUMINPUT / VIDIOC_S_INPUT        枚举并选择输入源
④ VIDIOC_ENUM_FMT                          确认支持目标像素格式
⑤ VIDIOC_S_FMT                             设置分辨率与像素格式
⑥ VIDIOC_REQBUFS                           向内核申请 N 块缓冲
⑦ VIDIOC_QUERYBUF + mmap                   逐块查信息并映射到用户空间
⑧ VIDIOC_QBUF × N + VIDIOC_STREAMON        全部入队，启动采集
   ┌──────────────────────────────────┐
   │ VIDIOC_DQBUF   取出一帧（有数据时返回）│
   │ 处理这一帧（存盘 / 转 RGB / 显示）     │
   │ VIDIOC_QBUF    把这帧还回去           │  ← 循环
   └──────────────────────────────────┘
⑨ VIDIOC_STREAMOFF + munmap + close        停止并释放

第 ⑧ 步之后是个永动的入队/出队循环。这一点是理解 streaming 模式的关键：内核手里始终握着「空 buffer 队列」，采集满一块就把它挪到「满 buffer 队列」；应用 DQBUF 从满队列摘一块，处理完 QBUF 还回空队列。队列里的 buffer 数量是不变的，永远是 REQBUFS 申请的那个数。

如果应用忘了 QBUF，空队列耗尽，采集就停了。这是「程序跑着跑着不出图了」最常见的原因。

三、查能力：VIDIOC_QUERYCAP #

c
struct v4l2_capability {
        __u8    driver[16];      /* 驱动名，如 "uvcvideo" */
        __u8    card[32];        /* 设备名 */
        __u8    bus_info[32];    /* 总线信息，如 "usb-s3c24xx-1" */
        __u32   version;         /* 内核版本，KERNEL_VERSION() 编码 */
        __u32   capabilities;    /* 设备能力位 */
        __u32   device_caps;     /* 当前这个节点的能力位（3.3+） */
        __u32   reserved[4];
};

c
int v4l2_querycap(int fd, struct v4l2_capability *cap)
{
    if (ioctl(fd, VIDIOC_QUERYCAP, cap) < 0) {
        perror("VIDIOC_QUERYCAP");
        return -1;
    }
    return 0;
}

能力位里要关心的：

标志 含义
V4L2_CAP_VIDEO_CAPTURE 支持视频捕获
V4L2_CAP_VIDEO_OUTPUT 支持视频输出
V4L2_CAP_VIDEO_OVERLAY 支持 overlay
V4L2_CAP_STREAMING 支持 mmap + 流式 IO
V4L2_CAP_READWRITE 支持 read / write
V4L2_CAP_DEVICE_CAPS device_caps 字段有效

这里有个 3.3 之后必须注意的分叉。 一个物理设备往往注册出多个 /dev/videoN 节点（比如一个 UVC 摄像头同时有采集节点和 metadata 节点），它们的 capabilities 是整个物理设备的能力合集，device_caps 才是当前这个节点的能力。

只查 capabilities 会误判：/dev/video1 是 metadata 节点，capabilities 里也有 V4L2_CAP_VIDEO_CAPTURE，但你对它做采集会失败。

正确写法：

c
cap.capabilities & V4L2_CAP_DEVICE_CAPS
    ? cap.device_caps          /* 新式：用节点能力 */
    : cap.capabilities;        /* 老驱动：只能退回设备能力 */

判定采集能力：

c
unsigned int caps = (cap.capabilities & V4L2_CAP_DEVICE_CAPS)
                    ? cap.device_caps : cap.capabilities;

if (!(caps & V4L2_CAP_VIDEO_CAPTURE)) {
    fprintf(stderr, "not a capture device\n");
    return -1;
}
if (!(caps & V4L2_CAP_STREAMING)) {
    fprintf(stderr, "no streaming support, fall back to read()\n");
}

四、选输入源：ENUMINPUT / S_INPUT #

一个设备可能有多个输入。比如 SoC 上的摄像头控制器，可能同时接了 MIPI CSI 和并口 DVP 两路，得选一路。

UVC 摄像头一般只有一路，这一步可以跳过。但只要设备有多路输入，就必须显式选。

c
struct v4l2_input {
    __u32        index;        /* 输入索引，用户空间填 */
    __u8         name[32];     /* 名字，驱动填 */
    __u32        type;         /* V4L2_INPUT_TYPE_TUNER / _CAMERA */
    __u32        audioset;     /* 关联的音频输入位掩码 */
    __u32        tuner;        /* 关联的调谐器索引 */
    v4l2_std_id  std;          /* 支持的视频标准（PAL/NTSC…） */
    __u32        status;       /* 输入状态位 */
    __u32        reserved[4];
};

index 是唯一一个由用户空间填的字段。驱动从 0 开始分配，索引必须连续，可以没有输入，但只要有输入，索引 0 一定存在。枚举到 EINVAL 就说明到末尾了。

c
int v4l2_enum_input(int fd, char *name_out)
{
    struct v4l2_input in;
    int found = 0;

    memset(&in, 0, sizeof(in));
    in.index = 0;
    while (ioctl(fd, VIDIOC_ENUMINPUT, &in) == 0) {
        printf("input %u: %s, std: 0x%08llx\n",
               in.index, in.name, (unsigned long long)in.std);
        if (in.index == 0) {
            strncpy(name_out, (char *)in.name, 31);
            found = 1;
        }
        in.index++;
    }
    return found ? 0 : -1;
}

status 字段值得看一眼：它是位掩码，每一位代表一种问题，比如没有电源、没有信号、没有同步锁。采集拿不到图的时候，先来这里确认输入源本身有没有信号。

选定输入：

c
struct v4l2_input in = { .index = 0 };
if (ioctl(fd, VIDIOC_S_INPUT, &in) < 0)
    perror("VIDIOC_S_INPUT");

驱动侧对应的是 vidioc_enum_input / vidioc_s_input 两个回调。vidioc_s_input 里驱动要真正切换硬件通路，索引非法返回 -EINVAL，硬件故障返回 -EIO。

五、协商格式：ENUM_FMT 与 S_FMT #

5.1 先问支持什么 #

c
struct v4l2_fmtdesc {
        __u32               index;
        enum v4l2_buf_type  type;
        __u32               flags;
        __u8                description[32];   /* 人读的描述 */
        __u32               pixelformat;       /* 四字符码，如 'YUYV' */
        __u32               reserved[4];
};

枚举方式和输入一样，index 从 0 递增直到 EINVAL：

c
int v4l2_enum_fmt(int fd, unsigned int want, enum v4l2_buf_type type)
{
    struct v4l2_fmtdesc desc;
    int found = 0;

    memset(&desc, 0, sizeof(desc));
    desc.type = type;
    desc.index = 0;

    while (ioctl(fd, VIDIOC_ENUM_FMT, &desc) == 0) {
        printf("pixelformat = %c%c%c%c, %s\n",
               desc.pixelformat & 0xFF,
               (desc.pixelformat >> 8) & 0xFF,
               (desc.pixelformat >> 16) & 0xFF,
               (desc.pixelformat >> 24) & 0xFF,
               desc.description);
        if (desc.pixelformat == want)
            found = 1;
        desc.index++;
    }
    return found ? 0 : -1;
}

pixelformat 是四字符码（FourCC），用小端序打包成 __u32。'YUYV' 展开就是 0x56595559，所以打印时要按字节拆。

常见格式：

FourCC 含义
V4L2_PIX_FMT_YUYV YUV 4:2:2 打包，每 4 字节两个像素
V4L2_PIX_FMT_UYVY / VYUY / YVYU 同上，字节序不同
V4L2_PIX_FMT_NV12 / NV21 YUV 4:2:0，Y 平面 + UV 交织平面
V4L2_PIX_FMT_YUV420 YUV 4:2:0 三平面
V4L2_PIX_FMT_RGB565 / RGB32 直接输出 RGB
V4L2_PIX_FMT_MJPEG / JPEG Motion-JPEG 压缩流
V4L2_PIX_FMT_H264 H.264 压缩流

UVC 摄像头几乎都支持 MJPEG 和 YUYV。如果带宽吃紧，选 MJPEG。它是压缩流，但要自己解码；YUYV 是裸数据，直接可用，代价是占用带宽大。在 USB 1.1（12Mbps）的板子上，320×240 YUYV@25fps 就要 3MB/s，勉强够，更高分辨率必须换 MJPEG。

5.2 再设格式 #

c
struct v4l2_format {
        enum v4l2_buf_type type;
        union {
                struct v4l2_pix_format          pix;    /* CAPTURE 时用这个 */
                struct v4l2_window              win;    /* OVERLAY */
                struct v4l2_vbi_format          vbi;
                struct v4l2_sliced_vbi_format   sliced;
                __u8    raw_data[200];
        } fmt;
};

struct v4l2_pix_format {
        __u32             width;
        __u32             height;
        __u32             pixelformat;
        enum v4l2_field   field;
        __u32             bytesperline;   /* 每行字节数，驱动可回填 */
        __u32             sizeimage;      /* 一帧总字节数 */
        enum v4l2_colorspace colorspace;
        __u32             priv;
};

fmt 是 union，哪个成员有效取决于 type。capture 用 fmt.pix。

c
int v4l2_s_fmt(int fd, unsigned int *width, unsigned int *height,
               unsigned int pixfmt, enum v4l2_buf_type type)
{
    struct v4l2_format fmt;

    memset(&fmt, 0, sizeof(fmt));
    fmt.type = type;
    fmt.fmt.pix.width       = *width;
    fmt.fmt.pix.height      = *height;
    fmt.fmt.pix.pixelformat = pixfmt;
    fmt.fmt.pix.field       = V4L2_FIELD_ANY;

    if (ioctl(fd, VIDIOC_S_FMT, &fmt) < 0) {
        perror("VIDIOC_S_FMT");
        return -1;
    }

    /* 关键：驱动可能改写了参数，必须回读 */
    *width  = fmt.fmt.pix.width;
    *height = fmt.fmt.pix.height;

    printf("actual: %ux%u, bytesperline=%u, sizeimage=%u\n",
           fmt.fmt.pix.width, fmt.fmt.pix.height,
           fmt.fmt.pix.bytesperline, fmt.fmt.pix.sizeimage);
    return 0;
}

S_FMT 是「请求」不是「命令」。 如果设备不支持你给的分辨率，内核不会报错，而是把结构体改写成它支持的最接近值并返回 0。所以：

  • 必须回读 width / height，后面的代码全部以回读值为准；
  • 申请 buffer 和计算行偏移时用 bytesperline，不要自己拿 width × bpp 算，驱动可能为了对齐在行尾填充。

field 填 V4L2_FIELD_ANY 让驱动决定隔行/逐行。摄像头发来的都是逐行，回读得到 V4L2_FIELD_NONE。

六、申请缓冲：VIDIOC_REQBUFS #

c
struct v4l2_requestbuffers {
        __u32                count;      /* 要几块，驱动可能改小 */
        enum v4l2_buf_type   type;
        enum v4l2_memory     memory;     /* MMAP / USERPTR / DMABUF */
        __u32                reserved[2];
};

c
struct v4l2_requestbuffers req;

memset(&req, 0, sizeof(req));
req.count  = 4;
req.type   = V4L2_BUF_TYPE_VIDEO_CAPTURE;
req.memory = V4L2_MEMORY_MMAP;

if (ioctl(fd, VIDIOC_REQBUFS, &req) < 0) {
    perror("VIDIOC_REQBUFS");
    return -1;
}

if (req.count < 2) {
    fprintf(stderr, "insufficient buffer memory, got %u\n", req.count);
    return -1;
}

count 是入参兼出参：你请求 4 块，内核如果内存不够会少给，返回实际数量。所以申请完必须检查 req.count，后面所有循环都以它为准。

memory 三种取值：

值 含义
V4L2_MEMORY_MMAP 内核分配，映射给用户态。最常用
V4L2_MEMORY_USERPTR 用户态自己 malloc，把指针给内核。省一次映射，但要求物理连续，多数设备不支持
V4L2_MEMORY_DMABUF 传 dma-buf fd，多设备共享同一块内存（零拷贝管线）

4 块是经验值。少于 2 块基本没法流式采集（一块在用、一块在填，没缓冲余量）；多了浪费内存，也不会更快。1080p YUYV 一块 4MB，4 块就是 16MB，嵌入式上要掂量。

七、查询与映射：QUERYBUF 与 mmap #

c
struct v4l2_buffer {
        __u32                   index;      /* 第几块，入参 */
        enum v4l2_buf_type      type;
        __u32                   bytesused;  /* 实际有效字节数 */
        __u32                   flags;
        enum v4l2_field         field;
        struct timeval          timestamp;  /* 采集时间戳 */
        struct v4l2_timecode    timecode;
        __u32                   sequence;   /* 帧序号，丢帧时跳变 */
        enum v4l2_memory        memory;
        union {
                __u32           offset;     /* MMAP 时：内核缓冲区偏移 */
                unsigned long   userptr;    /* USERPTR 时：用户指针 */
        } m;
        __u32                   length;     /* 这块的总大小 */
        __u32                   input;
        __u32                   reserved;
};

查询第 i 块并映射：

c
struct v4l2_buffer buf;
void *buffers[4];
size_t buf_len[4];

for (unsigned int i = 0; i < req.count; i++) {
    memset(&buf, 0, sizeof(buf));
    buf.type   = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    buf.memory = V4L2_MEMORY_MMAP;
    buf.index  = i;

    if (ioctl(fd, VIDIOC_QUERYBUF, &buf) < 0) {
        perror("VIDIOC_QUERYBUF");
        return -1;
    }

    buf_len[i] = buf.length;
    buffers[i] = mmap(NULL, buf.length, PROT_READ | PROT_WRITE,
                      MAP_SHARED, fd, buf.m.offset);
    if (buffers[i] == MAP_FAILED) {        /* 注意：不是 NULL，是 (void*)-1 */
        perror("mmap");
        return -1;
    }
}

两个必须说清的点：

一、mmap 的失败值是 MAP_FAILED，即 (void *)-1，不是 NULL。写成 if (!buffers[i]) 判不出来，后面拿 (void*)-1 去读就是段错误。

二、QUERYBUF 之前 buf 必须清零并填好 type / memory / index。 内核靠这三个值定位是哪块缓冲。只填 index 忘了 memory，memory 就是 0（V4L2_MEMORY_MMAP 恰好是 0），碰巧能过；但 DQBUF / QBUF 里同样别漏。

buf.m.offset 是这块缓冲在设备上的偏移，不是物理地址。mmap 的第四个参数 offset 要求页对齐，内核对齐好了才给出来。

映射之后，buffers[i] 就是这一帧数据的用户态地址，buf_len[i] 是它的大小。读它就是读图像，没有任何拷贝。

八、入队、启动、取帧：QBUF / STREAMON / DQBUF #

8.1 全部入队 #

c
for (unsigned int i = 0; i < req.count; i++) {
    memset(&buf, 0, sizeof(buf));
    buf.type   = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    buf.memory = V4L2_MEMORY_MMAP;
    buf.index  = i;

    if (ioctl(fd, VIDIOC_QBUF, &buf) < 0) {
        perror("VIDIOC_QBUF");
        return -1;
    }
}

这一步把 N 块空缓冲全部交给内核。STREAMON 之前必须全部入队，否则内核凑不出一个完整的队列。

8.2 启动采集 #

c
enum v4l2_buf_type type = V4L2_BUF_TYPE_VIDEO_CAPTURE;

if (ioctl(fd, VIDIOC_STREAMON, &type) < 0) {
    perror("VIDIOC_STREAMON");
    return -1;
}

注意参数是枚举的地址，不是枚举值本身。传值进去会返回 -EINVAL，这个坑很多人踩。

8.3 取一帧 #

c
int v4l2_wait_frame(int fd, int timeout_ms)
{
    struct pollfd pfd = { .fd = fd, .events = POLLIN };

    int ret = poll(&pfd, 1, timeout_ms);
    if (ret < 0) {
        if (errno == EINTR)
            return -2;                 /* 被信号打断，不是错误 */
        perror("poll");
        return -1;
    }
    if (ret == 0) {
        fprintf(stderr, "timeout waiting for frame\n");
        return -1;
    }
    if (pfd.revents & (POLLERR | POLLNVAL)) {
        fprintf(stderr, "poll error on video fd\n");
        return -1;
    }
    return 0;
}

int v4l2_dqbuf(int fd, struct v4l2_buffer *buf)
{
    memset(buf, 0, sizeof(*buf));
    buf->type   = V4L2_BUF_TYPE_VIDEO_CAPTURE;
    buf->memory = V4L2_MEMORY_MMAP;

    if (ioctl(fd, VIDIOC_DQBUF, buf) < 0) {
        if (errno == EAGAIN)
            return -2;                 /* 非阻塞模式，暂时没帧 */
        perror("VIDIOC_DQBUF");
        return -1;
    }
    return 0;
}

DQBUF 是阻塞的（除非 open 时加了 O_NONBLOCK）。实践中都先 poll 再 DQBUF。poll 能设超时，DQBUF 不能；设备一断线，裸 DQBUF 会永久挂住。

回来后从 buf.index 找到对应缓冲：

c
void *frame = buffers[buf.index];
size_t frame_len = buf.length;      /* 或 buf.bytesused */

buf.bytesused 是实际有效字节数，buf.length 是这块缓冲的总容量。对未压缩格式两者相等；对 MJPEG 这类变长压缩流，必须用 bytesused，length 是容量上限。

8.4 处理完还回去 #

c
/* 处理完这一帧，必须还回去，否则空队列耗尽，采集停止 */
if (ioctl(fd, VIDIOC_QBUF, &buf) < 0)
    perror("VIDIOC_QBUF (requeue)");

buf 原样传回就行，index / type / memory 都还在。这是最容易漏的一步，也是「采了几帧就不动了」的直接原因。

buf.sequence 是帧序号，连续 +1 表示没丢帧。如果它跳变，说明应用处理太慢或者 poll 超时，内核那边轮转跳过了。排查卡顿先看这个字段。

8.5 停止与释放 #

c
enum v4l2_buf_type type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
ioctl(fd, VIDIOC_STREAMOFF, &type);         /* 停止采集 */

for (unsigned int i = 0; i < req.count; i++)
    munmap(buffers[i], buf_len[i]);          /* 解除映射 */

close(fd);

顺序不能颠倒：先 STREAMOFF 再 munmap。还在 streaming 时解除映射，内核可能正在往那块地址 DMA，结果是随机崩溃或者显示花屏。

严格说，munmap 不等于 REQBUFS(0)，内核侧缓冲区要等 close(fd) 才释放。要提前还给内核，可以再调一次 VIDIOC_REQBUFS 且 count = 0。

九、完整控制流 #

把前面几节串成一个函数，这就是任何 V4L2 采集程序的骨架：

c
struct capture_ctx {
    int fd;
    unsigned int width, height;
    unsigned int pixfmt;
    unsigned int nbufs;
    void *buffers[8];
    size_t buf_len[8];
};

static int capture_start(struct capture_ctx *ctx, const char *dev)
{
    struct v4l2_capability cap;
    struct v4l2_format fmt;
    struct v4l2_requestbuffers req;
    struct v4l2_buffer buf;
    unsigned int caps, i;
    enum v4l2_buf_type type = V4L2_BUF_TYPE_VIDEO_CAPTURE;

    /* ① 打开 */
    ctx->fd = open(dev, O_RDWR);
    if (ctx->fd < 0) {
        perror(dev);
        return -1;
    }

    /* ② 查能力 */
    memset(&cap, 0, sizeof(cap));
    if (ioctl(ctx->fd, VIDIOC_QUERYCAP, &cap) < 0) {
        perror("VIDIOC_QUERYCAP");
        goto fail_fd;
    }
    caps = (cap.capabilities & V4L2_CAP_DEVICE_CAPS)
           ? cap.device_caps : cap.capabilities;

    if (!(caps & V4L2_CAP_VIDEO_CAPTURE) || !(caps & V4L2_CAP_STREAMING)) {
        fprintf(stderr, "%s: not a streaming capture device\n", dev);
        goto fail_fd;
    }
    printf("%s: %s / %s\n", dev, cap.driver, cap.card);

    /* ③ 选输入（设备只有一路时可跳过） */
    {
        struct v4l2_input in = { .index = 0 };
        ioctl(ctx->fd, VIDIOC_S_INPUT, &in);   /* 失败不致命 */
    }

    /* ④⑤ 协商格式 */
    if (v4l2_enum_fmt(ctx->fd, ctx->pixfmt, type) < 0) {
        fprintf(stderr, "pixel format not supported\n");
        goto fail_fd;
    }
    if (v4l2_s_fmt(ctx->fd, &ctx->width, &ctx->height, ctx->pixfmt, type) < 0)
        goto fail_fd;

    /* ⑥ 申请缓冲 */
    memset(&req, 0, sizeof(req));
    req.count  = ctx->nbufs;
    req.type   = type;
    req.memory = V4L2_MEMORY_MMAP;
    if (ioctl(ctx->fd, VIDIOC_REQBUFS, &req) < 0) {
        perror("VIDIOC_REQBUFS");
        goto fail_fd;
    }
    if (req.count < 2) {
        fprintf(stderr, "insufficient buffers: %u\n", req.count);
        goto fail_fd;
    }
    ctx->nbufs = req.count;

    /* ⑦ 查询并映射 */
    for (i = 0; i < ctx->nbufs; i++) {
        memset(&buf, 0, sizeof(buf));
        buf.type = type;
        buf.memory = V4L2_MEMORY_MMAP;
        buf.index = i;
        if (ioctl(ctx->fd, VIDIOC_QUERYBUF, &buf) < 0) {
            perror("VIDIOC_QUERYBUF");
            goto fail_bufs;
        }
        ctx->buf_len[i] = buf.length;
        ctx->buffers[i] = mmap(NULL, buf.length, PROT_READ | PROT_WRITE,
                               MAP_SHARED, ctx->fd, buf.m.offset);
        if (ctx->buffers[i] == MAP_FAILED) {
            perror("mmap");
            goto fail_maps;
        }
    }

    /* ⑧ 全部入队后启动 */
    for (i = 0; i < ctx->nbufs; i++) {
        memset(&buf, 0, sizeof(buf));
        buf.type = type;
        buf.memory = V4L2_MEMORY_MMAP;
        buf.index = i;
        if (ioctl(ctx->fd, VIDIOC_QBUF, &buf) < 0) {
            perror("VIDIOC_QBUF");
            goto fail_maps;
        }
    }
    if (ioctl(ctx->fd, VIDIOC_STREAMON, &type) < 0) {
        perror("VIDIOC_STREAMON");
        goto fail_maps;
    }
    return 0;

fail_maps:
    while (i--)
        munmap(ctx->buffers[i], ctx->buf_len[i]);
fail_bufs:
fail_fd:
    close(ctx->fd);
    ctx->fd = -1;
    return -1;
}

采集循环：

c
for (;;) {
    struct v4l2_buffer buf;
    int ret;

    ret = v4l2_wait_frame(ctx.fd, 3000);
    if (ret == -2) continue;              /* 被信号打断 */
    if (ret < 0)   break;

    ret = v4l2_dqbuf(ctx.fd, &buf);
    if (ret == -2) continue;              /* 暂时没帧 */
    if (ret < 0)   break;

    handle_frame(ctx.buffers[buf.index], buf.bytesused);

    /* 这一行漏了，采集就会停在 nbufs 帧之后 */
    if (ioctl(ctx.fd, VIDIOC_QBUF, &buf) < 0) {
        perror("requeue");
        break;
    }
}

十、易错点 #

现象 原因 处理
采了几帧就不动了 忘了把 DQBUF 取出的缓冲 QBUF 回去 循环里补上
DQBUF 永久阻塞 没先 poll，或设备已断线 一律 poll + 超时
mmap 判断失效 用 !ptr 判 MAP_FAILED 判 == MAP_FAILED
STREAMON 返回 EINVAL 传了枚举值而不是枚举地址 ioctl(fd, VIDIOC_STREAMON, &type)
分辨率不是自己设的值 S_FMT 会静默改写 回读 width/height
画面错位、花屏 自己用 width × bpp 算行宽 用回读的 bytesperline
metadata 节点也能查能力但采不了 只看了 capabilities 优先用 device_caps
req.count 比申请的小 内核内存不够 以返回值为准，检查是否 < 2
streaming 中 munmap 崩溃 顺序错了 先 STREAMOFF
MJPEG 帧读到半截 用了 length 而不是 bytesused 压缩流用 bytesused
帧率上不去 用了 read 模式 改 mmap + streaming
USB 带宽不够导致掉帧 YUYV 裸数据太大 改 MJPEG 或降分辨率

V4L2 的 ioctl 有几十个，本篇只覆盖采集主链路。音频、调参（VIDIOC_S_CTRL）、裁剪（VIDIOC_S_CROP）、多平面（V4L2_BUF_TYPE_VIDEO_CAPTURE_MPLANE，MIPI CSI 常用）这些不在主链路上，需要时查 Documentation/userspace-api/media/v4l/。
