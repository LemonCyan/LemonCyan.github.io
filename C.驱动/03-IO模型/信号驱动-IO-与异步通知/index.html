阻塞和非阻塞访问、poll() 提供了较多解决设备访问的机制，但如果有了异步通知，整套机制就更加完善了。

一、异步通知是什么 #

异步通知的意思是：一旦设备就绪，就主动通知应用程序。这样应用程序根本不需要查询设备状态。

这一点非常类似于硬件上"中断"的概念，比较准确的称谓是"信号驱动的异步 I/O"。信号是在软件层次上对中断机制的一种模拟——一个进程收到一个信号与处理器收到一个中断请求，在原理上可以说是一样的。信号是异步的，一个进程不必通过任何操作等待信号的到达，事实上，进程也不知道信号到底什么时候到达。

1.1 四种方式的关系 #

阻塞 I/O      等待设备可访问后再访问        （被动等）
非阻塞 I/O    查询设备是否可访问            （主动问）
poll/select   同时查询多个设备              （批量问）
异步通知      设备通知自身可访问            （设备主动说）

由此可见，这几种 I/O 方式可以互为补充：

方式 发起方 触发时机
阻塞 应用 应用调用 read 时
非阻塞 应用（查询） 应用调用 read 时
多路复用 应用（查询） 应用调用 poll 时
异步通知 驱动 设备就绪时

1.2 作用与影响 #

  • 阻塞方式下，应用程序无需轮询设备是否可以访问；
  • 非阻塞方式下，由中断进行通知；
  • 异步通知则是由驱动发起，主动通知应用程序。

二、编程模型 #

2.1 Linux 信号 #

Linux 系统中，异步通知使用信号来实现。

c
void (*signal(int signum, void (*handler)(int)))(int);

这个原型比较难理解，可以分解为：

c
typedef void (*sighandler_t)(int);
sighandler_t signal(int signum, sighandler_t handler);

第一个参数是指定信号的值，第二个参数是指定针对前面信号的处理函数。

2.2 应用端：捕获信号 #

三项准备工作，顺序很重要：

c
/* 1. 设置信号处理函数 */
signal(SIGIO, sigterm_handler);

/* 2. 告诉内核：这个信号的接收者是我 */
fcntl(STDIN_FILENO, F_SETOWN, getpid());

/* 3. 开启 FASYNC，让驱动知道"这个文件要异步通知" */
oflags = fcntl(STDIN_FILENO, F_GETFL);
fcntl(STDIN_FILENO, F_SETFL, oflags | FASYNC);

完整示例：

c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <fcntl.h>

#define MAX_LEN 64

/* 信号处理函数：信号到达时读走数据 */
void sigterm_handler(int signo)
{
    char data[MAX_LEN];
    int len;

    len = read(STDIN_FILENO, &data, MAX_LEN);
    data[len] = 0;
    printf("Input available: %s\n", data);
    exit(0);
}

int main(void)
{
    int oflags;

    signal(SIGIO, sigterm_handler);
    fcntl(STDIN_FILENO, F_SETOWN, getpid());

    oflags = fcntl(STDIN_FILENO, F_GETFL);
    fcntl(STDIN_FILENO, F_SETFL, oflags | FASYNC);

    /* 建立死循环，防止程序结束 */
    while (1)
        ;

    return 0;
}

素材原文此处有三处笔误：fctcl（应为 fcntl）、whlie(1)（应为 while(1)）、main() 无返回类型、return ; 无返回值。已修正。

三、驱动端：三项工作 #

为了使设备支持异步通知机制，驱动程序中涉及以下三项工作：

# 工作 由谁完成
1 支持 F_SETOWN 命令，能在这个控制命令处理中设置 filp->f_owner 为对应的进程 ID 已由内核完成，驱动无须处理
2 支持 F_SETFL 命令处理，每当 FASYNC 标志改变时，驱动中的 fasync() 函数得以执行 驱动必须实现 fasync()
3 在设备资源可获得时，调用 kill_fasync() 函数激发相应的信号 驱动必须调用

也就是说，驱动实际只做第 2、3 项。设备驱动中异步通知编程比较简单，主要用到一个结构体 + 两个函数：

c
/* 一个结构体 */
struct fasync_struct;

/* 两个函数 */
int  fasync_helper(int fd, struct file *filp, int mode, struct fasync_struct **fa);
void kill_fasync(struct fasync_struct **fa, int sig, int band);

3.1 挂到设备结构体上 #

和其他结构体指针一样，把 fasync_struct 指针放进设备结构体：

c
struct xxx_dev {
    struct cdev cdev;
    /* ... */
    struct fasync_struct *async_queue;   /* 异步结构体指针 */
};

3.2 fasync 方法模板 #

在驱动的 fasync() 函数中，只需简单地把该函数的 3 个参数以及 fasync_struct 结构体指针的指针作为第四个参数传入 fasync_helper()：

c
static int xxx_fasync(int fd, struct file *filp, int mode)
{
    struct xxx_dev *dev = filp->private_data;

    return fasync_helper(fd, filp, mode, &dev->async_queue);
}

fasync_helper() 会自己维护那个链表——你不需要 allocate、不需要 free、不需要加锁。mode 为真时加入链表，为假时移出。

3.3 在资源可获得时发信号 #

在设备资源可获得时调用 kill_fasync() 释放 SIGIO 信号。第三个参数：可读时为 POLL_IN，可写时为 POLL_OUT。

c
static ssize_t xxx_write(struct file *filp, const char __user *buf,
                         size_t count, loff_t *ppos)
{
    struct xxx_dev *dev = filp->private_data;
    /* ... 写入数据 ... */

    if (dev->async_queue)
        kill_fasync(&dev->async_queue, SIGIO, POLL_IN);

    return count;
}

注意 if (dev->async_queue) 这个判断——如果没有任何进程启用异步通知，这个指针是 NULL，kill_fasync 虽然能处理 NULL，但显式判断更清晰、也省一次函数调用。

3.4 release 里必须注销 #

最后在文件关闭时，必须将文件从异步通知列表中删除：

c
static int xxx_release(struct inode *inode, struct file *filp)
{
    xxx_fasync(-1, filp, 0);    /* 第三个参数 0 表示移出 */

    /* ... 其他清理 ... */
    return 0;
}

这是最容易漏的一步。忘了它，进程退出后 async_queue 里还挂着一条指向已释放 struct file 的记录，下次 kill_fasync 就会访问已释放内存——典型 use-after-free，且往往要等到几天后随机崩一次才暴露。

3.5 file_operations 中的位置 #

c
static struct file_operations xxx_ops = {
    .owner  = THIS_MODULE,
    .open   = xxx_open,
    .release = xxx_release,
    .read   = xxx_read,
    .write  = xxx_write,
    .fasync = xxx_fasync,      /* ← 异步通知的入口 */
};

.fasync 是内核在 FASYNC 标志变化时回调的钩子。它的签名是：

c
int (*fasync)(int, struct file *, int);

四、完整实例 #

4.1 驱动 hello.c #

c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>
#include <linux/fcntl.h>

static int major = 250;
static int minor = 0;
static dev_t devno;
static struct class *cls;
static struct device *test_device;

static char temp[64] = {0};
static struct fasync_struct *fasync;

static int hello_open(struct inode *inode, struct file *filep)
{
    return 0;
}

static int hello_release(struct inode *inode, struct file *filep)
{
    hello_fasync(-1, filep, 0);     /* 从异步通知链表移除 */
    return 0;
}

static ssize_t hello_read(struct file *filep, char __user *buf, size_t len, loff_t *pos)
{
    if (len > 64)
        len = 64;

    if (copy_to_user(buf, temp, len))
        return -EFAULT;

    return len;
}

static ssize_t hello_write(struct file *filep, const char __user *buf,
                           size_t len, loff_t *pos)
{
    if (len > 64)
        len = 64;

    if (copy_from_user(temp, buf, len))
        return -EFAULT;

    printk("write %s\n", temp);

    kill_fasync(&fasync, SIGIO, POLL_IN);   /* 通知应用层"有数据了" */
    return len;
}

static int hello_fasync(int fd, struct file *file, int on)
{
    return fasync_helper(fd, file, on, &fasync);
}

static struct file_operations hello_ops = {
    .open    = hello_open,
    .release = hello_release,
    .read    = hello_read,
    .write   = hello_write,
    .fasync  = hello_fasync,
};

static int hello_init(void)
{
    int ret;

    devno = MKDEV(major, minor);
    ret = register_chrdev(major, "hello", &hello_ops);

    cls = class_create(THIS_MODULE, "myclass");
    if (IS_ERR(cls)) {
        unregister_chrdev(major, "hello");
        return -EBUSY;
    }

    test_device = device_create(cls, NULL, devno, NULL, "hello");
    if (IS_ERR(test_device)) {
        class_destroy(cls);
        unregister_chrdev(major, "hello");
        return -EBUSY;
    }
    return 0;
}

static void hello_exit(void)
{
    device_destroy(cls, devno);
    class_destroy(cls);
    unregister_chrdev(major, "hello");
    printk("hello_exit\n");
}

MODULE_LICENSE("GPL");
module_init(hello_init);
module_exit(hello_exit);

注意：素材原文的 hello_ops 漏写了 .fasync = hello_fasync——那样内核永远不会调用 fasync_helper，应用侧的 F_SETFL|FASYNC 不会生效，kill_fasync 时 fasync 链表始终为空，信号永远发不出去。本篇已补上。另外原文 hello_fasync 定义在 hello_ops 之后却没写前置声明，本篇调整了顺序。

4.2 应用 test.c #

c
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>
#include <signal.h>

static int fd, len;
static char buf[64] = {0};

void func(int signo)
{
    printf("signo %d\n", signo);
    read(fd, buf, 64);
    printf("buf=%s\n", buf);
}

int main(void)
{
    int flage, i = 0;

    fd = open("/dev/hello", O_RDWR);
    if (fd < 0) {
        perror("open fail\n");
        return -1;
    }

    fcntl(fd, F_SETOWN, getpid());
    flage = fcntl(fd, F_GETFL);
    fcntl(fd, F_SETFL, flage | FASYNC);

    signal(SIGIO, func);

    while (1) {
        sleep(1);
        printf("%d\n", i++);
    }

    close(fd);
    return 0;
}

运行后：应用每秒打印一个递增数字（说明它没有阻塞），另一个终端执行 echo hi > /dev/hello，应用立刻在信号处理函数里打印 signo 29 和 buf=hi。

五、几个容易踩的坑 #

5.1 顺序：先 F_SETOWN 还是先 FASYNC #

必须先 F_SETOWN 再 F_SETFL | FASYNC。反过来，内核可能在知道"该通知谁"之前就已经走了 fasync()，f_owner 尚未设置，信号发不出去。

标准顺序：signal → F_SETOWN → F_SETFL|FASYNC。

5.2 信号会丢 #

SIGIO 默认是非实时信号（编号 29）。非实时信号不排队——如果信号到来时前一个 SIGIO 还没被处理完，第二次到来的信号会被丢弃。

这意味着：短时间内写入多次，处理函数可能只被调用一次。异步通知只保证"有一次通知"，不保证"每次都通知"。需要不丢事件时，应该在信号处理函数里循环读干净，或者改用实时信号 / sigqueue。

5.3 信号处理函数里只能调 async-signal-safe 函数 #

printf、malloc 都不是 async-signal-safe 的。上面示例里在 handler 中用 printf 在工程上是不严谨的——它可能在 printf 内部持锁时被信号打断而死锁。生产代码应改用 write(STDOUT_FILENO, ...) 或只置一个 volatile sig_atomic_t 标志位，主循环里再处理。

5.4 中断上下文可以 kill_fasync #

kill_fasync() 内部只操作链表和一个唤醒，可以在中断上下文调用。这正是它能在硬件中断里直接通知应用的原因——比唤醒等待队列更"轻"。

六、四种方式怎么选 #

场景 推荐
单设备、应用本来就一直等着 阻塞 I/O
单设备、应用要兼顾别的事 非阻塞 + 自己轮询
多设备同时等 poll/select
事件稀疏、应用主循环不能被打断 异步通知（SIGIO）
一次通知不能丢 异步通知 + 内部计数器，或退回 poll
需要内核完全代办、连拷贝都不等 Posix AIO（aio_read）
